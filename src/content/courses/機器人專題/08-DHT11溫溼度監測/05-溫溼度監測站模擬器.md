```game-js
const temp = document.getElementById('temp');
const humidity = document.getElementById('humidity');
const interval = document.getElementById('interval');
const toggle = document.getElementById('toggle');
const status = document.getElementById('status');
let timer = null;

function renderRow(id, text) {
  const row = document.getElementById(id);
  row.setAttribute('aria-label', text);
  const cells = Array.from(text.slice(0, 16).padEnd(16), (character) => {
    const cell = document.createElement('span');
    cell.setAttribute('aria-hidden', 'true');
    cell.textContent = character;
    return cell;
  });
  row.replaceChildren(...cells);
}

renderRow('lcd-temp', 'T: -- C');
renderRow('lcd-humidity', 'H: -- % RH');

function sample() {
  renderRow('lcd-temp', 'T: ' + Number(temp.value).toFixed(1) + ' C');
  renderRow('lcd-humidity', 'H: ' + Number(humidity.value).toFixed(1) + ' % RH');
}

function startSampling() {
  timer = setInterval(sample, Number(interval.value));
  status.textContent = '取樣中，每 ' + Number(interval.value) / 1000 + ' 秒更新';
}

for (const input of [temp, humidity]) {
  input.addEventListener('input', () => {
    document.getElementById(input.id + '-value').textContent = input.value;
  });
}

toggle.addEventListener('click', () => {
  if (timer === null) {
    startSampling();
    toggle.textContent = '停止';
    return;
  }
  clearInterval(timer);
  timer = null;
  status.textContent = '已停止';
  toggle.textContent = '開始';
});

interval.addEventListener('change', () => {
  if (timer === null) return;
  clearInterval(timer);
  startSampling();
});
```
