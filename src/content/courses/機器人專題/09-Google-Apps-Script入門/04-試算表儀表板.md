## 把原始紀錄和畫面分開

保留「溫溼度紀錄」當資料來源，再新增「儀表板」分頁。先確認資料表至少有兩筆資料。若需要第二筆，可以將上一節的合法測試資料改成 28 °C 和 65 % RH 再送一次。

1. 在儀表板的 A1、B1、C1 依序填「最近更新時間」、「最新溫度（°C）」、「最新相對溼度（% RH）」。
2. 把下面三條公式依序貼到 A2、B2、C2，每格一條。三條公式都從「溫溼度紀錄」的 A 欄由下往上找最後一格有時間的紀錄，再取出同一列的時間、溫度和溼度。資料表中間有空列也能跳過。

```text
=XLOOKUP(TRUE,ARRAYFORMULA('溫溼度紀錄'!A2:A<>""),'溫溼度紀錄'!A2:A,"",0,-1)
=XLOOKUP(TRUE,ARRAYFORMULA('溫溼度紀錄'!A2:A<>""),'溫溼度紀錄'!B2:B,"",0,-1)
=XLOOKUP(TRUE,ARRAYFORMULA('溫溼度紀錄'!A2:A<>""),'溫溼度紀錄'!C2:C,"",0,-1)
```

如果 A2 顯示一串數字，選取 A2，點「格式 → 數字 → 日期時間」。公式回傳的是試算表裡的日期時間值，可以依需要調整顯示格式。

3. 在資料分頁選 A、B 兩欄，插入折線圖。確認 X 軸是時間，數列是溫度，第一列當標題。將圖表剪下並貼到儀表板，再確認資料範圍仍指向「溫溼度紀錄」的整個 A、B 欄。
4. 再送一筆不同的測試值，確認紀錄新增一列、最新數值更新，折線圖也加入新的資料點。整欄範圍可以讓後續新增的紀錄進入圖表。

下章會用 ESP32 送資料，沿用這個儀表板驗收。

## 完成範例｜試算表會長這樣

點選 A2、B2 或 C2，可以在公式列看到各欄的公式。切換到「溫溼度紀錄」看看原始資料，再按「模擬收到一筆」並回到「儀表板」，核對最新數字和折線圖的最後一點。

```game-js
const byId = (id) => document.getElementById(id);
const readings = [[25, 60], [27, 63], [26, 61], [28, 65]];
const sampleCycle = [[29, 67], [27, 64], [30, 69], [26, 62]];
let selectedCell = 'A2';
let activeSheet = 'dashboard';

function timeLabel(index) {
  const minutes = 9 * 60 + index;
  const hour = String(Math.floor(minutes / 60) % 24).padStart(2, '0');
  return hour + ':' + String(minutes % 60).padStart(2, '0');
}

function selectCell(cell) {
  selectedCell = cell;
  const column = cell[0];
  const range = "'溫溼度紀錄'!" + column + '2:' + column;
  byId('address').textContent = cell;
  byId('formula-text').textContent = '=XLOOKUP(TRUE,ARRAYFORMULA(' + "'溫溼度紀錄'!A2:A" + '<>""),' + range + ',"",0,-1)';
  for (const address of ['A2', 'B2', 'C2']) {
    byId(address).setAttribute('aria-pressed', String(address === cell));
  }
}

function showSheet(sheet) {
  activeSheet = sheet;
  for (const name of ['dashboard', 'log']) {
    byId(name).hidden = name !== sheet;
    byId(name + '-tab').setAttribute('aria-pressed', String(name === sheet));
  }
  if (sheet === 'dashboard') {
    selectCell(selectedCell);
  } else {
    byId('address').textContent = 'A1';
    byId('formula-text').textContent = '時間';
  }
}

function drawChart() {
  const chart = byId('chart');
  chart.replaceChildren();
  function add(tag, attributes, label) {
    const element = document.createElementNS('http://www.w3.org/2000/svg', tag);
    for (const [key, value] of Object.entries(attributes)) element.setAttribute(key, value);
    if (label !== undefined) element.textContent = label;
    chart.append(element);
  }
  const recent = readings.slice(-8);
  const x = (index) => 49 + index * 398 / (recent.length - 1);
  const y = (temperature) => 172 - (temperature - 24) * 20;
  for (const value of [24, 26, 28, 30, 32]) {
    add('line', { x1: 49, x2: 447, y1: y(value), y2: y(value), stroke: '#dadce0' });
    add('text', { x: 40, y: y(value) + 4, 'text-anchor': 'end', 'font-size': 12, fill: '#5f6368' }, value + '°');
  }
  add('polyline', { points: recent.map(([temp], i) => x(i) + ',' + y(temp)).join(' '), stroke: '#4285f4', 'stroke-width': 2.5, fill: 'none' });
  recent.forEach(([temp], i) => {
    add('circle', { cx: x(i), cy: y(temp), r: 3.5, fill: '#4285f4' });
    add('text', { x: x(i), y: 195, 'text-anchor': 'middle', 'font-size': 12, fill: '#5f6368' }, timeLabel(readings.length - recent.length + i));
  });
  chart.setAttribute('aria-label', '最近溫度（°C）：' + recent.map(([temp]) => temp).join('、'));
}

function appendRow(number, values) {
  const row = byId('records').insertRow();
  const heading = document.createElement('th');
  heading.scope = 'row';
  heading.textContent = number;
  row.append(heading);
  for (const value of values) row.insertCell().textContent = value;
}

function refreshDashboard() {
  const [temperature, humidity] = readings.at(-1);
  byId('A2').textContent = timeLabel(readings.length - 1);
  byId('B2').textContent = temperature;
  byId('C2').textContent = humidity;
  drawChart();
}

for (const address of ['A2', 'B2', 'C2']) byId(address).addEventListener('click', () => selectCell(address));
for (const sheet of ['dashboard', 'log']) byId(sheet + '-tab').addEventListener('click', () => showSheet(sheet));

byId('add').addEventListener('click', () => {
  const index = readings.length;
  const reading = sampleCycle[(index - 4) % sampleCycle.length];
  readings.push(reading);
  appendRow(index + 2, [timeLabel(index), ...reading]);
  refreshDashboard();
  if (activeSheet === 'log') byId('log').scrollTop = byId('log').scrollHeight;
  byId('status').textContent = '第 ' + (index + 2) + ' 列已新增，最新讀值為 ' + reading[0] + ' °C、' + reading[1] + ' % RH。折線圖最多顯示最近 8 筆。';
});

appendRow(1, ['時間', '溫度（°C）', '相對溼度（% RH）']);
readings.forEach((reading, index) => appendRow(index + 2, [timeLabel(index), ...reading]));
selectCell(selectedCell);
refreshDashboard();
```
