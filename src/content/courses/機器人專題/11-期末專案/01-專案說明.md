```game-js
const slides = [...document.querySelectorAll('.slide')];
const previous = document.getElementById('previous');
const next = document.getElementById('next');
const position = document.getElementById('position');
let index = 0;

function showSlide(target, moveFocus = false) {
  index = Math.max(0, Math.min(slides.length - 1, target));
  slides.forEach((slide, number) => { slide.hidden = number !== index; });
  previous.disabled = index === 0;
  next.disabled = index === slides.length - 1;
  position.textContent = (index + 1) + ' / ' + slides.length;
  if (moveFocus) slides[index].querySelector('h2').focus();
}

previous.addEventListener('click', () => showSlide(index - 1, true));
next.addEventListener('click', () => showSlide(index + 1, true));
document.addEventListener('keydown', (event) => {
  if (event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
    event.preventDefault();
    showSlide(index + (event.key === 'ArrowRight' ? 1 : -1), true);
  }
});
showSlide(0);
```
