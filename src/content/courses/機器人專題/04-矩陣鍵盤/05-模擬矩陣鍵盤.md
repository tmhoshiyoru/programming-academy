```game-js
// 全域狀態控制 (對應板載 LED 與設定)
let rVal = 255;
let gVal = 0;
let bVal = 0;
let currentBrightness = 50; // 0 ~ 255

// DOM 元件取得
const rgbLed = document.getElementById('rgb-led');
const colorNameText = document.getElementById('color-name');
const brightnessValText = document.getElementById('brightness-val');
const serialLog = document.getElementById('serial-log');
const btnClearSerial = document.getElementById('btn-clear-serial');
const keyButtons = document.querySelectorAll('.key-btn');

// 初始化 Serial 畫面
function init() {
  logSerial("==================================================");
  logSerial(" ESP32-S3 Keypad RGB & Brightness Controller");
  logSerial(" 1:Red, 2:Green, 3:Blue, 0:OFF");
  logSerial(" '*': Brightness -25 , '#': Brightness +25");
  logSerial("==================================================");
  updateLedHardware();
}

// 模擬板載 LED 視覺渲染
function updateLedHardware() {
  // 計算縮放後的顏色 Alpha/亮度比例
  const factor = currentBrightness / 255;
  const renderR = Math.round(rVal * factor);
  const renderG = Math.round(gVal * factor);
  const renderB = Math.round(bVal * factor);

  // 渲染 DOM 顏色與發光 Glow 效果
  if (renderR === 0 && renderG === 0 && renderB === 0) {
    rgbLed.style.backgroundColor = '#1a1a1a';
    rgbLed.style.boxShadow = 'none';
  } else {
    const colorStr = `rgb(${renderR}, ${renderG}, ${renderB})`;
    rgbLed.style.backgroundColor = colorStr;
    rgbLed.style.boxShadow = `0 0 ${Math.max(8, currentBrightness / 8)}px ${colorStr}`;
  }

  // 更新介面狀態文字
  brightnessValText.textContent = currentBrightness;
  
  if (rVal === 255 && gVal === 0 && bVal === 0) colorNameText.textContent = "紅色 (RED)";
  else if (rVal === 0 && gVal === 255 && bVal === 0) colorNameText.textContent = "綠色 (GREEN)";
  else if (rVal === 0 && gVal === 0 && bVal === 255) colorNameText.textContent = "藍色 (BLUE)";
  else if (rVal === 0 && gVal === 0 && bVal === 0) colorNameText.textContent = "關閉 (OFF)";
  else colorNameText.textContent = "自訂色彩";
}

// 印出訊息至 Serial Monitor
function logSerial(text) {
  const line = document.createElement('div');
  line.className = 'serial-line';
  line.textContent = text;
  serialLog.appendChild(line);
  serialLog.scrollTop = serialLog.scrollHeight;
}

// 模擬矩陣鍵盤按鈕邏輯
function handleKeyPress(key) {
  switch (key) {
    case '1':
      rVal = 255; gVal = 0; bVal = 0;
      logSerial(">>> Color: RED");
      updateLedHardware();
      break;

    case '2':
      rVal = 0; gVal = 255; bVal = 0;
      logSerial(">>> Color: GREEN");
      updateLedHardware();
      break;

    case '3':
      rVal = 0; gVal = 0; bVal = 255;
      logSerial(">>> Color: BLUE");
      updateLedHardware();
      break;

    case '0':
      rVal = 0; gVal = 0; bVal = 0;
      logSerial(">>> Color: OFF");
      updateLedHardware();
      break;

    case '*':
      currentBrightness -= 25;
      if (currentBrightness < 0) {
        currentBrightness = 0;
      }
      logSerial(`>>> Brightness Decreased: ${currentBrightness}`);
      updateLedHardware();
      break;

    case '#':
      currentBrightness += 25;
      if (currentBrightness > 255) {
        currentBrightness = 255;
      }
      logSerial(`>>> Brightness Increased: ${currentBrightness}`);
      updateLedHardware();
      break;

    default:
      logSerial(`Pressed key '${key}' (Unassigned)`);
      break;
  }
}

// 綁定按鍵監聽
keyButtons.forEach(btn => {
  btn.addEventListener('click', () => {
    const key = btn.getAttribute('data-key');
    handleKeyPress(key);
  });
});

// 清空 Log 按鈕
btnClearSerial.addEventListener('click', () => {
  serialLog.innerHTML = '';
});

// 啟動模擬器
init();
```
