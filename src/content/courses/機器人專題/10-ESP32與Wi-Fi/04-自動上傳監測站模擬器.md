```game-js
const byId = (id) => document.getElementById(id);
const temperature = byId('temp');
const humidity = byId('humidity');
const wifi = byId('wifi');
const gas = byId('gas');
const rows = byId('rows');
const clockFormat = new Intl.DateTimeFormat('zh-TW', {
  timeZone: 'Asia/Taipei', hour: '2-digit', minute: '2-digit', second: '2-digit', hourCycle: 'h23'
});
let playing = false;
let lastSampleSlot = 0;
let lastUploadSlot = 0;
let temperatureOffset = 0;
let humidityOffset = 0;

function renderRow(id, text) {
  const row = byId(id);
  row.setAttribute('aria-label', text);
  row.replaceChildren(...Array.from(text.padEnd(16), (character) => {
    const cell = document.createElement('span');
    cell.setAttribute('aria-hidden', 'true');
    cell.textContent = character;
    return cell;
  }));
}

function updateControls() {
  byId('temp-value').textContent = temperature.value;
  byId('humidity-value').textContent = humidity.value;
  byId('wifi-state').textContent = wifi.checked ? '已連線' : '已中斷';
  byId('signal').classList.toggle('off', !wifi.checked);
}

function rtcLabel(seconds) { return clockFormat.format(seconds * 1000); }

function tryUpload(reading, clockSeconds) {
  if (!wifi.checked) {
    byId('reply').textContent = '沒有新回覆（Wi-Fi 已中斷）';
    return 'LCD 已更新。Wi-Fi 中斷，試算表沒有新增紀錄。';
  }

  byId('payload').textContent = JSON.stringify(reading);
  if (!gas.checked) {
    byId('reply').textContent = '無法存取網頁應用程式';
    return 'LCD 已更新。資料已送出，但 GAS 無法存取，試算表沒有新增紀錄。';
  }

  const row = rows.insertRow();
  const number = document.createElement('th');
  number.scope = 'row';
  number.textContent = rows.rows.length;
  row.append(number);
  for (const value of [rtcLabel(clockSeconds), reading.temperature, reading.humidity]) {
    row.insertCell().textContent = value;
  }
  rows.parentElement.parentElement.scrollTop = rows.parentElement.parentElement.scrollHeight;
  byId('reply').textContent = 'OK';
  return 'GAS 已寫入第 ' + rows.rows.length + ' 列。';
}

function nextOffset(previous, maxStep, limit) {
  const step = Math.floor(Math.random() * (2 * maxStep + 1)) - maxStep;
  return Math.max(-limit, Math.min(limit, previous + step));
}

function measuredValue(input, offset) {
  const value = Math.round(Number(input.value) * 10) + offset;
  return Math.max(Number(input.min) * 10, Math.min(Number(input.max) * 10, value)) / 10;
}

function sample(clockSeconds) {
  temperatureOffset = nextOffset(temperatureOffset, 2, 10);
  humidityOffset = nextOffset(humidityOffset, 10, 40);
  const reading = {
    temperature: measuredValue(temperature, temperatureOffset),
    humidity: measuredValue(humidity, humidityOffset)
  };
  renderRow('lcd-temp', 'T: ' + reading.temperature.toFixed(1) + ' C');
  renderRow('lcd-humidity', 'H: ' + reading.humidity.toFixed(1) + ' % RH');

  const slot = Math.floor(clockSeconds / 10);
  if (slot > lastUploadSlot) {
    lastUploadSlot = slot;
    byId('status').textContent = tryUpload(reading, clockSeconds);
  } else {
    byId('status').textContent = 'LCD 已更新。距離下個上傳時段還有 ' + (10 - clockSeconds % 10) + ' 秒。';
  }
}

function tick() {
  const seconds = Math.floor(Date.now() / 1000);
  byId('rtc').textContent = 'RTC ' + rtcLabel(seconds);
  const sampleSlot = Math.floor(seconds / 2);
  if (playing && sampleSlot > lastSampleSlot) {
    lastSampleSlot = sampleSlot;
    sample(seconds);
  }
}

function pausePlayback() {
  playing = false;
  byId('play').textContent = '▶ 播放';
}

byId('play').addEventListener('click', () => {
  if (playing) {
    pausePlayback();
    byId('status').textContent = '已暫停取樣與上傳。RTC 繼續顯示現在時間。';
    return;
  }
  const seconds = Math.floor(Date.now() / 1000);
  lastSampleSlot = Math.floor(seconds / 2);
  lastUploadSlot = Math.max(lastUploadSlot, Math.floor(seconds / 10));
  playing = true;
  byId('play').textContent = '⏸ 暫停';
  byId('status').textContent = '播放中，等下一次取樣與上傳。';
});
function reset() {
  pausePlayback();
  lastUploadSlot = 0;
  temperatureOffset = 0;
  humidityOffset = 0;
  temperature.value = '25';
  humidity.value = '60';
  wifi.checked = true;
  gas.checked = true;
  updateControls();
  while (rows.rows.length > 1) rows.deleteRow(-1);
  renderRow('lcd-temp', 'T: -- C');
  renderRow('lcd-humidity', 'H: -- % RH');
  byId('payload').textContent = '尚未送出';
  byId('reply').textContent = '等待請求';
  byId('status').textContent = '按播放開始取樣，並等下一個 10 秒時段上傳。';
}

temperature.addEventListener('input', updateControls);
humidity.addEventListener('input', updateControls);
wifi.addEventListener('change', updateControls);
byId('reset').addEventListener('click', reset);
reset();
tick();
setInterval(tick, 250);
```
