## 任務

沿用第 8 章接線與 LCD 顯示，讓 ESP32-S3 每 2 秒更新 LCD，在 RTC 的 00、10、20、30、40、50 秒附近，各上傳一次最新的有效讀值。試算表新增時間、溫度、相對溼度三欄，儀表板跟著更新。

## 上傳前要改的地方

1. 在主程式填入自己的 `ssid`、`password` 和 GAS 的 `gasUrl`（以 `/exec` 結尾）。
2. 將 LCD 的 `0x27` 換成掃描到的 I2C 位址（若你的位址就是 `0x27`，直接沿用）。
3. 在同一份 Arduino 草稿建立 `google_roots.h`，把[上一節](../02-wi-fi串接gas/)最下方的憑證分頁整段貼上。

上傳前先試著指出兩個計時變數。哪一個記錄上次讀取感測器的時間？哪一個記錄已處理過的 10 秒時段？

上傳要等待網路回覆，該輪 LCD 更新可能延後。LCD 用 millis() 每 2 秒取樣，GAS 則按 RTC 時段上傳。若取樣稍晚，仍可在該時段內上傳一次。若開機校時或連線失敗，LCD 仍能讀取和顯示，修正網路後重啟再測。

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <WiFiClientSecure.h>
#include <time.h>
#include "google_roots.h"

// DHT11 的 DATA 接 GPIO 10。
DHT dht(10, DHT11);
// LCD 的 I2C 位址、每行字數和行數。
LiquidCrystal_I2C lcd(0x27, 16, 2);
const unsigned long sampleInterval = 2000;
unsigned long lastSample = 0;
time_t lastUploadSlot = 0;
bool uploadReady = false;

const char* ssid = "填入 Wi-Fi 名稱";
const char* password = "填入 Wi-Fi 密碼";

bool connectWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  Serial.print("Connecting");
  const unsigned long started = millis();

  while (WiFi.status() != WL_CONNECTED && millis() - started < 15000) {
    delay(250);
    Serial.print('.');
  }
  Serial.println();

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi timeout");
    return false;
  }

  Serial.print("Wi-Fi connected, IP: ");
  Serial.println(WiFi.localIP());
  return true;
}

const char* gasUrl = "貼上 GAS 的 /exec 網址";

bool syncClock() {
  configTzTime("CST-8", "time.google.com", "pool.ntp.org");
  const unsigned long started = millis();
  while (time(nullptr) < 1700000000 && millis() - started < 15000) {
    delay(250);
  }
  return time(nullptr) >= 1700000000;
}

bool sendReading(float temperature, float humidity) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi disconnected");
    return false;
  }

  char payload[80];
  snprintf(payload, sizeof(payload),
           "{\"temperature\":%.1f,\"humidity\":%.1f}", temperature, humidity);

  WiFiClientSecure client;
  client.setCACert(googleRoots);
  HTTPClient http;
  http.setConnectTimeout(5000);
  http.setTimeout(5000);
  http.setFollowRedirects(HTTPC_DISABLE_FOLLOW_REDIRECTS);

  if (!http.begin(client, gasUrl)) return false;
  http.addHeader("Content-Type", "application/json");
  int status = http.POST(String(payload));

  if (status == 302 || status == 303) {
    const String redirect = http.getLocation();
    http.end();

    if (!redirect.startsWith("https://script.googleusercontent.com/")) {
      Serial.println("Unexpected redirect. Check GAS deployment access.");
      return false;
    }
    if (!http.begin(client, redirect)) return false;
    status = http.GET();
  }

  String reply = status == 200 ? http.getString() : "";
  http.end();
  reply.trim();
  Serial.printf("HTTP %d\n", status);
  if (reply.length()) Serial.println(reply);
  return status == 200 && reply == "OK";
}

void setup() {
  Serial.begin(115200);
  // 啟動 DHT11，讓程式可以向感測器讀取資料。
  dht.begin();
  // 設定 I2C：SDA 接 GPIO 8，SCL 接 GPIO 9，時脈為 100 kHz。
  Wire.begin(8, 9, 100000);
  // 初始化 LCD、開啟背光，先顯示等待訊息。
  lcd.init();
  lcd.backlight();
  lcd.print("Waiting...");
  lastSample = millis();

  // 放在原本 setup() 的結尾，先讓 LCD 完成初始化。
  uploadReady = connectWiFi() && syncClock();
  if (!uploadReady) Serial.println("Upload unavailable. Check Wi-Fi and clock.");
  // TODO 1：校時成功後，用 Unix 秒數算出目前的 10 秒時段。
  // 先記下它，讓第一次上傳從下一個時段開始。
  if (uploadReady) lastUploadSlot = time(nullptr) / ____;
}

void printReading(uint8_t row, const char* label, float value, const char* unit) {
  lcd.setCursor(0, row);
  // 用 16 個空格清掉整行，再回到行首寫入新數值。
  lcd.print("                ");
  lcd.setCursor(0, row);
  lcd.print(label);
  lcd.print(value, 1);
  lcd.print(unit);
}

void loop() {
  const unsigned long now = millis();
  // now 是目前的毫秒數，lastSample 是上次取樣時的毫秒數。
  // 兩者相減就是經過的時間。還沒滿 2000 毫秒，就先結束這一輪 loop()。
  // return 不會停住程式，Arduino 會再呼叫 loop()，重新檢查時間。
  // 前幾章用「經過時間 >= 間隔」才進入 if 區塊。這裡先用相反條件
  // 跳過還沒到時間的情況，所以下方的讀取程式只有到時間才會執行。
  if (now - lastSample < sampleInterval) return;
  // 記下這次取樣的時間，下一次從這裡重新計算間隔。
  lastSample = now;

  // 分別讀取相對溼度（% RH）和攝氏溫度（°C）。
  const float humidity = dht.readHumidity();
  const float temperature = dht.readTemperature();
  // 任一讀值失敗時顯示錯誤，結束本次更新。
  if (isnan(temperature) || isnan(humidity)) {
    Serial.println("DHT11 read failed");
    lcd.setCursor(0, 0);
    lcd.print("DHT11 read error");
    lcd.setCursor(0, 1);
    lcd.print("Check wiring    ");
    return;
  }

  Serial.printf("T: %.1f C, H: %.1f %% RH\n", temperature, humidity);
  // LCD 行索引從 0 開始。第一行顯示溫度，第二行顯示溼度。
  printReading(0, "T: ", temperature, " C");
  printReading(1, "H: ", humidity, " % RH");

  const time_t nowOnClock = time(nullptr);
  // TODO 2：把這次讀到的 RTC 秒數換成時段編號，算法和 setup() 相同。
  const time_t currentSlot = nowOnClock / ____;
  // TODO 3：填入比較運算子。只有目前時段比已處理的時段更新，才上傳。
  if (uploadReady && currentSlot ____ lastUploadSlot) {
    lastUploadSlot = currentSlot; // 先標記，避免同一時段重送。
    struct tm localTime;
    localtime_r(&nowOnClock, &localTime);
    char clockText[9];
    strftime(clockText, sizeof(clockText), "%H:%M:%S", &localTime);
    Serial.printf("RTC %s | T: %.1f C, H: %.1f %% RH\n", clockText, temperature, humidity);
    // TODO 4：找到 loop() 前半段的兩個讀值變數，依序填入溫度和溼度。

    Serial.println(uploaded ? "Reading logged" : "Upload not confirmed");
  }
}
```

## 階段成果驗收

- 在 LCD 上辨認溫度和相對溼度。說明為什麼 LCD 更新多次，試算表才新增一列。
- 觀察一整分鐘，核對 00、10、20、30、40、50 秒附近各新增一筆，並比對序列監控視窗與試算表中的溫溼度讀值。

  <strong style="color: #b45309;">時間差是正常的。序列監控視窗顯示 ESP32 準備上傳的時間，試算表記的是 GAS 收到資料的時間。實測試算表大約晚 3～4 秒，實際差距會隨網路狀況而變。驗收時請確認每個時段各有一筆，且溫溼度讀值一致。</strong>
- 開啟儀表板，確認最新溫溼度與最後一列一致，溫度折線圖也出現新資料點。
- 若只看到 LCD 更新，先看序列監控視窗的 Wi-Fi、校時與 HTTP 回覆，再依第 9 章的測試流程定位問題。
