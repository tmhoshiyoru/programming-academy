```game-js
let timer = null;
let currentState = 0;
let previousMillis = 0;

// 取得元件控制權
const lightRed = document.getElementById('light-red');
const lightYellow = document.getElementById('light-yellow');
const lightGreen = document.getElementById('light-green');

const pinGnd = document.getElementById('pin-gnd');
const pin19 = document.getElementById('pin-19');
const pin20 = document.getElementById('pin-20');
const pin21 = document.getElementById('pin-21');

const wireGnd = document.getElementById('wire-gnd');
const wire19 = document.getElementById('wire-19');
const wire20 = document.getElementById('wire-20');
const wire21 = document.getElementById('wire-21');

const mpinGnd = document.getElementById('mpin-gnd');
const mpin19 = document.getElementById('mpin-19');
const mpin20 = document.getElementById('mpin-20');
const mpin21 = document.getElementById('mpin-21');

const statusText = document.getElementById('status-text');
const feedback = document.getElementById('feedback');

// 設定燈號與硬體線路發光狀態
function setTrafficLight(r, y, g) {
  // 控制 LED 燈
  lightRed.classList.toggle('active', r);
  lightYellow.classList.toggle('active', y);
  lightGreen.classList.toggle('active', g);

  // 控制導线與腳位 (GND 通電迴路永遠同步)
  const isAnyOn = r || y || g;
  pinGnd.classList.toggle('active', isAnyOn);
  wireGnd.classList.toggle('active', isAnyOn);
  mpinGnd.classList.toggle('active', isAnyOn);

  // 紅燈路徑 (GPIO 19)
  pin19.classList.toggle('active', r);
  wire19.classList.toggle('active', r);
  mpin19.classList.toggle('active', r);

  // 黃燈路徑 (GPIO 20)
  pin20.classList.toggle('active', y);
  wire20.classList.toggle('active', y);
  mpin20.classList.toggle('active', y);

  // 綠燈路徑 (GPIO 21)
  pin21.classList.toggle('active', g);
  wire21.classList.toggle('active', g);
  mpin21.classList.toggle('active', g);
}

function resetSimulator() {
  if (timer) clearInterval(timer);
  currentState = 0;
  previousMillis = 0;
  setTrafficLight(false, false, false);
  statusText.textContent = "待命中";
  feedback.className = "feedback";
}

function validateCode() {
  const case1 = document.getElementById('code-case1').value;
  const case2 = document.getElementById('code-case2').value;

  // 驗證 Case 1 (黃燈)
  const hasYellowLight = /setTrafficLight\(\s*false\s*,\s*true\s*,\s*false\s*\)/.test(case1);
  const hasYellowTime = /2000/.test(case1);
  const hasNextState2 = /currentState\s*=\s*2/.test(case1);

  // 驗證 Case 2 (紅燈)
  const hasRedLight = /setTrafficLight\(\s*true\s*,\s*false\s*,\s*false\s*\)/.test(case2);
  const hasRedTime = /5000/.test(case2);
  const hasNextState0 = /currentState\s*=\s*0/.test(case2);

  if (!hasYellowLight || !hasYellowTime || !hasNextState2) {
    showFeedback("黃燈階段 (Case 1) 邏輯錯誤！請確認：1. 點亮黃燈 (false, true, false) 2. 時間檢查 >= 2000 3. 切換至 currentState = 2", false);
    return false;
  }

  if (!hasRedLight || !hasRedTime || !hasNextState0) {
    showFeedback("紅燈階段 (Case 2) 邏輯錯誤！請確認：1. 點亮紅燈 (true, false, false) 2. 時間檢查 >= 5000 3. 切換至 currentState = 0", false);
    return false;
  }

  showFeedback("語法檢查正確！開始執行硬體模擬切換...", true);
  return true;
}

function showFeedback(msg, isSuccess) {
  feedback.textContent = msg;
  feedback.className = `feedback ${isSuccess ? 'success' : 'error'}`;
}

function startSimulation() {
  resetSimulator();
  if (!validateCode()) return;

  previousMillis = Date.now();

  timer = setInterval(() => {
    const currentMillis = Date.now();
    const elapsed = currentMillis - previousMillis;

    switch (currentState) {
      case 0: // 綠燈 (5秒)
        setTrafficLight(false, false, true);
        statusText.textContent = `🟢 綠燈通行中 [GPIO 21 導通] (${Math.ceil((5000 - elapsed)/1000)}s)`;
        if (elapsed >= 5000) {
          currentState = 1;
          previousMillis = currentMillis;
        }
        break;

      case 1: // 黃燈 (2秒)
        setTrafficLight(false, true, false);
        statusText.textContent = `🟡 黃燈減速中 [GPIO 20 導通] (${Math.ceil((2000 - elapsed)/1000)}s)`;
        if (elapsed >= 2000) {
          currentState = 2;
          previousMillis = currentMillis;
        }
        break;

      case 2: // 紅燈 (5秒)
        setTrafficLight(true, false, false);
        statusText.textContent = `🔴 紅燈停止中 [GPIO 19 導通] (${Math.ceil((5000 - elapsed)/1000)}s)`;
        if (elapsed >= 5000) {
          currentState = 0;
          previousMillis = currentMillis;
        }
        break;
    }
  }, 100);
}

document.getElementById('btn-run').addEventListener('click', startSimulation);
document.getElementById('btn-reset').addEventListener('click', resetSimulator);
```
