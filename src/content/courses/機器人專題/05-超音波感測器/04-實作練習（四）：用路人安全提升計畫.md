這個專案將結合 HC-SR04 超音波感測器、紅綠燈模組 與 millis() 時間控制，打造一個模擬真實路口的「感應式觸發號誌系統」。首先，請將紅綠燈模組也接到麵包板上吧！

## 一、 專案設計
系統平時維持在預設的閃爍狀態，當偵測到有行人靠近時，自動啟動一連串的切換流程：
```PlainText
 ┌────────────────────────────────────────────────────────┐
 │ 【狀態 0】STATE_IDLE_FLASHING                           │
 │  平時狀態：黃燈每 0.5 秒閃爍一次                          │
 └──────────────────────────┬─────────────────────────────┘
                            │
                            │ 觸發條件：距離 < 15 cm
                            ▼
 ┌────────────────────────────────────────────────────────┐
 │ 【狀態 1】STATE_PRE_RED                                 │
 │  警告狀態：黃燈常亮 1 秒（提示即將變紅燈）                 │
 └──────────────────────────┬─────────────────────────────┘
                            │
                            │ 時間滿 1 秒
                            ▼
 ┌────────────────────────────────────────────────────────┐
 │ 【狀態 2】STATE_RED_LIGHT                               │
 │  停止狀態：紅燈亮 5 秒                                   │
 └──────────────────────────┬─────────────────────────────┘
                            │
                            │ 時間滿 5 秒
                            ▼
 ┌────────────────────────────────────────────────────────┐
 │ 【狀態 3】STATE_GREEN_LIGHT                             │
 │  通行狀態：綠燈亮 5 秒                                   │
 └──────────────────────────┬─────────────────────────────┘
                            │
                            │ 時間滿 5 秒
                            ▼
               (回到【狀態 0】繼續閃黃燈)
```

## 二、 狀態控管說明（#define 巨集）
為了維持程式的可讀性，我們使用 #define 巨集來為每個狀態命名：
```c
#define STATE_IDLE_FLASHING 0  // 平時閃黃燈
#define STATE_PRE_RED       1  // 黃燈常亮 (準備變紅燈)
#define STATE_RED_LIGHT     2  // 紅燈亮
#define STATE_GREEN_LIGHT   3  // 綠燈亮
```

```c=

#define TRIG_PIN   41
#define ECHO_PIN   42

#define RED_PIN    19
#define YELLOW_PIN 20
#define GREEN_PIN  21

#define STATE_IDLE_FLASHING 0
#define STATE_PRE_RED       1
#define STATE_RED_LIGHT     2
#define STATE_GREEN_LIGHT   3

UltraSonicDistanceSensor distanceSensor(TRIG_PIN, ECHO_PIN);

int currentState = STATE_IDLE_FLASHING;

unsigned long lastSensorTime = 0;
const unsigned long SENSOR_INTERVAL = 200;

unsigned long lastFlashTime = 0;
const unsigned long FLASH_INTERVAL = 500;
bool yellowState = false;

unsigned long stateStartTime = 0;
float currentDistance = 0.0;

void setLights(bool red, bool yellow, bool green) {
  digitalWrite(RED_PIN, red ? HIGH : LOW);
  digitalWrite(YELLOW_PIN, yellow ? HIGH : LOW);
  digitalWrite(GREEN_PIN, green ? HIGH : LOW);
}

void setup() {
  Serial.begin(115200);

  pinMode(RED_PIN, OUTPUT);
  pinMode(YELLOW_PIN, OUTPUT);
  pinMode(GREEN_PIN, OUTPUT);

  setLights(false, false, false);
  Serial.println("Traffic Light Practice Initialized.");
}

void loop() {
  unsigned long currentMillis = millis();

  if (currentMillis - lastSensorTime >= SENSOR_INTERVAL) {
    lastSensorTime = currentMillis;
    currentDistance = distanceSensor.measureDistanceCm();
  }

  switch (currentState) {

    case STATE_IDLE_FLASHING:
      if (currentMillis - lastFlashTime >= FLASH_INTERVAL) {
        lastFlashTime = currentMillis;
        yellowState = !yellowState;
        setLights(false, yellowState, false);
      }

      // =======================================================
      // 任務 A：當距離小於 15 cm 時，觸發狀態切換至 STATE_PRE_RED
      // =======================================================
```
