```game-js
// --- 常數定義 ---
const STATE_IDLE_FLASHING = 0;
const STATE_PRE_RED       = 1;
const STATE_RED_LIGHT     = 2;
const STATE_GREEN_LIGHT   = 3;

// --- 全域變數 ---
let currentState = STATE_IDLE_FLASHING;
let currentDistance = 999;

let stateStartTime = Date.now();
let lastFlashTime = Date.now();
let yellowState = false;

// 拖曳狀態控制
let isDragging = false;
let dragOffsetX = 0;
let dragOffsetY = 0;

// DOM 元件
const viewport = document.getElementById('viewport');
const pedestrian = document.getElementById('pedestrian');
const sensor = document.getElementById('sensor');
const distanceDisplay = document.getElementById('distance-display');
const timerDisplay = document.getElementById('timer-display');
const stateDisplay = document.getElementById('state-display');

const lightRed = document.getElementById('light-red');
const lightYellow = document.getElementById('light-yellow');
const lightGreen = document.getElementById('light-green');

// 初始化事件與主迴圈
function init() {
  setupDragEvents();
  requestAnimationFrame(gameLoop);
}

// 設定拖曳事件 (滑鼠與觸控)
function setupDragEvents() {
  const startDrag = (clientX, clientY) => {
    isDragging = true;
    const pedRect = pedestrian.getBoundingClientRect();
    dragOffsetX = clientX - pedRect.left;
    dragOffsetY = clientY - pedRect.top;
  };

  const moveDrag = (clientX, clientY) => {
    if (!isDragging) return;
    const viewRect = viewport.getBoundingClientRect();
    
    // 計算相對於 viewport 的新座標
    let newX = clientX - viewRect.left - dragOffsetX;
    let newY = clientY - viewRect.top - dragOffsetY;

    // 邊界限制，防止拖出畫面
    newX = Math.max(0, Math.min(viewRect.width - 26, newX));
    newY = Math.max(0, Math.min(viewRect.height - 26, newY));

    pedestrian.style.left = `${newX}px`;
    pedestrian.style.top = `${newY}px`;
  };

  const stopDrag = () => {
    isDragging = false;
  };

  // 滑鼠事件
  pedestrian.addEventListener('mousedown', (e) => startDrag(e.clientX, e.clientY));
  window.addEventListener('mousemove', (e) => moveDrag(e.clientX, e.clientY));
  window.addEventListener('mouseup', stopDrag);

  // 觸控事件 (支援行動裝置)
  pedestrian.addEventListener('touchstart', (e) => {
    if (e.touches.length > 0) startDrag(e.touches[0].clientX, e.touches[0].clientY);
  });
  window.addEventListener('touchmove', (e) => {
    if (e.touches.length > 0) moveDrag(e.touches[0].clientX, e.touches[0].clientY);
  });
  window.addEventListener('touchend', stopDrag);
}

// 主遊戲與號誌更新迴圈
function gameLoop() {
  calculateDistance();
  updateStateMachine();
  requestAnimationFrame(gameLoop);
}

// 計算行人距離超音波感測器的模擬公分數 (15 cm 模擬)
function calculateDistance() {
  const sensorRect = sensor.getBoundingClientRect();
  const pedRect = pedestrian.getBoundingClientRect();

  const sensorX = sensorRect.left + sensorRect.width / 2;
  const sensorY = sensorRect.top + sensorRect.height / 2;
  const pedX = pedRect.left + pedRect.width / 2;
  const pedY = pedRect.top + pedRect.height / 2;

  const pixelDist = Math.hypot(sensorX - pedX, sensorY - pedY);

  // 每 8px 代表 1cm
  currentDistance = Math.round(pixelDist / 8);
  distanceDisplay.textContent = `${currentDistance} cm`;
}

// 控制燈號熄滅/亮起
function setLights(red, yellow, green) {
  lightRed.classList.toggle('active', red);
  lightYellow.classList.toggle('active', yellow);
  lightGreen.classList.toggle('active', green);
}

// 狀態機核心邏輯
function updateStateMachine() {
  const now = Date.now();
  const elapsedSeconds = Math.floor((now - stateStartTime) / 1000);

  switch (currentState) {

    // 【狀態 0：平時閃黃燈】
    case STATE_IDLE_FLASHING:
      stateDisplay.textContent = "平時閃黃燈 (待命中)";
      stateDisplay.className = "dash-value status-idle";
      timerDisplay.textContent = "--";

      if (now - lastFlashTime >= 500) {
        lastFlashTime = now;
        yellowState = !yellowState;
        setLights(false, yellowState, false);
      }

      if (currentDistance <= 15) {
        currentState = STATE_PRE_RED;
        stateStartTime = now;
      }
      break;

    // 【狀態 1：黃燈常亮 1 秒】
    case STATE_PRE_RED:
      stateDisplay.textContent = "偵測到行人！黃燈常亮";
      stateDisplay.className = "dash-value status-warning";
      setLights(false, true, false);

      const yellowRemain = 1 - elapsedSeconds;
      timerDisplay.textContent = `${Math.max(1, yellowRemain)} 秒`;

      if (now - stateStartTime >= 1000) {
        currentState = STATE_RED_LIGHT;
        stateStartTime = now;
      }
      break;

    // 【狀態 2：紅燈亮 5 秒】
    case STATE_RED_LIGHT:
      stateDisplay.textContent = "紅燈停 (行人通行)";
      stateDisplay.className = "dash-value status-stop";
      setLights(true, false, false);

      const redRemain = 5 - elapsedSeconds;
      timerDisplay.textContent = `${Math.max(1, redRemain)} 秒`;

      if (now - stateStartTime >= 5000) {
        currentState = STATE_GREEN_LIGHT;
        stateStartTime = now;
      }
      break;

    // 【狀態 3：綠燈亮 5 秒】
    case STATE_GREEN_LIGHT:
      stateDisplay.textContent = "綠燈行 (車輛通行)";
      stateDisplay.className = "dash-value status-go";
      setLights(false, false, true);

      const greenRemain = 5 - elapsedSeconds;
      timerDisplay.textContent = `${Math.max(1, greenRemain)} 秒`;

      if (now - stateStartTime >= 5000) {
        currentState = STATE_IDLE_FLASHING;
      }
      break;
  }
}

// 啟動模擬器
init();
```
