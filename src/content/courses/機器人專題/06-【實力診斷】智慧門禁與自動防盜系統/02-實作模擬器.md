```game-js
// === 系統狀態與變數 ===
let currentStage = 1;
let passwordInput = "";
const TARGET_PASSWORD = "1234";
let errorCount = 0;

// 定時器與鎖定狀態 (模擬 millis 控制)
let isLocked = false;
let isWarning = false;
let blinkInterval = null;

// === DOM 元素存取 ===
const lightRed = document.getElementById('light-red');
const lightYellow = document.getElementById('light-yellow');
const lightGreen = document.getElementById('light-green');

const keypadDisplay = document.getElementById('keypad-input');
const distanceSlider = document.getElementById('distance-slider');
const distanceVal = document.getElementById('distance-val');
const ultrasonicGroup = document.getElementById('ultrasonic-group');
const consoleOutput = document.getElementById('console-output');
const stageDesc = document.getElementById('stage-desc');

// === 初始化 ===
document.addEventListener('DOMContentLoaded', () => {
    initKeypad();
    initStageSelector();
    initUltrasonic();
    document.getElementById('btn-clear-serial').addEventListener('click', clearConsole);
});

// 印出 Serial Console 訊息
function logSerial(msg, type = 'system') {
    const time = new Date().toLocaleTimeString('zh-TW', { hour12: false });
    const logItem = document.createElement('div');
    logItem.className = `log-item ${type}`;
    logItem.textContent = `[${time}] ${msg}`;
    consoleOutput.appendChild(logItem);
    consoleOutput.scrollTop = consoleOutput.scrollHeight;
}

function clearConsole() {
    consoleOutput.innerHTML = '';
}

// === 燈號控制函數 ===
function setLights(red, yellow, green) {
    lightRed.classList.toggle('active', red);
    lightYellow.classList.toggle('active', yellow);
    lightGreen.classList.toggle('active', green);
}

function clearBlink() {
    if (blinkInterval) {
        clearInterval(blinkInterval);
        blinkInterval = null;
    }
}

// === 切換關卡選單 ===
function initStageSelector() {
    const stageBtns = document.querySelectorAll('.stage-btn');
    stageBtns.forEach(btn => {
        btn.addEventListener('click', (e) => {
            stageBtns.forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            currentStage = parseInt(btn.dataset.stage);
            resetSystemState();
            updateStageUI();
        });
    });
}

function updateStageUI() {
    if (currentStage === 1) {
        stageDesc.innerHTML = '當前關卡 1 目標：輸入預設密碼 <strong>1234</strong> 解鎖。';
        ultrasonicGroup.classList.add('disabled');
        distanceSlider.disabled = true;
    } else if (currentStage === 2) {
        stageDesc.innerHTML = '當前關卡 2 目標：嘗試拖動距離滑桿至 <strong>< 10cm</strong> 觸發自動預警。';
        ultrasonicGroup.classList.remove('disabled');
        distanceSlider.disabled = false;
    } else if (currentStage === 3) {
        stageDesc.innerHTML = '當前關卡 3 目標：連續輸入錯誤密碼 <strong>3 次</strong>，體驗 10 秒非阻塞鎖定。';
        ultrasonicGroup.classList.remove('disabled');
        distanceSlider.disabled = false;
    }
    logSerial(`[System] 切換至關卡 ${currentStage}`, 'system');
}

function resetSystemState() {
    clearBlink();
    isLocked = false;
    isWarning = false;
    errorCount = 0;
    passwordInput = "";
    keypadDisplay.textContent = "----";
    setLights(true, false, false); // 恢復待命紅燈
}

// === 矩陣鍵盤邏輯 ===
function initKeypad() {
    const keys = document.querySelectorAll('.key-btn');
    keys.forEach(key => {
        key.addEventListener('click', () => {
            if (isLocked) {
                logSerial('[Keypad] 系統鎖定中，按鍵輸入無效！', 'error');
                return;
            }

            const val = key.textContent;
            if (val === 'C') {
                passwordInput = "";
            } else if (val === '#') {
                verifyPassword();
                return;
            } else {
                if (passwordInput.length < 4) {
                    passwordInput += val;
                }
            }
            keypadDisplay.textContent = passwordInput.padEnd(4, '-');
        });
    });
}

// 密碼比對邏輯
function verifyPassword() {
    if (passwordInput.length < 4) {
        logSerial('[Error] 密碼長度不足 4 位！', 'error');
        return;
    }

    if (passwordInput === TARGET_PASSWORD) {
        // 密碼正確
        clearBlink();
        setLights(false, false, true); // 綠燈亮
        logSerial('[Access Granted] 密碼正確，允許進入！', 'grant');
        
        setTimeout(() => {
            if (!isLocked && !isWarning) setLights(true, false, false);
        }, 3000);
    } else {
        // 密碼錯誤
        logSerial('[Password Error] 密碼錯誤！', 'error');
        errorCount++;

        // 關卡三：連續錯 3 次觸發鎖定
        if (currentStage === 3 && errorCount >= 3) {
            triggerLockdown();
            return;
        }

        // 閃爍紅燈 3 次
        let blinkCount = 0;
        clearBlink();
        blinkInterval = setInterval(() => {
            lightRed.classList.toggle('active');
            blinkCount++;
            if (blinkCount >= 6) {
                clearBlink();
                setLights(true, false, false);
            }
        }, 200);
    }

    passwordInput = "";
    keypadDisplay.textContent = "----";
}

// === 關卡二：超音波感測器 ===
function initUltrasonic() {
    distanceSlider.addEventListener('input', (e) => {
        const dist = parseInt(e.target.value);
        distanceVal.textContent = dist;

        if (currentStage >= 2 && !isLocked) {
            if (dist < 10 && !isWarning) {
                // 觸發警報
                isWarning = true;
                logSerial(`[Warning: Intruder Detected!] 距離: ${dist} cm`, 'warn');
                startYellowBlink();
            } else if (dist >= 10 && isWarning) {
                // 解除警報
                isWarning = false;
                clearBlink();
                setLights(true, false, false);
                logSerial(`[System] 目標離開，恢復待命狀態 (距離: ${dist} cm)`, 'system');
            }
        }
    });
}

// 黃燈快速閃爍 (以 0.2 秒代表 millis 模擬)
function startYellowBlink() {
    clearBlink();
    let state = false;
    blinkInterval = setInterval(() => {
        state = !state;
        setLights(false, state, false);
    }, 200);
}

// === 關卡三：非阻塞鎖定 (10 秒) ===
function triggerLockdown() {
    isLocked = true;
    clearBlink();
    setLights(true, false, false); // 紅燈常亮
    logSerial('[System Locked! Wait 10 seconds...]', 'error');

    let remainingTime = 10;
    const lockTimer = setInterval(() => {
        remainingTime--;
        if (remainingTime > 0) {
            logSerial(`[Lockdown] 強制鎖定剩餘 ${remainingTime} 秒...`, 'warn');
        } else {
            clearInterval(lockTimer);
            isLocked = false;
            errorCount = 0;
            logSerial('[System] 鎖定解除，系統恢復待命！', 'grant');
        }
    }, 1000);
}
```
