## 題目

跑長編碼用「位元、連續次數」表示一段相同的位元。例如 `0000001111111001111` 可以拆成 `(0,6) (1,7) (0,2) (1,4)`。

每個碼字由 1 個原始位元和 3 個表示次數的位元組成。例如連續 6 個 `0` 編成 `0110`。3 個位元最多表示 7，因此同一段若超過 7 個，就每滿 7 個輸出一組，再接著計數。

- **輸入**　原題第一列為測資筆數，之後每列一筆，長度小於 500。本動畫一次處理一筆。
- **輸出**　以空格隔開各個 4 位元碼字，最後附上壓縮後位元數除以原始位元數、四捨五入後的整數百分比。遇到 `0` 和 `1` 以外的字元就輸出 `-1`。資料重複度低時，壓縮率可能超過 100%。

## 觀察動畫

輸入一筆測資，或載入固定、隨機範例後按「執行」。觀察相同位元如何分組，以及每個碼字如何由位元與次數組成。遇到非法字元時，動畫會停在該格，整筆輸出改成 `-1`。

```game-js
const source = document.querySelector('#source');
const status = document.querySelector('#status');
const viewport = document.querySelector('#viewport');
const cells = document.querySelector('#cells');
const scanner = document.querySelector('#scanner');
const step = document.querySelector('#step');
const result = document.querySelector('#result');
const rate = document.querySelector('#rate');
const examples = [
  '00010000000111111101111111',
  '11111100000000000000111111111111110000',
  '0011  1111100000101010',
  'HINET0800000123',
];
let runId = 0;

function reset() {
  runId += 1;
  viewport.hidden = true;
  viewport.classList.remove('invalid');
  result.classList.remove('invalid');
  cells.replaceChildren();
  step.textContent = '';
  result.value = 'string s = "";';
  rate.hidden = true;
  rate.textContent = '';
  return runId;
}

function loadExample(value) {
  reset();
  source.value = value;
  status.textContent = '已載入範例，按「執行」開始。';
}

function pause(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

for (const button of document.querySelectorAll('[data-example]')) {
  button.addEventListener('click', () => loadExample(examples[Number(button.dataset.example)]));
}
document.querySelector('#random').addEventListener('click', () => {
  const groups = 3 + Math.floor(Math.random() * 5);
  let bit = Math.floor(Math.random() * 2);
  let value = '';
  for (let i = 0; i < groups; i += 1) {
    value += String(bit).repeat(1 + Math.floor(Math.random() * 12));
    bit = 1 - bit;
  }
  loadExample(value);
});

const runButton = document.querySelector('#run');
source.addEventListener('keydown', (event) => {
  if (event.key === 'Enter' && !event.isComposing) runButton.click();
});

runButton.addEventListener('click', async () => {
  const currentRun = reset();
  const bits = Array.from(source.value);
  if (bits.length === 0 || bits.length >= 500) {
    status.textContent = '請輸入一筆 1 到 499 字元的測資。';
    return;
  }

  const tiles = bits.map((bit) => {
    const tile = document.createElement('span');
    tile.className = 'cell';
    tile.textContent = bit === ' ' ? '␣' : bit;
    return tile;
  });
  cells.append(...tiles);
  scanner.style.setProperty('--travel', '0ms');
  scanner.style.left = '2.4rem';
  scanner.textContent = '掃描';
  viewport.hidden = false;
  viewport.scrollLeft = 0;
  status.textContent = '從左往右掃描，同一組最多收 7 個位元。';
  await pause(160);
  if (currentRun !== runId) return;

  const delay = Math.min(480, Math.max(40, Math.floor(24000 / bits.length)));
  const travel = Math.floor(delay / 2);
  scanner.style.setProperty('--travel', travel + 'ms');
  const words = [];
  let s = '';
  let count = 0;

  for (let index = 0; index < bits.length; index += 1) {
    const bit = bits[index];
    const tile = tiles[index];
    const center = tile.offsetLeft + tile.offsetWidth / 2;
    scanner.style.left = center + 'px';
    viewport.scrollTo({ left: center - viewport.clientWidth / 2, behavior: 'instant' });
    await pause(travel);
    if (currentRun !== runId) return;
    tiles[index - 1]?.classList.remove('active');
    tile.classList.add('active');

    if (bit !== '0' && bit !== '1') {
      viewport.classList.add('invalid');
      result.classList.add('invalid');
      scanner.textContent = '停止';
      status.textContent = '第 ' + (index + 1) + ' 個字元不是 0 或 1，這筆測資無效。';
      s = '-1';
      result.value = 's = "' + s + '";';
      step.textContent = '遇到非法字元「' + tile.textContent + '」，將 s 指定為 "-1"，立即結束。';
      return;
    }

    tile.classList.add(words.length % 2 ? 'group-b' : 'group-a');
    count += 1;
    scanner.textContent = bit + ' × ' + count;
    step.textContent = '目前累積 ' + count + ' 個 ' + bit + '。';
    if (count === 7 || bits[index + 1] !== bit) {
      const lengthBits = count.toString(2).padStart(3, '0');
      const word = bit + lengthBits;
      words.push(word);
      s += (s ? ' ' : '') + word;
      result.value = 'string s = "' + s + '";';
      step.textContent = '位元 ' + bit + ' ＋ 次數 ' + count + '（' + lengthBits + '）→ ' + word;
      count = 0;
    }

    const nextBit = bits[index + 1];
    const changesBit = (nextBit === '0' || nextBit === '1') && nextBit !== bit;
    await pause(delay - travel + (changesBit ? delay : 0));
    if (currentRun !== runId) return;
  }

  const compressed = words.length * 4;
  const percentage = Math.round(compressed * 100 / bits.length);
  rate.textContent = '壓縮率 ' + percentage + '%';
  rate.hidden = false;
  scanner.textContent = '完成';
  status.textContent = '編碼完成，共 ' + words.length + ' 個碼字。';
  step.textContent = words.length + ' × 4 = ' + compressed + ' 位元。壓縮率為 '
    + compressed + ' ÷ ' + bits.length + ' × 100%，四捨五入為 ' + percentage + '%。';
});
reset();
```
