## 題目

從最右邊開始數，個位是第 1 位，十位是第 2 位。把奇數位上的數字加總為 A，把偶數位上的數字加總為 B，秘密差就是兩者差的絕對值 |A − B|。

例如，263541 的 A = 6 + 5 + 1 = 12，B = 2 + 3 + 4 = 9，所以秘密差是 3。

- **輸入**　一行十進位正整數 X，最多 1000 位，以換行結尾。
- **輸出**　一行秘密差，以十進位表示並以換行結尾。

| 範例輸入 | 範例輸出 |
| --- | --- |
| 263541 | 3 |
| 131 | 1 |

## 觀察動畫

輸入正整數或載入範例，再按「執行」。字串由左至右標成 `s[0]` 到 `s[n-1]`，方塊上的「第 i 位」表示字串索引。判斷題目的奇偶位時，仍從右數，使用 `n - i`。奇數位往上移，偶數位往下移。

讓 `sum` 從 0 開始。每一步先用 `s[i] - '0'` 把字元轉成數值，再依位次決定加或減。走完整個字串後，`sum` 就是 A − B，再取絕對值。

```game-js
const controls = document.querySelector('#controls');
const input = document.querySelector('#digits');
const status = document.querySelector('#status');
const animation = document.querySelector('#animation');
const viewport = document.querySelector('#viewport');
const cells = document.querySelector('#cells');
const cursor = document.querySelector('#cursor');
const effect = document.querySelector('#effect');
const calculation = document.querySelector('#calculation');
const reducedMotion = matchMedia('(prefers-reduced-motion: reduce)');
let flash;
let runId = 0;

function reset() {
  runId += 1;
  animation.hidden = true;
  cells.className = '';
  cells.replaceChildren();
  cursor.hidden = true;
  flash?.cancel();
  calculation.textContent = '';
  return runId;
}

function pause(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function loadExample(digits) {
  reset();
  input.value = digits;
  status.textContent = '已載入 ' + digits.length + ' 位的範例，按「執行」開始。';
}

for (const button of controls.querySelectorAll('[data-example]')) {
  button.addEventListener('click', () => loadExample(button.dataset.example));
}

document.querySelector('#random-example').addEventListener('click', () => {
  const length = 1 + Math.floor(Math.random() * 100);
  let digits = String(1 + Math.floor(Math.random() * 9));
  for (let i = 1; i < length; i += 1) {
    digits += Math.floor(Math.random() * 10);
  }
  loadExample(digits);
});

const runButton = document.querySelector('#run');
input.addEventListener('keydown', (event) => {
  if (event.key === 'Enter' && !event.isComposing) runButton.click();
});

runButton.addEventListener('click', async () => {
  const currentRun = reset();
  const digits = input.value.trim();
  if (!/^[0-9]{1,1000}$/.test(digits) || !/[1-9]/.test(digits)) {
    status.textContent = '請輸入 1 到 1000 位的十進位正整數。';
    input.focus();
    return;
  }

  const steps = Array.from(digits, (digit, index) => {
    const place = digits.length - index;
    const odd = place % 2 === 1;
    const cell = document.createElement('div');
    cell.className = odd ? 'cell odd' : 'cell even';

    const label = document.createElement('small');
    label.textContent = 's[' + index + ']';
    const value = document.createElement('strong');
    value.textContent = digit;
    const position = document.createElement('small');
    position.textContent = '第 ' + index + ' 位';
    cell.append(label, value, position);
    return { digit, index, odd, cell };
  });

  cells.append(...steps.map(({ cell }) => cell));
  animation.hidden = false;
  viewport.scrollLeft = 0;
  status.textContent = '先從左到右排列字串索引。';
  await pause(500);
  if (currentRun !== runId) return;

  cells.classList.add('colored');
  status.textContent = '從右邊數位次，標出奇數位和偶數位。';
  await pause(600);
  if (currentRun !== runId) return;

  cells.classList.add('separated');
  status.textContent = '奇數位往上，偶數位往下。接著從左往右累加。';
  calculation.textContent = 'sum = 0';
  await pause(550);
  if (currentRun !== runId) return;

  let sum = 0;
  let previousCell;
  // 短字串有清楚的移動與停頓，長字串則逐位加速播放。
  const delay = Math.max(20, Math.min(750, Math.floor(25000 / steps.length)));
  const travel = Math.floor(delay * 0.42);
  cursor.style.setProperty('--travel', '0ms');
  cursor.style.left = '2rem';
  cursor.textContent = 'sum = 0';
  cursor.hidden = false;
  await pause(100);
  if (currentRun !== runId) return;
  cursor.style.setProperty('--travel', travel + 'ms');

  for (const { digit, index, odd, cell } of steps) {
    const center = cell.offsetLeft + cell.offsetWidth / 2;
    cursor.style.left = center + 'px';
    viewport.scrollTo({
      left: center - viewport.clientWidth / 2,
      behavior: reducedMotion.matches || delay < 350 ? 'instant' : 'smooth',
    });
    await pause(travel);
    if (currentRun !== runId) return;

    const value = digit.charCodeAt(0) - '0'.charCodeAt(0);
    const before = sum;
    sum += odd ? value : -value;
    previousCell?.classList.remove('active');
    cell.classList.add('active');
    previousCell = cell;

    cursor.textContent = 'sum = ' + sum;
    flash?.cancel();
    effect.className = odd ? 'odd' : 'even';
    effect.style.left = center + 'px';
    effect.textContent = (odd ? '＋' : '−') + value;
    const offset = odd ? '-3.6rem' : '3.6rem';
    flash = effect.animate([
      { opacity: 1, transform: 'translate(-50%, calc(-50% + ' + offset + '))' },
      { opacity: 0, transform: reducedMotion.matches ? undefined : 'translate(-50%, -50%)' },
    ], { duration: delay - travel, easing: 'ease-in' });
    calculation.textContent = 's[' + index + "]：'" + digit + "' − '0' = " + value
      + '，sum = ' + before + (odd ? ' ＋ ' : ' − ') + value + ' = ' + sum;
    await pause(delay - travel);
    if (currentRun !== runId) return;
  }

  const answer = Math.abs(sum);
  if (sum < 0) {
    cursor.textContent = '|' + sum + '|';
    calculation.textContent = 'sum 是負數，取絕對值 |' + sum + '|。';
    await pause(450);
    if (currentRun !== runId) return;
    cursor.textContent = 'abs = ' + answer;
    flash = cursor.animate(
      reducedMotion.matches
        ? [{ backgroundColor: '#965ac7' }, { backgroundColor: '#633c91' }]
        : [{ scale: 1 }, { scale: 1.2, backgroundColor: '#965ac7' }, { scale: 1 }],
      { duration: 450, easing: 'ease-in-out' },
    );
  }
  status.textContent = '計算完成。秘密差 = |' + sum + '| = ' + answer;
  calculation.textContent = '輸出：' + answer;
});
```
