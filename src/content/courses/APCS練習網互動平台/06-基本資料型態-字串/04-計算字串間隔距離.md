## 題目

給定長度為 2 到 100 的英文字母字串，以及一個至少出現兩次的目標字母。不分大小寫，依序輸出相鄰兩次出現的距離，以空格分隔。距離是後一次索引減前一次索引。

例如在 `ABCDAAEFeDaBDCBCBcBCbCBBd` 中搜尋 `A`，命中的索引依序為 0、4、5、10，所以輸出 `4 1 5`。

```game-html
  <label for="target">目標字母</label>
  <input id="target" value="A" autocomplete="off" spellcheck="false">
  <div class="buttons">
    <button id="run" type="button">執行</button>

    <button id="random" type="button">載入隨機範例</button>
```

```game-js
const source = document.querySelector('#source');
const target = document.querySelector('#target');
const status = document.querySelector('#status');
const viewport = document.querySelector('#viewport');
const cells = document.querySelector('#cells');
const scanner = document.querySelector('#scanner');
const step = document.querySelector('#step');
const result = document.querySelector('#result');
const sample = 'ABCDAAEFeDaBDCBCBcBCbCBBd';
let runId = 0;

function reset() {
  runId += 1;
  viewport.hidden = true;
  cells.replaceChildren();
  step.textContent = '';
  result.value = '';
  return runId;
}

function loadExample(text, letter) {
  reset();
  source.value = text;
  target.value = letter;
  status.textContent = '已載入範例，按「執行」開始。';
}

function pause(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

document.querySelector('#example').addEventListener('click', () => loadExample(sample, 'A'));
document.querySelector('#random').addEventListener('click', () => {
  const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  const letter = alphabet[Math.floor(Math.random() * alphabet.length)];
  const chars = Array.from({ length: 14 }, () => {
    const char = alphabet[Math.floor(Math.random() * alphabet.length)];
    return Math.random() < .5 ? char.toLowerCase() : char;
  });
  chars[Math.floor(Math.random() * 7)] = letter;
  chars[7 + Math.floor(Math.random() * 7)] = letter.toLowerCase();
  loadExample(chars.join(''), letter);
});

document.querySelector('#run').addEventListener('click', async () => {
  const currentRun = reset();
  const text = source.value;
  const letter = target.value;
  if (!/^[A-Za-z]{2,100}$/.test(text) || !/^[A-Za-z]$/.test(letter)) {
    status.textContent = '請輸入 2 到 100 個英文字母，以及一個目標字母。';
    return;
  }

  const normalized = text.toLowerCase();
  const key = letter.toLowerCase();
  const first = normalized.indexOf(key);
  if (first === -1 || normalized.indexOf(key, first + 1) === -1) {
    status.textContent = '目標字母至少要出現兩次（不分大小寫）。';
    return;
  }

  const tiles = Array.from(text, (char, index) => {
    const tile = document.createElement('div');
    tile.className = 'cell';
    const label = document.createElement('small');
    label.textContent = 's[' + index + ']';
    tile.append(label, char);
    return tile;
  });
  cells.append(...tiles);
  scanner.style.setProperty('--travel', '0ms');
  scanner.style.left = '2.7rem';
  scanner.textContent = '找 ' + letter;
  viewport.hidden = false;
  viewport.scrollLeft = 0;
  status.textContent = '正在找「' + letter + '」，大小寫視為相同。';
  await pause(240);
  if (currentRun !== runId) return;

  const travel = text.length <= 30 ? 190 : 85;
  scanner.style.setProperty('--travel', travel + 'ms');
  const distances = [];
  let previous = -1;

  for (let index = 0; index < text.length; index += 1) {
    const tile = tiles[index];
    const center = tile.offsetLeft + tile.offsetWidth / 2;
    scanner.style.left = center + 'px';
    viewport.scrollTo({ left: center - viewport.clientWidth / 2, behavior: 'instant' });
    await pause(travel);
    if (currentRun !== runId) return;
    tiles[index - 1]?.classList.remove('active');
    tile.classList.add('active');

    const hit = normalized[index] === key;
    if (hit) {
      tile.classList.add('hit', 'endpoint');
      if (previous === -1) {
        step.textContent = '首次找到「' + text[index] + '」，記住索引 ' + index + '。';
      } else {
        const distance = index - previous;
        distances.push(distance);
        result.value = distances.join(' ');
        step.textContent = '前一次索引 ' + previous + '，這一次索引 ' + index
          + '。距離 = ' + index + ' − ' + previous + ' = ' + distance;
      }
      await pause(text.length <= 30 ? 850 : 230);
      if (currentRun !== runId) return;
      tiles[previous]?.classList.remove('endpoint');
      previous = index;
    } else {
      await pause(65);
      if (currentRun !== runId) return;
    }
  }

  tiles[previous].classList.remove('endpoint');
  tiles.at(-1).classList.remove('active');
  scanner.textContent = '完成';
  status.textContent = '共找到 ' + (distances.length + 1) + ' 次，輸出 ' + distances.length + ' 段距離。';
});
```
