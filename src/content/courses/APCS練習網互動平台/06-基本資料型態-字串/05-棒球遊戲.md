## 題目

九位球員依棒次輪流打擊，第九棒之後回到第一棒。每次輪到一位球員，就讀取他的下一筆打擊結果。

- **安打**　`1B`、`2B`、`3B`、`HR` 分別讓擊球員和場上的跑壘員前進 1、2、3、4 個壘包。每位球員回到本壘就得 1 分。
- **出局**　`FO`、`GO`、`SO` 各算一個出局。跑壘員留在原位，每三個出局清空壘包，棒次繼續往下輪。

輸入共十行。前九行各以打擊次數開頭，後接該球員的打擊結果。第十行是要停止的累計出局數。輸出停止時的總得分。

## 觀察動畫

載入範例或貼上十行測資，再按「執行」。球員圓牌上的數字是棒次。範例 1 在第 3 個出局時停止，得分為 0。範例 2 使用相同打擊紀錄，在第 6 個出局時停止，得分為 5。

```game-js
const source = document.querySelector('#source');
const field = document.querySelector('#field');
const status = document.querySelector('#status');
const scoreDisplay = document.querySelector('#score');
const outsDisplay = document.querySelector('#outs');
const inningOuts = document.querySelector('#inning-outs');
const atBat = document.querySelector('#at-bat');
const log = document.querySelector('#log');
const result = document.querySelector('#result');
const lineupPanel = document.querySelector('#lineup');
const lineupBody = document.querySelector('#lineup-body');
const lineupCaption = document.querySelector('#lineup-caption');
const sample = "5 1B 1B FO GO 1B\n5 1B 2B FO FO SO\n4 SO HR SO 1B\n4 FO FO FO HR\n4 1B 1B 1B 1B\n4 GO GO 3B GO\n4 1B GO GO SO\n4 SO GO 2B 2B\n4 3B GO GO FO";
const positions = [[50, 82], [82, 50], [50, 18], [18, 50]];
const hitBases = { '1B': 1, '2B': 2, '3B': 3, HR: 4 };
const outCodes = new Set(['FO', 'GO', 'SO']);
let runId = 0;

function reset() {
  runId += 1;
  field.querySelectorAll('.player').forEach((player) => player.remove());
  scoreDisplay.textContent = '0';
  outsDisplay.textContent = '0';
  inningOuts.textContent = '0';
  atBat.textContent = '等待打擊';
  log.replaceChildren();
  lineupPanel.hidden = true;
  lineupBody.replaceChildren();
  result.hidden = true;
  return runId;
}

function pause(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function place(player, base) {
  const [left, top] = positions[base % 4];
  player.style.left = left + '%';
  player.style.top = top + '%';
}

function record(message) {
  const entry = document.createElement('li');
  entry.textContent = message;
  log.append(entry);
  log.scrollTop = log.scrollHeight;
}

function loadExample(outs) {
  reset();
  source.value = sample + '\n' + outs;
  status.textContent = '按「執行」，模擬到第 ' + outs + ' 個出局。';
}

for (const button of document.querySelectorAll('[data-outs]')) {
  button.addEventListener('click', () => loadExample(button.dataset.outs));
}
loadExample(3);

document.querySelector('#run').addEventListener('click', async () => {
  const currentRun = reset();
  const lines = source.value.trim().split(/\r?\n/);
  const limit = Number(lines[9]);
  const lineup = lines.slice(0, 9).map((line) => line.trim().split(/\s+/));
  const validLineup = lineup.every(([count, ...codes]) =>
    /^[1-5]$/.test(count) && codes.length === Number(count)
    && codes.every((code) => Object.hasOwn(hitBases, code) || outCodes.has(code)));
  if (lines.length !== 10 || !/^\d+$/.test(lines[9].trim()) || limit < 1 || limit > 27 || !validLineup) {
    status.textContent = '請輸入九行有效的打擊紀錄，以及一行 1 到 27 的目標出局數。';
    return;
  }

  const turns = [];
  let availableOuts = 0;
  for (let turn = 0; turn < 45 && availableOuts < limit; turn += 1) {
    const batter = turn % 9 + 1;
    const code = lineup[batter - 1][Math.floor(turn / 9) + 1];
    if (!code) break;
    turns.push({ batter, code, attempt: Math.floor(turn / 9) });
    if (outCodes.has(code)) availableOuts += 1;
  }
  if (availableOuts < limit) {
    status.textContent = '依棒次輪流讀取時，打擊紀錄不足以達到指定出局數。';
    return;
  }

  const battingCells = lineup.map(([count, ...codes], index) => {
    const row = lineupBody.insertRow();
    const heading = document.createElement('th');
    heading.scope = 'row';
    heading.textContent = (index + 1) + '（' + count + '次）';
    row.append(heading);
    return Array.from({ length: 5 }, (_, attempt) => {
      const cell = row.insertCell();
      cell.textContent = codes[attempt] ?? '—';
      return cell;
    });
  });
  lineupCaption.textContent = '打擊測資 · 目標 ' + limit + ' 出局';
  lineupPanel.hidden = false;
  status.textContent = '比賽開始，目標是累計 ' + limit + ' 個出局。';
  let runners = [];
  let score = 0;
  let outs = 0;
  for (const { batter, code, attempt } of turns) {
    const battingCell = battingCells[batter - 1][attempt];
    battingCell.className = 'current';
    const player = document.createElement('span');
    player.className = 'player';
    player.textContent = batter;
    place(player, 0);
    field.append(player);
    atBat.textContent = '第 ' + batter + ' 棒 · ' + code;
    await pause(740);
    if (currentRun !== runId) return;

    if (outCodes.has(code)) {
      outs += 1;
      outsDisplay.textContent = outs + ' / ' + limit;
      inningOuts.textContent = outs % 3 || 3;
      record('第 ' + batter + ' 棒 ' + code + '，累計 ' + outs + ' 個出局。');
      player.classList.add('leaving');
      if (outs % 3 === 0) {
        record('三個出局，清空壘包。');
        for (const runner of runners) runner.player.classList.add('leaving');
      }
      await pause(670);
      if (currentRun !== runId) return;
      player.remove();
      if (outs % 3 === 0) {
        for (const runner of runners) runner.player.remove();
        runners = [];
        inningOuts.textContent = '0';
      }
      battingCell.className = 'done';
      continue;
    }

    runners.push({ player, base: 0 });
    const before = score;
    for (let move = 0; move < hitBases[code]; move += 1) {
      for (const runner of runners) {
        runner.base += 1;
        place(runner.player, runner.base);
      }
      await pause(450);
      if (currentRun !== runId) return;
      for (const runner of runners) {
        if (runner.base === 4) {
          score += 1;
          runner.player.classList.add('scored', 'leaving');
        }
      }
      scoreDisplay.textContent = score;
      runners = runners.filter((runner) => runner.base < 4);
    }
    record('第 ' + batter + ' 棒 ' + code + '，這球得 ' + (score - before) + ' 分。');
    await pause(490);
    if (currentRun !== runId) return;
    field.querySelectorAll('.scored').forEach((runner) => runner.remove());
    battingCell.className = 'done';
  }

  status.textContent = '已達到第 ' + limit + ' 個出局，比賽模擬結束。';
  result.textContent = '輸出 ' + score;
  result.hidden = false;
});
```
