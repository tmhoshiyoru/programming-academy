```game-js
document.addEventListener('DOMContentLoaded', () => {
  let lcmStepsData = [];
  let dateStepsData = [];
  let currentLcmStep = 0;
  let currentDateStep = 0;

  const calcBtn = document.getElementById('calc-btn');
  const errorMsg = document.getElementById('error-msg');

  // 輔助函式：判斷閏年
  function isLeapYear(y) {
    return (y % 4 === 0 && y % 100 !== 0) || (y % 400 === 0);
  }

  // 輔助函式：取得該月天數
  function getDaysInMonth(y, m) {
    const days = [0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
    if (m === 2 && isLeapYear(y)) return 29;
    return days[m];
  }

  function padZero(n) {
    return String(n).padStart(2, '0');
  }

  // 輾轉相除法
  function calcGCD(a, b) {
    let steps = [];
    let x = a, y = b;
    while (y !== 0) {
      let r = x % y;
      steps.push(`${x} ÷ ${y} = ${Math.floor(x / y)} ... 餘數 ${r}`);
      x = y;
      y = r;
    }
    return { gcd: x, steps };
  }

  // 執行整體計算
  function calculateAll() {
    errorMsg.style.display = 'none';

    const numKids = parseInt(document.getElementById('num-kids').value, 10);
    const periodsStr = document.getElementById('periods').value.trim();
    const baseDateStr = document.getElementById('base-date').value.trim();

    const arr = periodsStr.split(/\s+/).map(Number).filter(n => !isNaN(n) && n > 0);

    if (arr.length !== numKids) {
      showError(`週期數量 (${arr.length}) 與小孩人數 (${numKids}) 不符合！`);
      return;
    }

    const dParts = baseDateStr.split('/');
    if (dParts.length !== 3) {
      showError('請輸入正確的日期格式 (YYYY/MM/DD)');
      return;
    }

    let y = parseInt(dParts[0], 10);
    let m = parseInt(dParts[1], 10);
    let d = parseInt(dParts[2], 10);

    if (isNaN(y) || isNaN(m) || isNaN(d) || m < 1 || m > 12 || d < 1 || d > getDaysInMonth(y, m)) {
      showError('輸入的日期不合法！');
      return;
    }

    // --- 第一階段：LCM 步驟建構 ---
    lcmStepsData = [];
    let curLCM = arr[0];
    lcmStepsData.push({
      title: '起始第一個週期數字',
      html: `初始比較基準為：<strong>${curLCM}</strong>`
    });

    for (let i = 1; i < arr.length; i++) {
      let nextNum = arr[i];
      let { gcd, steps } = calcGCD(curLCM, nextNum);
      let nextLCM = (curLCM * nextNum) / gcd;

      let gcdLog = steps.map(s => `<div>${s}</div>`).join('');
      lcmStepsData.push({
        title: `求 LCM(${curLCM}, ${nextNum})`,
        html: `
          <strong>1. 輾轉相除法求 GCD：</strong>
          <div class="code-box">${gcdLog}</div>
          得出 GCD(${curLCM}, ${nextNum}) = <strong>${gcd}</strong><br><br>
          <strong>2. 套用公式求 LCM：</strong><br>
          LCM = (${curLCM} × ${nextNum}) ÷ ${gcd} = <strong>${nextLCM}</strong>
        `
      });
      curLCM = nextLCM;
    }

    lcmStepsData.push({
      isFinal: true,
      title: 'LCM 計算完成',
      html: `所有小孩週期的最小公倍數為：<strong>${curLCM} 天</strong>。<br>意思是每隔 ${curLCM} 天所有小孩就會同時回家。`
    });

    // --- 第二階段：日期逐月拆解步驟建構 ---
    dateStepsData = [];
    let remDays = curLCM;
    let curY = y, curM = m, curD = d;

    dateStepsData.push({
      title: '推算的起點與總天數',
      html: `起始日期：<strong>${curY}/${padZero(curM)}/${padZero(curD)}</strong><br>需要推算增加的天數：<strong>${remDays} 天</strong>`
    });

    let isFirstMonth = true;

    while (remDays > 0) {
      let monthDays = getDaysInMonth(curY, curM);

      if (isFirstMonth) {
        let daysToFill = monthDays - curD; // 當月剩餘可走天數
        if (remDays <= daysToFill) {
          curD += remDays;
          dateStepsData.push({
            isFinal: true,
            title: `在當月 (${curY}年${curM}月) 即可完成`,
            html: `剩餘 ${remDays} 天 &le; 當月剩餘天數 (${daysToFill} 天)<br>直接加在日數上：${curD - remDays} + ${remDays} = <strong>${curD} 日</strong>`
          });
          remDays = 0;
        } else {
          remDays -= (daysToFill + 1); // 加上跨到下個月1號的1天
          dateStepsData.push({
            title: `1. 填滿首月：${curY} 年 ${curM} 月`,
            html: `當月共有 ${monthDays} 天。從 ${curD} 號走到月底並進入下個月 1 號需消耗 <strong>${daysToFill + 1}</strong> 天。<br>
                   剩餘天數變更為：<div class="code-box">${remDays + daysToFill + 1} - ${daysToFill + 1} = ${remDays} 天</div>`
          });
          curM++;
          if (curM > 12) { curM = 1; curY++; }
          curD = 1;
        }
        isFirstMonth = false;
      } else {
        let isLeap = (curM === 2 && isLeapYear(curY));
        if (remDays >= monthDays) {
          remDays -= monthDays;
          dateStepsData.push({
            title: `跨越整個月：${curY} 年 ${curM} 月 ${isLeap ? '<span class="badge">閏年 29 天</span>' : ''}`,
            html: `${curM} 月共 ${monthDays} 天，扣除整個月天數。<br>
                   剩餘天數變更為：<div class="code-box">${remDays + monthDays} - ${monthDays} = ${remDays} 天</div>`
          });
          curM++;
          if (curM > 12) { curM = 1; curY++; }
        } else {
          curD = curD + remDays;
          dateStepsData.push({
            isFinal: true,
            title: `落入最終月份：${curY} 年 ${curM} 月`,
            html: `剩餘天數 ${remDays} 天小於當月總天數 (${monthDays} 天)。<br>
                   從 1 號開始推進 ${remDays} 天：<br>
                   🎉 最終答案為：<strong>${curY}/${padZero(curM)}/${padZero(curD)}</strong>`
          });
          remDays = 0;
        }
      }
    }

    // 更新摘要
    document.getElementById('sum-lcm').textContent = `${curLCM} 天`;
    document.getElementById('sum-date').textContent = `${curY}/${padZero(curM)}/${padZero(curD)}`;

    // 重置並渲染卡片
    currentLcmStep = 0;
    currentDateStep = 0;
    renderLcmCard();
    renderDateCard();
  }

  function showError(msg) {
    errorMsg.textContent = msg;
    errorMsg.style.display = 'block';
  }

  // 渲染 LCM 卡片
  function renderLcmCard() {
    const card = document.getElementById('lcm-card-display');
    const data = lcmStepsData[currentLcmStep];
    card.className = `interactive-card ${data.isFinal ? 'final' : ''}`;
    card.innerHTML = `<div class="card-title">${data.title}</div>${data.html}`;

    document.getElementById('lcm-step-indicator').textContent = `步驟 ${currentLcmStep + 1} / ${lcmStepsData.length}`;
    document.getElementById('lcm-prev-btn').disabled = (currentLcmStep === 0);
    document.getElementById('lcm-next-btn').disabled = (currentLcmStep === lcmStepsData.length - 1);
  }

  // 渲染 Date 卡片
  function renderDateCard() {
    const card = document.getElementById('date-card-display');
    const data = dateStepsData[currentDateStep];
    card.className = `interactive-card ${data.isFinal ? 'final' : ''}`;
    card.innerHTML = `<div class="card-title">${data.title}</div>${data.html}`;

    document.getElementById('date-step-indicator').textContent = `步驟 ${currentDateStep + 1} / ${dateStepsData.length}`;
    document.getElementById('date-prev-btn').disabled = (currentDateStep === 0);
    document.getElementById('date-next-btn').disabled = (currentDateStep === dateStepsData.length - 1);
  }

  // 頁籤切換
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      document.querySelectorAll('.tab-pane').forEach(p => p.classList.remove('active'));

      btn.classList.add('active');
      document.getElementById(btn.dataset.tab).classList.add('active');
    });
  });

  // 按鈕事件監聽
  document.getElementById('lcm-prev-btn').addEventListener('click', () => {
    if (currentLcmStep > 0) { currentLcmStep--; renderLcmCard(); }
  });
  document.getElementById('lcm-next-btn').addEventListener('click', () => {
    if (currentLcmStep < lcmStepsData.length - 1) { currentLcmStep++; renderLcmCard(); }
  });

  document.getElementById('date-prev-btn').addEventListener('click', () => {
    if (currentDateStep > 0) { currentDateStep--; renderDateCard(); }
  });
  document.getElementById('date-next-btn').addEventListener('click', () => {
    if (currentDateStep < dateStepsData.length - 1) { currentDateStep++; renderDateCard(); }
  });

  calcBtn.addEventListener('click', calculateAll);

  // 初始化首次執行
  calculateAll();
});
```
