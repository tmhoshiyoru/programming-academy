```game-js
// HTML 元素綁定
const kSlider = document.getElementById('kSlider');
const kValue = document.getElementById('kValue');
const calcRangeBtn = document.getElementById('calcRangeBtn');
const mathRangeDisplay = document.getElementById('mathRangeDisplay');
const rangeResultText = document.getElementById('rangeResultText');
const evenOnlyCheck = document.getElementById('evenOnlyCheck');

const startBtn = document.getElementById('startBtn');
const stepBtn = document.getElementById('stepBtn');
const resetBtn = document.getElementById('resetBtn');

const currentI = document.getElementById('currentI');
const currentSquare = document.getElementById('currentSquare');
const digitsContainer = document.getElementById('digitsContainer');
const statusBox = document.getElementById('statusBox');
const logPanel = document.getElementById('logPanel');

// 變數設定
let k = 3;
let minI = 0n;
let maxI = 0n;
let currentVal = 0n;
let isRangeCalculated = false;
let isRunning = false;
let autoTimer = null;
let foundResult = false;

// 格式化顯示 10 的次方的 HTML
function formatPower(pow) {
  if (pow === 0) return '1';
  return `10<sup>${pow}</sup>`;
}

// 精確計算 BigInt 平方根
function bigintSqrt(value) {
  if (value < 0n) return 0n;
  if (value === 0n) return 0n;
  let x0 = value / 2n;
  if (x0 === 0n) x0 = 1n;
  let x1 = (x0 + value / x0) / 2n;
  while (x1 < x0) {
    x0 = x1;
    x1 = (x0 + value / x0) / 2n;
  }
  return x0;
}

// 向上取整的 BigInt 開根號
function bigintCeilSqrt(value) {
  const sqrt = bigintSqrt(value);
  if (sqrt * sqrt === value) {
    return sqrt;
  }
  return sqrt + 1n;
}

// 展現數學算式與更新 UI
function updateMathFormula() {
  k = parseInt(kSlider.value);
  kValue.textContent = k;

  const minPow = k - 1;
  const maxPow = k;
  
  if (k === 1) {
    mathRangeDisplay.innerHTML = `平方數範圍：1 &le; i<sup>2</sup> &le; 9`;
  } else {
    mathRangeDisplay.innerHTML = `平方數範圍：${formatPower(minPow)} &le; i<sup>2</sup> &le; (${formatPower(maxPow)} - 1)`;
  }

  resetSimulation();
}

// 步驟 1：開根號確定範圍
function calculateRange() {
  k = parseInt(kSlider.value);
  let minSquare = 10n ** BigInt(k - 1);
  let maxSquare = (10n ** BigInt(k)) - 1n;

  if (k === 1) {
    minSquare = 1n;
  }

  minI = bigintCeilSqrt(minSquare);
  maxI = bigintSqrt(maxSquare);

  // 若勾選「僅測試偶數」，且起點為奇數，則主動向後調整為下一個偶數
  if (evenOnlyCheck.checked && minI % 2n !== 0n) {
    minI += 1n;
  }

  currentVal = minI;

  rangeResultText.innerHTML = `&radic;${minSquare} &le; i &le; &radic;${maxSquare} &nbsp;&Rightarrow;&nbsp; <b>${minI} &le; i &le; ${maxI}</b>`;
  
  const modeText = evenOnlyCheck.checked ? ' (僅測試偶數底數)' : ' (連續測試所有整數)';
  addLog(`完成開根號：底數 i 搜尋範圍訂為 ${minI} ~ ${maxI}${modeText}`, 'info');

  isRangeCalculated = true;
  startBtn.disabled = false;
  stepBtn.disabled = false;
  calcRangeBtn.disabled = true;
  kSlider.disabled = true;
  evenOnlyCheck.disabled = true;

  statusBox.textContent = `範圍已確定 (i=${minI}~${maxI})，請點擊「自動搜尋」或「單步執行」`;
  digitsContainer.innerHTML = '<span class="placeholder-text">點擊「自動搜尋」開始尋找偶數完全平方數</span>';
}

// 檢查是否全為偶數
function isAllEvenDigits(numBigInt) {
  const str = numBigInt.toString();
  const result = {
    isAllEven: true,
    digits: []
  };

  for (let i = 0; i < str.length; i++) {
    const digit = parseInt(str[i]);
    const isEven = digit % 2 === 0;
    result.digits.push({ digit, isEven });
    if (!isEven) {
      result.isAllEven = false;
    }
  }
  return result;
}

// Log 紀錄
function addLog(msg, type = '') {
  const entry = document.createElement('div');
  entry.className = `log-entry ${type}`;
  entry.textContent = `> ${msg}`;
  logPanel.appendChild(entry);
  logPanel.scrollTop = logPanel.scrollHeight;
}

// 渲染數字卡片
function renderDigits(digitsInfo) {
  digitsContainer.innerHTML = '';
  digitsInfo.digits.forEach(item => {
    const box = document.createElement('div');
    box.className = `digit-box ${item.isEven ? 'even' : 'odd'}`;
    box.textContent = item.digit;
    digitsContainer.appendChild(box);
  });
}

// 執行一步測試
function step() {
  if (currentVal > maxI || foundResult) {
    stopAutoRun();
    return;
  }

  const square = currentVal * currentVal;
  currentI.textContent = currentVal.toString();
  currentSquare.textContent = square.toString();

  const check = isAllEvenDigits(square);
  renderDigits(check);

  if (check.isAllEven) {
    foundResult = true;
    statusBox.textContent = `找到解答！k=${k} 時最小全偶數平方數為：${square}`;
    statusBox.className = 'status-box success';
    addLog(`成功！找到最小全偶數平方數：${square} (底數 i = ${currentVal})`, 'success');
    stopAutoRun();
    stepBtn.disabled = true;
    startBtn.disabled = true;
  } else {
    addLog(`測試 i=${currentVal}, i²=${square} -> 含有奇數，遞增搜尋...`, 'fail');
    
    // 依據勾選狀態決定步長 (僅偶數則 +2，否則 +1)
    const stepSize = evenOnlyCheck.checked ? 2n : 1n;
    currentVal += stepSize;

    if (currentVal > maxI) {
      statusBox.textContent = '搜尋結束，該範圍內無符合條件的數。';
      stopAutoRun();
      stepBtn.disabled = true;
    }
  }
}

// 開始自動搜尋
function startAutoRun() {
  if (foundResult || !isRangeCalculated) return;
  isRunning = true;
  startBtn.textContent = '暫停搜尋';
  stepBtn.disabled = true;
  statusBox.textContent = '正在搜尋中...';
  statusBox.className = 'status-box searching';
  
  autoTimer = setInterval(() => {
    step();
  }, 500); 
}

// 暫停自動搜尋
function stopAutoRun() {
  isRunning = false;
  clearInterval(autoTimer);
  autoTimer = null;
  startBtn.textContent = '自動搜尋';
  if (!foundResult && currentVal <= maxI && isRangeCalculated) {
    stepBtn.disabled = false;
  }
}

// 重置狀態
function resetSimulation() {
  stopAutoRun();
  isRangeCalculated = false;
  foundResult = false;
  
  currentI.textContent = '-';
  currentSquare.textContent = '-';
  digitsContainer.innerHTML = '<span class="placeholder-text">請先完成步驟 1 確定範圍後開始搜尋</span>';
  rangeResultText.textContent = '點擊上方按鈕計算底數 i 的搜尋區間';

  statusBox.textContent = '請調整 k 值並點擊「開根號確定底數範圍」';
  statusBox.className = 'status-box';
  
  logPanel.innerHTML = '';
  addLog(`設定位數 k=${k}，等待計算搜尋範圍...`);
  
  calcRangeBtn.disabled = false;
  startBtn.disabled = true;
  stepBtn.disabled = true;
  kSlider.disabled = false;
  evenOnlyCheck.disabled = false;
}

// 事件監聽
kSlider.addEventListener('input', updateMathFormula);
calcRangeBtn.addEventListener('click', calculateRange);

startBtn.addEventListener('click', () => {
  if (isRunning) {
    stopAutoRun();
  } else {
    startAutoRun();
  }
});

stepBtn.addEventListener('click', () => {
  stopAutoRun();
  step();
});

resetBtn.addEventListener('click', resetSimulation);

// 初始化
updateMathFormula();
```
