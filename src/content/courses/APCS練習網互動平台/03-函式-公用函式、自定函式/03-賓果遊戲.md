```game-js
const board = [
  1, 9, 23, 17, 18,
  10, 24, 8, 16, 2,
  11, 13, 3, 7, 19,
  12, 4, 15, 6, 20,
  25, 14, 5, 22, 21
];

// 題目範例 1：選擇 10 可得 2 分
const ex1Called = [1, 9, 23, 17, 24, 8, 16, 2, 11, 13, 3, 7, 19, 12, 4, 15, 20, 25, 14, 5, 22, 21];

// 題目範例 2：選擇 6 和 10 均可得 2 分 (瑞奇會選比較小的 6)
const ex2Called = [1, 9, 23, 17, 24, 8, 16, 2, 11, 13, 3, 7, 19, 12, 4, 15, 20, 25, 14, 5, 22];

let marked = Array(25).fill(false);
let selectedCandidateNum = null;

const boardEl = document.getElementById('bingo-board');
const lineCanvas = document.getElementById('line-canvas');
const markedCountEl = document.getElementById('marked-count');
const baseScoreEl = document.getElementById('base-score');
const bestChoiceEl = document.getElementById('best-choice');
const analysisDetailEl = document.getElementById('analysis-detail');
const candidatesTbody = document.getElementById('candidates-tbody');

const btnEx1 = document.getElementById('btn-ex1');
const btnEx2 = document.getElementById('btn-ex2');

function loadState(calledArray) {
  marked = Array(25).fill(false);
  calledArray.forEach(num => {
    const idx = board.indexOf(num);
    if (idx !== -1) marked[idx] = true;
  });
  selectedCandidateNum = null;
  render();
}

function checkLines(currentMarked) {
  let completedLines = [];

  // 1. 檢查 5 列 (Rows)
  for (let r = 0; r < 5; r++) {
    let complete = true;
    let indices = [];
    for (let c = 0; c < 5; c++) {
      let idx = r * 5 + c;
      indices.push(idx);
      if (!currentMarked[idx]) complete = false;
    }
    if (complete) completedLines.push({ type: 'row', index: r, indices });
  }

  // 2. 檢查 5 行 (Columns)
  for (let c = 0; c < 5; c++) {
    let complete = true;
    let indices = [];
    for (let r = 0; r < 5; r++) {
      let idx = r * 5 + c;
      indices.push(idx);
      if (!currentMarked[idx]) complete = false;
    }
    if (complete) completedLines.push({ type: 'col', index: c, indices });
  }

  // 3. 主對角線 (Diagonal 1)
  let d1Indices = [0, 6, 12, 18, 24];
  if (d1Indices.every(idx => currentMarked[idx])) {
    completedLines.push({ type: 'diag1', indices: d1Indices });
  }

  // 4. 副對角線 (Diagonal 2)
  let d2Indices = [4, 8, 12, 16, 20];
  if (d2Indices.every(idx => currentMarked[idx])) {
    completedLines.push({ type: 'diag2', indices: d2Indices });
  }

  return completedLines;
}

function getBestChoice(candidates) {
  if (candidates.length === 0) return null;
  let best = candidates[0];
  for (let i = 1; i < candidates.length; i++) {
    if (candidates[i].score > best.score) {
      best = candidates[i];
    } else if (candidates[i].score === best.score) {
      if (candidates[i].num < best.num) {
        best = candidates[i];
      }
    }
  }
  return best;
}

function render() {
  const baseLines = checkLines(marked);
  const baseScore = baseLines.length;
  
  // 計算所有待選號碼的得分
  let candidates = [];
  for (let num = 1; num <= 25; num++) {
    const idx = board.indexOf(num);
    if (idx !== -1 && !marked[idx]) {
      let tempMarked = [...marked];
      tempMarked[idx] = true;
      const linesAfter = checkLines(tempMarked);
      candidates.push({
        num,
        idx,
        score: linesAfter.length,
        linesAfter
      });
    }
  }

  const best = getBestChoice(candidates);

  // 更新左側資訊
  markedCountEl.textContent = marked.filter(Boolean).length;
  baseScoreEl.textContent = `${baseScore} 分`;
  bestChoiceEl.textContent = best ? `${best.num} (可得 ${best.score} 分)` : "無";

  // 繪製盤面
  boardEl.innerHTML = '';
  for (let i = 0; i < 25; i++) {
    const num = board[i];
    const cell = document.createElement('div');
    cell.className = 'cell';
    cell.textContent = num;

    if (marked[i]) {
      cell.classList.add('marked');
    } else {
      if (best && num === best.num) {
        cell.classList.add('best-tag');
      }
      if (selectedCandidateNum === num) {
        cell.classList.add('selected');
      }
      cell.addEventListener('click', () => {
        selectedCandidateNum = num;
        render();
      });
    }
    boardEl.appendChild(cell);
  }

  // 渲染待選表格
  candidatesTbody.innerHTML = '';
  candidates.sort((a, b) => a.num - b.num);
  candidates.forEach(c => {
    const tr = document.createElement('tr');
    if (selectedCandidateNum === c.num) tr.classList.add('active');
    if (best && c.num === best.num) tr.classList.add('best-row');

    const isBest = best && c.num === best.num;
    tr.innerHTML = `
      <td>${c.num}</td>
      <td>${c.score} 分</td>
      <td>${isBest ? '★ 瑞奇的最佳選擇' : (c.score > baseScore ? '有額外連線' : '無連線')}</td>
    `;
    tr.style.cursor = 'pointer';
    tr.addEventListener('click', () => {
      selectedCandidateNum = c.num;
      render();
    });
    candidatesTbody.appendChild(tr);
  });

  // 渲染點選詳情與 SVG 連線
  renderAnalysisAndLines(candidates, best, baseScore);
}

function renderAnalysisAndLines(candidates, best, baseScore) {
  lineCanvas.innerHTML = '';

  let activeMarked = [...marked];
  let currentCandidate = null;

  if (selectedCandidateNum !== null) {
    currentCandidate = candidates.find(c => c.num === selectedCandidateNum);
    if (currentCandidate) {
      activeMarked[currentCandidate.idx] = true;
    }
  }

  const activeLines = checkLines(activeMarked);

  if (currentCandidate) {
    const scoreDiff = currentCandidate.score - baseScore;
    let detailText = `<strong>試算號碼：[ ${currentCandidate.num} ]</strong><br/>`;
    detailText += `標記前分數：${baseScore} 分<br/>`;
    detailText += `標記後總分數：<strong>${currentCandidate.score} 分</strong> (淨增加 ${scoreDiff} 分)<br/>`;
    
    if (best && currentCandidate.num === best.num) {
      detailText += `<span style="color: #16a34a; font-weight: bold;">符合瑞奇策略：此為最高分且數值最小的選擇！</span>`;
    } else {
      detailText += `<span style="color: #d97706;">非最佳解。最佳選擇為 [ ${best ? best.num : '--'} ]。</span>`;
    }
    analysisDetailEl.innerHTML = detailText;
  } else {
    analysisDetailEl.innerHTML = `請點擊右側盤面上任意「白底待選號碼」以查看連線試算。`;
  }

  // 繪製劃線 (SVG)
  if (activeLines.length > 0 && boardEl.children.length === 25) {
    const boardRect = boardEl.getBoundingClientRect();
    
    activeLines.forEach(line => {
      const startCell = boardEl.children[line.indices[0]].getBoundingClientRect();
      const endCell = boardEl.children[line.indices[4]].getBoundingClientRect();

      const x1 = startCell.left + startCell.width / 2 - boardRect.left;
      const y1 = startCell.top + startCell.height / 2 - boardRect.top;
      const x2 = endCell.left + endCell.width / 2 - boardRect.left;
      const y2 = endCell.top + endCell.height / 2 - boardRect.top;

      const svgLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      svgLine.setAttribute('x1', x1);
      svgLine.setAttribute('y1', y1);
      svgLine.setAttribute('x2', x2);
      svgLine.setAttribute('y2', y2);
      svgLine.setAttribute('stroke', '#ef4444');
      svgLine.setAttribute('stroke-width', '4');
      svgLine.setAttribute('stroke-linecap', 'round');
      svgLine.setAttribute('opacity', '0.85');
      lineCanvas.appendChild(svgLine);
    });
  }
}

btnEx1.addEventListener('click', () => loadState(ex1Called));
btnEx2.addEventListener('click', () => loadState(ex2Called));

// 預設載入範例 1
loadState(ex1Called);
```
