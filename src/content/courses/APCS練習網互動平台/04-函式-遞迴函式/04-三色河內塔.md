```game-js
let totalRings = 3;
let pegs = { A: [], B: [], C: [] };
let moves = [];
let currentStep = 0;
let isAutoPlaying = false;
let autoPlayTimer = null;

const diskCountSelect = document.getElementById('disk-count');
const btnAutoPlay = document.getElementById('btn-auto-play');
const btnNextStep = document.getElementById('btn-next-step');
const btnReset = document.getElementById('btn-reset');
const messageBanner = document.getElementById('message-banner');
const logContent = document.getElementById('log-content');
const currentStageBadge = document.getElementById('current-stage-badge');
const currentExplanationText = document.getElementById('current-explanation-text');

function getDiskColorClass(diskId) {
  const mod = diskId % 3;
  if (mod === 1) return 'flower';
  if (mod === 2) return 'white';
  return 'gray';
}

function init() {
  totalRings = parseInt(diskCountSelect.value);
  stopAutoPlay();

  pegs = { A: [], B: [], C: [] };
  for (let i = totalRings; i >= 1; i--) {
    pegs.A.push(i);
  }

  currentStep = 0;
  moves = [];

  generateMovesFromCppLogic(totalRings);

  renderDisks();
  updateLogText();
  highlightPhaseCard(null);
  messageBanner.textContent = `初始化完成：共 ${totalRings} 個套環。已載入標準遞迴解析流程。`;
  currentStageBadge.textContent = "準備就緒";
  currentExplanationText.textContent = `點擊「單步移動」或「自動播放」，觀看如何將 ${totalRings} 個套環逐步解開！`;
}

function generateMovesFromCppLogic(num) {
  moves = [];

  function recordMove(ring, from, to, type, info) {
    moves.push({ ring, from, to, type, info });
  }

  function hanoi(x, y, z, n) {
    if (n === 1) {
      recordMove(1, x, z, 'HANOI', `將頂層套環 1 從 ${x} 搬到 ${z}`);
    } else if (n > 1) {
      hanoi(x, z, y, n - 1);
      recordMove(n, x, z, 'HANOI', `標準河內塔遞迴：將套環 ${n} 從 ${x} 搬到 ${z}`);
      hanoi(y, x, z, n - 1);
    }
  }

  function hanoi3(position, n) {
    if (n !== 0) {
      if (position === 'A') {
        hanoi('A', 'C', 'B', n - 3);
        recordMove(n - 2, 'A', 'C', 'PHASE_A', `[柱A步驟1] 將花環 (${n-2}) 暫存移至 C`);
        hanoi('B', 'A', 'C', n - 3);
        recordMove(n - 1, 'A', 'B', 'PHASE_A', `[柱A步驟2] 將白環 (${n-1}) 歸位移至 B`);
        hanoi('C', 'B', 'A', n - 3);
        recordMove(n - 2, 'C', 'B', 'PHASE_A', `[柱A步驟3] 將花環 (${n-2}) 疊上 B 讓出 C`);
        hanoi('A', 'C', 'B', n - 3);
        recordMove(n, 'A', 'C', 'PHASE_A', `[柱A步驟4] 將灰環 (${n}) 歸位移至 C`);
        hanoi('B', 'A', 'C', n - 3);
        recordMove(n - 2, 'B', 'A', 'PHASE_A', `[柱A步驟5] 將花環 (${n-2}) 歸位移至 A`);
        hanoi3('C', n - 3);
      } else if (position === 'C') {
        hanoi('C', 'A', 'B', n - 3);
        recordMove(n - 2, 'C', 'A', 'PHASE_C', `[柱C步驟1] 將花環 (${n-2}) 歸位移至 A`);
        hanoi('B', 'C', 'A', n - 3);
        recordMove(n - 1, 'C', 'B', 'PHASE_C', `[柱C步驟2] 將白環 (${n-1}) 歸位移至 B`);
        hanoi3('A', n - 3);
      }
    }
  }

  hanoi3('A', num);
}

function highlightPhaseCard(type) {
  document.getElementById('card-phase-hanoi').classList.remove('active');
  document.getElementById('card-phase-A').classList.remove('active');
  document.getElementById('card-phase-C').classList.remove('active');

  if (type === 'HANOI') {
    document.getElementById('card-phase-hanoi').classList.add('active');
  } else if (type === 'PHASE_A') {
    document.getElementById('card-phase-A').classList.add('active');
  } else if (type === 'PHASE_C') {
    document.getElementById('card-phase-C').classList.add('active');
  }
}

function renderDisks() {
  document.querySelectorAll('.disk').forEach(el => el.remove());
  const container = document.getElementById('pegs-container');
  const containerWidth = container.clientWidth;
  const containerHeight = container.clientHeight;
  const pegAreaWidth = containerWidth * 0.32;

  const minDiskWidthPercent = 25;
  const maxDiskWidthPercent = 90;

  const availableHeight = containerHeight - 50;
  const rawDiskHeight = Math.floor(availableHeight / totalRings);
  const diskHeight = Math.max(4, Math.min(18, rawDiskHeight));
  const fontSize = diskHeight >= 11 ? '0.72rem' : '0px';

  const pegKeys = ['A', 'B', 'C'];

  pegKeys.forEach((pegKey, pegIndex) => {
    const pegList = pegs[pegKey];
    pegList.forEach((diskSize, diskIndex) => {
      const diskEl = document.createElement('div');
      diskEl.className = `disk ${getDiskColorClass(diskSize)}`;

      if (fontSize !== '0px') {
        diskEl.textContent = diskSize;
        diskEl.style.fontSize = fontSize;
      }

      let widthRatio;
      if (totalRings === 1) {
        widthRatio = maxDiskWidthPercent / 100;
      } else {
        const stepRatio = (diskSize - 1) / (totalRings - 1);
        widthRatio = (minDiskWidthPercent + stepRatio * (maxDiskWidthPercent - minDiskWidthPercent)) / 100;
      }

      const diskPixelWidth = pegAreaWidth * widthRatio;
      diskEl.style.width = `${diskPixelWidth}px`;
      diskEl.style.height = `${diskHeight}px`;
      diskEl.style.borderRadius = `${Math.min(4, diskHeight / 2)}px`;

      const pegCenter = (pegIndex + 0.5) * (containerWidth / 3);
      const diskLeft = pegCenter - (diskPixelWidth / 2);
      diskEl.style.left = `${diskLeft}px`;

      const bottomOffset = 18 + diskIndex * (diskHeight + (totalRings > 10 ? 1 : 2));
      diskEl.style.bottom = `${bottomOffset}px`;

      container.appendChild(diskEl);
    });
  });
}

function executeStep() {
  if (currentStep >= moves.length) {
    stopAutoPlay();
    messageBanner.textContent = "🎉 所有套環已精確分流歸位！(花色在 A，白色在 B，灰色在 C)";
    currentStageBadge.textContent = "任務完成";
    currentExplanationText.textContent = `成功以最小步數 ${moves.length} 步完美解開三色河內塔！`;
    highlightPhaseCard(null);
    return;
  }

  const step = moves[currentStep];
  const disk = pegs[step.from].pop();
  pegs[step.to].push(disk);

  currentStep++;
  renderDisks();
  updateLogText();

  highlightPhaseCard(step.type);

  messageBanner.textContent = `步驟 ${currentStep} / ${moves.length}：ring ${step.ring} : ${step.from} => ${step.to}`;

  if (step.type === 'HANOI') {
    currentStageBadge.textContent = "經典河內塔移開上方";
  } else if (step.type === 'PHASE_A') {
    currentStageBadge.textContent = "柱 A 核心 5 步處置";
  } else if (step.type === 'PHASE_C') {
    currentStageBadge.textContent = "柱 C 核心 2 步處置";
  }
  currentExplanationText.textContent = step.info;
}

function updateLogText() {
  if (currentStep === 0) {
    logContent.textContent = "等待執行...";
    return;
  }

  let text = "";
  for (let i = 0; i < currentStep; i++) {
    const m = moves[i];
    text += `ring ${m.ring} : ${m.from} => ${m.to}\n`;
  }

  if (currentStep === moves.length) {
    text += `共需${moves.length}個移動`;
  } else {
    text += `\n[進行中... 目前第 ${currentStep}/${moves.length} 步]`;
  }

  logContent.textContent = text;
  const box = document.getElementById('output-box');
  box.scrollTop = box.scrollHeight;
}

function stopAutoPlay() {
  isAutoPlaying = false;
  btnAutoPlay.textContent = "▶ 自動播放";
  btnAutoPlay.classList.remove('secondary');
  btnAutoPlay.classList.add('primary');
  if (autoPlayTimer) {
    clearInterval(autoPlayTimer);
    autoPlayTimer = null;
  }
}

function startAutoPlay() {
  if (currentStep >= moves.length) currentStep = 0;
  isAutoPlaying = true;
  btnAutoPlay.textContent = "⏸ 暫停播放";
  btnAutoPlay.classList.remove('primary');
  btnAutoPlay.classList.add('secondary');

  executeStep();
  const speed = totalRings > 6 ? 80 : 450;
  autoPlayTimer = setInterval(() => {
    if (currentStep < moves.length) {
      executeStep();
    } else {
      stopAutoPlay();
    }
  }, speed);
}

diskCountSelect.addEventListener('change', init);

btnAutoPlay.addEventListener('click', () => {
  if (isAutoPlaying) {
    stopAutoPlay();
  } else {
    startAutoPlay();
  }
});

btnNextStep.addEventListener('click', () => {
  if (isAutoPlaying) stopAutoPlay();
  executeStep();
});

btnReset.addEventListener('click', init);

window.addEventListener('resize', renderDisks);

init();
```
