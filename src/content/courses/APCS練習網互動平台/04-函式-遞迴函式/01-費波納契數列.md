```game-js
document.addEventListener('DOMContentLoaded', () => {
  const nInput = document.getElementById('n-input');
  const calcBtn = document.getElementById('calc-btn');
  const stepBtn = document.getElementById('step-btn');
  
  const resultVal = document.getElementById('result-val');
  const callsVal = document.getElementById('calls-val');
  const cppTimeVal = document.getElementById('cpp-time-val');
  const warningBox = document.getElementById('warning-box');
  
  const treeViewport = document.getElementById('tree-viewport');
  const treeWrapper = document.getElementById('tree-wrapper');
  const treeSvg = document.getElementById('tree-edges');
  const treeNodes = document.getElementById('tree-nodes');
  const barChart = document.getElementById('bar-chart');

  let animationTimer = null;

  calcBtn.addEventListener('click', () => {
    let n = parseInt(nInput.value);
    if (isNaN(n) || n < 0) n = 0;
    if (n > 100) n = 100;
    nInput.value = n;

    runAnalysis(n);
    renderTree(n, false);
  });

  stepBtn.addEventListener('click', () => {
    let n = parseInt(nInput.value);
    if (isNaN(n) || n < 0) n = 0;
    if (n > 100) n = 100;
    nInput.value = n;

    runAnalysis(n);
    renderTree(n, true);
  });

  // 精確使用 BigInt 運算，支援 N 到 100 以上的大數與呼叫次數
  function calculateMetrics(n) {
    let fib = [0n, 1n];
    for (let i = 2; i <= n + 1; i++) {
      fib[i] = fib[i - 1] + fib[i - 2];
    }

    const value = fib[n];
    
    // 總呼叫次數公式：TotalCalls(N) = 2 * Fib(N+1) - 1
    const totalCalls = 2n * fib[n + 1] - 1n;

    // 計算每一個 F(k) 被呼叫的次數：
    // 對於 k >= 1，F(k) 在 F(N) 樹中出現的次數等於 Fib(N - k + 1)
    // 對於 k = 0，被呼叫次數等於 Fib(N - 1) (當 N>=1)
    const subproblemCounts = {};
    for (let k = n; k >= 0; k--) {
      if (k === 0) {
        subproblemCounts[k] = n === 0 ? 1n : fib[n - 1];
      } else {
        subproblemCounts[k] = fib[n - k + 1];
      }
    }

    return { value, totalCalls, subproblemCounts };
  }

  // 估算 C++ 執行時間（假設現代 CPU 極速 C++ 每秒可跑 5 * 10^8 次遞迴呼叫）
  function estimateCppTime(totalCalls) {
    const opsPerSecond = 500000000n; // 5 x 10^8 ops/sec
    
    if (totalCalls < 1000n) {
      return "< 0.001 毫秒";
    }

    const seconds = Number(totalCalls / opsPerSecond);
    
    if (seconds < 1) {
      const ms = (Number(totalCalls) / 500000).toFixed(2);
      return `約 ${ms} 毫秒`;
    }
    if (seconds < 60) {
      return `約 ${seconds.toFixed(1)} 秒`;
    }
    if (seconds < 3600) {
      return `約 ${(seconds / 60).toFixed(1)} 分鐘`;
    }
    if (seconds < 86400) {
      return `約 ${(seconds / 3600).toFixed(1)} 小時`;
    }
    const days = seconds / 86400;
    if (days < 365) {
      return `約 ${days.toFixed(1)} 天`;
    }
    const years = days / 365;
    if (years < 10000) {
      return `約 ${years.toFixed(1)} 年`;
    }
    if (years < 1e9) {
      return `約 ${(years / 1e4).toFixed(1)} 萬年`;
    }
    return `約 ${(years / 1e8).toFixed(1)} 億年 (超乎想像)`;
  }

  function runAnalysis(n) {
    const metrics = calculateMetrics(n);
    
    resultVal.textContent = metrics.value.toString();
    callsVal.textContent = metrics.totalCalls.toLocaleString();
    
    const timeEst = estimateCppTime(metrics.totalCalls);
    cppTimeVal.textContent = timeEst;

    if (n >= 35) {
      warningBox.style.display = 'block';
    } else {
      warningBox.style.display = 'none';
    }

    renderBarChart(metrics.subproblemCounts, metrics.totalCalls);
  }

  function renderBarChart(counts, totalCalls) {
    barChart.innerHTML = '';
    const keys = Object.keys(counts).map(Number).sort((a, b) => b - a);
    
    // 取最大呼叫次數作為 100% 基準
    let maxCount = 1n;
    keys.forEach(k => {
      if (counts[k] > maxCount) maxCount = counts[k];
    });

    keys.forEach(key => {
      const count = counts[key];
      // 算比例 (使用 BigInt 安全轉換 percentage)
      let percentage = Number((count * 100n) / maxCount);
      if (count > 0n && percentage < 2) percentage = 2; // 保留最小微小長度

      const item = document.createElement('div');
      item.className = 'bar-item';
      
      const isRedundant = count > 1n;
      const fillClass = isRedundant ? 'bar-fill high' : 'bar-fill';

      item.innerHTML = `
        <span class="bar-label">F(${key})</span>
        <div class="bar-track">
          <div class="${fillClass}" style="width: ${percentage}%"></div>
        </div>
        <span class="bar-count">${count.toLocaleString()} 次</span>
      `;
      barChart.appendChild(item);
    });
  }

  function buildTreeData(n, maxDepth) {
    let nodeId = 0;
    let leafCounter = 0;

    function createNode(val, depth) {
      const id = ++nodeId;
      const isDuplicate = (val < n - 1 && depth > 1);

      if (val <= 1 || depth >= maxDepth) {
        const xIndex = leafCounter++;
        return { id, val, depth, isDuplicate, xIndex, children: [] };
      }

      const left = createNode(val - 1, depth + 1);
      const right = createNode(val - 2, depth + 1);

      const xIndex = (left.xIndex + right.xIndex) / 2;
      return { id, val, depth, isDuplicate, xIndex, children: [left, right] };
    }

    return createNode(n, 1);
  }

  function renderTree(n, animated = false) {
    if (animationTimer) clearInterval(animationTimer);
    treeSvg.innerHTML = '';
    treeNodes.innerHTML = '';

    const displayDepthLimit = n > 5 ? 5 : 6;
    const treeData = buildTreeData(n, displayDepthLimit);

    const levelHeight = 55;
    const nodeSpacing = 45;

    let maxLeaves = 0;
    function countLeaves(node) {
      if (node.children.length === 0) {
        maxLeaves++;
        return;
      }
      node.children.forEach(countLeaves);
    }
    countLeaves(treeData);

    const canvasWidth = Math.max(treeViewport.clientWidth, maxLeaves * nodeSpacing + 60);
    const canvasHeight = Math.max(treeViewport.clientHeight, displayDepthLimit * levelHeight + 60);

    treeWrapper.style.width = `${canvasWidth}px`;
    treeWrapper.style.height = `${canvasHeight}px`;
    treeSvg.setAttribute('width', canvasWidth);
    treeSvg.setAttribute('height', canvasHeight);

    const nodesToRender = [];
    const edgesToRender = [];

    function processLayout(node) {
      const x = 30 + node.xIndex * nodeSpacing;
      const y = 35 + (node.depth - 1) * levelHeight;
      node.px = x;
      node.py = y;

      nodesToRender.push(node);

      node.children.forEach(child => {
        processLayout(child);
        edgesToRender.push({
          x1: x, y1: y,
          x2: child.px, y2: child.py,
          isDuplicate: child.isDuplicate
        });
      });
    }

    processLayout(treeData);

    if (!animated) {
      edgesToRender.forEach(e => drawEdge(e));
      nodesToRender.forEach(n => drawNode(n));
      centerTree(treeData.px);
    } else {
      let i = 0;
      nodesToRender.sort((a,b) => a.depth - b.depth || a.xIndex - b.xIndex);

      animationTimer = setInterval(() => {
        if (i < nodesToRender.length) {
          const currNode = nodesToRender[i];
          drawNode(currNode);
          
          const connectedEdges = edgesToRender.filter(e => e.x2 === currNode.px && e.y2 === currNode.py);
          connectedEdges.forEach(drawEdge);

          i++;
        } else {
          clearInterval(animationTimer);
        }
      }, 120);

      centerTree(treeData.px);
    }
  }

  function drawNode(node) {
    const el = document.createElement('div');
    el.className = 'node';
    if (node.isDuplicate) el.classList.add('duplicate');
    if (node.val <= 1) el.classList.add('leaf');
    
    el.textContent = `F(${node.val})`;
    el.style.left = `${node.px}px`;
    el.style.top = `${node.py}px`;

    treeNodes.appendChild(el);
  }

  function drawEdge(edge) {
    const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    line.setAttribute('x1', edge.x1);
    line.setAttribute('y1', edge.y1);
    line.setAttribute('x2', edge.x2);
    line.setAttribute('y2', edge.y2);
    line.setAttribute('class', edge.isDuplicate ? 'tree-line duplicate' : 'tree-line');
    
    treeSvg.appendChild(line);
  }

  function centerTree(rootX) {
    const viewportWidth = treeViewport.clientWidth;
    treeViewport.scrollLeft = rootX - viewportWidth / 2;
  }

  // 初始化預設帶入 N = 20 展現顯著效果
  runAnalysis(20);
  renderTree(20, false);
});
```
