```game-js
document.addEventListener('DOMContentLoaded', () => {
  const inputData = document.getElementById('input-data');
  const btnExample = document.getElementById('btn-example');
  const btnRandom = document.getElementById('btn-random');
  const outputContainer = document.getElementById('output-container');

  const exampleData = `4
IF
ELSE
END_IF
ENDPROGRAM
IF
S
ELSE
END_IF
IF
ELSE
S
END_IF
ENDPROGRAM
S
S
S
ENDPROGRAM
S
IF
S
S
ELSE
IF
IF
S
ELSE
S
END_IF
S
ELSE
S
END_IF
END_IF
S
ENDPROGRAM`;

  btnExample.addEventListener('click', () => {
    inputData.value = exampleData;
    parseAndRender();
  });

  btnRandom.addEventListener('click', () => {
    const progCount = Math.floor(Math.random() * 3) + 2;
    let result = `${progCount}\n`;

    function generateStatements(depth = 0) {
      let lines = [];
      const count = Math.floor(Math.random() * 2) + 1;
      for (let i = 0; i < count; i++) {
        if (depth < 2 && Math.random() < 0.5) {
          lines.push('IF');
          lines = lines.concat(generateStatements(depth + 1));
          lines.push('ELSE');
          lines = lines.concat(generateStatements(depth + 1));
          lines.push('END_IF');
        } else {
          lines.push('S');
        }
      }
      return lines;
    }

    for (let p = 0; p < progCount; p++) {
      const stmts = generateStatements();
      result += stmts.join('\n') + '\nENDPROGRAM\n';
    }

    inputData.value = result.trim();
    parseAndRender();
  });

  inputData.addEventListener('input', parseAndRender);

  function parseAndRender() {
    const text = inputData.value.trim();
    if (!text) {
      outputContainer.innerHTML = '<div class="placeholder-text">請在左側輸入測資或點擊按鈕載入範例</div>';
      return;
    }

    const rawTokens = text.split(/\s+/).filter(t => t.length > 0);
    if (rawTokens.length === 0) return;

    let tokenIdx = 0;
    let numPrograms = parseInt(rawTokens[0], 10);

    if (isNaN(numPrograms)) {
      numPrograms = null;
    } else {
      tokenIdx = 1;
    }

    outputContainer.innerHTML = '';
    let progIndex = 1;

    while (tokenIdx < rawTokens.length) {
      const progTokens = [];
      while (tokenIdx < rawTokens.length && rawTokens[tokenIdx] !== 'ENDPROGRAM') {
        progTokens.push(rawTokens[tokenIdx]);
        tokenIdx++;
      }

      const hasEndProgram = (tokenIdx < rawTokens.length && rawTokens[tokenIdx] === 'ENDPROGRAM');
      if (hasEndProgram) tokenIdx++;

      if (progTokens.length > 0 || hasEndProgram) {
        renderProgram(progIndex++, progTokens, hasEndProgram);
      }
    }
  }

  function renderProgram(index, tokens, hasEndProgram) {
    const progDiv = document.createElement('div');
    progDiv.className = 'program-block';

    const parseResult = parseBlock(tokens, 0);

    const header = document.createElement('div');
    header.className = 'program-header';
    
    const title = document.createElement('span');
    title.textContent = `程式 #${index}`;

    const badge = document.createElement('span');
    badge.className = 'path-badge';
    
    if (parseResult.error || parseResult.nextIdx < tokens.length || !hasEndProgram) {
      badge.textContent = '語法錯誤';
      badge.style.backgroundColor = '#ef4444';
    } else {
      badge.textContent = `總執行路徑數：${parseResult.paths}`;
    }

    header.appendChild(title);
    header.appendChild(badge);
    progDiv.appendChild(header);

    const treeRoot = document.createElement('div');
    treeRoot.className = 'tree-root';

    if (parseResult.error) {
      const errDiv = document.createElement('div');
      errDiv.className = 'error-msg';
      errDiv.textContent = `解析失敗：${parseResult.error}`;
      treeRoot.appendChild(errDiv);
    } else {
      buildDOMTree(parseResult.ast, treeRoot);
    }

    if (hasEndProgram) {
      const endLine = document.createElement('div');
      endLine.className = 'line kw-endprog';
      endLine.textContent = 'ENDPROGRAM';
      treeRoot.appendChild(endLine);
    }

    progDiv.appendChild(treeRoot);
    outputContainer.appendChild(progDiv);
  }

  // 遞迴解析語法樹並計算路徑數
  function parseBlock(tokens, idx) {
    const nodes = [];
    let totalPaths = 1n; // 採用 BigInt 計算

    while (idx < tokens.length) {
      const tok = tokens[idx];
      if (tok === 'ELSE' || tok === 'END_IF') {
        break;
      }

      if (tok === 'S') {
        nodes.push({ type: 'S', paths: 1n });
        idx++;
      } else if (tok === 'IF') {
        idx++; // 跳過 IF
        const ifBranch = parseBlock(tokens, idx);
        if (ifBranch.error) return { error: ifBranch.error };
        idx = ifBranch.nextIdx;

        if (idx >= tokens.length || tokens[idx] !== 'ELSE') {
          return { error: 'IF 缺少對應的 ELSE' };
        }
        idx++; // 跳過 ELSE

        const elseBranch = parseBlock(tokens, idx);
        if (elseBranch.error) return { error: elseBranch.error };
        idx = elseBranch.nextIdx;

        if (idx >= tokens.length || tokens[idx] !== 'END_IF') {
          return { error: '結構缺少 END_IF' };
        }
        idx++; // 跳過 END_IF

        const ifBranchPaths = ifBranch.paths;
        const elseBranchPaths = elseBranch.paths;
        const totalIfPaths = ifBranchPaths + elseBranchPaths;

        nodes.push({
          type: 'IF_STRUCTURE',
          paths: totalIfPaths,
          ifBranchPaths: ifBranchPaths,
          elseBranchPaths: elseBranchPaths,
          ifBranch: ifBranch.nodes,
          elseBranch: elseBranch.nodes
        });

        totalPaths *= totalIfPaths;
      } else {
        return { error: `未知關鍵字: ${tok}` };
      }
    }

    return {
      nodes: nodes,
      paths: totalPaths,
      nextIdx: idx,
      ast: nodes
    };
  }

  // 渲染 DOM，包含顯著相乘標示與詳細分支路徑數
  function buildDOMTree(nodes, container) {
    let previousIfNode = null;

    nodes.forEach((node) => {
      if (node.type === 'S') {
        const line = document.createElement('div');
        line.className = 'line kw-s';
        line.textContent = 'S';
        container.appendChild(line);
      } else if (node.type === 'IF_STRUCTURE') {
        // 若前面已有 IF 結構，顯示路徑相乘說明卡
        if (previousIfNode) {
          const multDiv = document.createElement('div');
          multDiv.className = 'multiply-indicator';
          multDiv.innerHTML = `<span>✖</span> <span>獨立選擇結構：前段路徑與後段路徑相乘 (× ${node.paths})</span>`;
          container.appendChild(multDiv);
        }
        previousIfNode = node;

        const ifGroup = document.createElement('div');
        ifGroup.className = 'if-group';

        // 1. IF Header（標頭與收合時詳細說明）
        const ifHeader = document.createElement('div');
        ifHeader.className = 'collapsible-header';
        
        const toggleIcon = document.createElement('span');
        toggleIcon.className = 'toggle-icon';
        toggleIcon.textContent = '▼';

        const ifText = document.createElement('span');
        ifText.className = 'kw-if';
        ifText.textContent = 'IF';

        // 收合時顯示 IF 與 ELSE 各別分支數與總計
        const summary = document.createElement('span');
        summary.className = 'collapsed-summary';
        summary.textContent = `[收合] IF分支: ${node.ifBranchPaths} 種 | ELSE分支: ${node.elseBranchPaths} 種 (合計: ${node.paths} 種)`;

        ifHeader.appendChild(toggleIcon);
        ifHeader.appendChild(ifText);
        ifHeader.appendChild(summary);

        // 2. 主體內容
        const collapsibleBody = document.createElement('div');
        collapsibleBody.className = 'collapsible-body';

        // IF 分支色階區塊
        const ifBranchBlock = document.createElement('div');
        ifBranchBlock.className = 'branch-block branch-if';
        buildDOMTree(node.ifBranch, ifBranchBlock);

        // ELSE 標籤
        const elseLine = document.createElement('div');
        elseLine.className = 'line kw-else';
        elseLine.textContent = 'ELSE';

        // ELSE 分支色階區塊
        const elseBranchBlock = document.createElement('div');
        elseBranchBlock.className = 'branch-block branch-else';
        buildDOMTree(node.elseBranch, elseBranchBlock);

        // END_IF 標籤 (顯著 Highlight)
        const endIfLine = document.createElement('div');
        endIfLine.className = 'line kw-endif';
        endIfLine.textContent = 'END_IF';

        // 組裝主體
        collapsibleBody.appendChild(ifBranchBlock);
        collapsibleBody.appendChild(elseLine);
        collapsibleBody.appendChild(elseBranchBlock);
        collapsibleBody.appendChild(endIfLine);

        ifGroup.appendChild(ifHeader);
        ifGroup.appendChild(collapsibleBody);

        // 綁定收合事件
        ifHeader.addEventListener('click', (e) => {
          e.stopPropagation();
          ifGroup.classList.toggle('collapsed');
        });

        container.appendChild(ifGroup);
      }
    });
  }

  // 預設載入範例
  btnExample.click();
});
```
