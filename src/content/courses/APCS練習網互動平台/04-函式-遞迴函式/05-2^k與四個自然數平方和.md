```game-js
document.addEventListener("DOMContentLoaded", () => {
  const kInput = document.getElementById("k-input");
  const startBtn = document.getElementById("start-btn");
  const prevBtn = document.getElementById("prev-btn");
  const nextBtn = document.getElementById("next-btn");
  const pauseBtn = document.getElementById("pause-btn");
  const speedSlider = document.getElementById("speed-slider");
  const speedVal = document.getElementById("speed-val");

  const canvas = document.getElementById("pie-canvas");
  const ctx = canvas.getContext("2d");

  const stepStatus = document.getElementById("step-status");
  const valTotal = document.getElementById("val-total");
  const valDepth = document.getElementById("val-depth");
  const valPath = document.getElementById("val-path");
  const valRem = document.getElementById("val-rem");
  const valMsg = document.getElementById("val-msg");
  const resultBox = document.getElementById("result-box");

  let steps = [];
  let currentStepIndex = -1;
  let timer = null;
  let isPlaying = false;

  const sliceColors = ["#3b82f6", "#10b981", "#f59e0b", "#8b5cf6"];
  const unassignedColor = "#e2e8f0";

  const speedMap = { "1": 1200, "2": 800, "3": 500, "4": 250, "5": 100 };
  const speedTextMap = { "1": "極慢", "2": "慢速", "3": "中速", "4": "快速", "5": "極快" };

  speedSlider.addEventListener("input", (e) => {
    speedVal.textContent = speedTextMap[e.target.value];
    if (isPlaying) restartTimer();
  });

  startBtn.addEventListener("click", () => {
    let k = parseInt(kInput.value, 10);
    if (isNaN(k) || k < 1) k = 1;
    if (k > 10) k = 10;
    kInput.value = k;
    
    generateSteps(k);
    currentStepIndex = 0;
    renderStep(currentStepIndex);
    play();
  });

  pauseBtn.addEventListener("click", () => {
    isPlaying ? pause() : play();
  });

  prevBtn.addEventListener("click", () => {
    pause();
    if (currentStepIndex > 0) {
      currentStepIndex--;
      renderStep(currentStepIndex);
    }
  });

  nextBtn.addEventListener("click", () => {
    pause();
    if (currentStepIndex < steps.length - 1) {
      currentStepIndex++;
      renderStep(currentStepIndex);
    }
  });

  function play() {
    isPlaying = true;
    pauseBtn.textContent = "暫停";
    pauseBtn.disabled = false;
    restartTimer();
  }

  function pause() {
    isPlaying = false;
    pauseBtn.textContent = "播放";
    if (timer) clearInterval(timer);
  }

  function restartTimer() {
    if (timer) clearInterval(timer);
    timer = setInterval(() => {
      if (currentStepIndex < steps.length - 1) {
        currentStepIndex++;
        renderStep(currentStepIndex);
      } else {
        pause();
      }
    }, speedMap[speedSlider.value]);
  }

  function generateSteps(k) {
    steps = [];
    const totalTarget = Math.pow(2, k);
    const results = [];

    // 當 k 為奇數時，無解
    if (k % 2 !== 0) {
      steps.push({
        totalTarget,
        depth: 0,
        target: totalTarget,
        path: [],
        results: [],
        message: "k 為奇數，數學證明無解！",
        status: "FINISHED"
      });
      return;
    }

    function recordStep(depth, target, path, message, status = "SEARCHING") {
      steps.push({
        totalTarget,
        depth,
        target,
        path: [...path],
        results: [...results],
        message,
        status
      });
    }

    recordStep(0, totalTarget, [], "準備開始遞迴搜尋...");

    function solve(depth, target, startNum, path) {
      if (depth === 3) {
        recordStep(3, target, path, `達到深度 3，檢查剩餘 ${target} 是否為完全平方數...`);
        const root = Math.sqrt(target);
        const lastNum = Math.round(root);

        if (lastNum * lastNum === target && lastNum >= startNum) {
          const fullPath = [...path, lastNum];
          results.push(fullPath.join(" "));
          recordStep(4, 0, fullPath, `找到合法組合：${fullPath.join(" + ")}`, "SUCCESS");
        } else {
          recordStep(3, target, path, `剩餘數值 ${target} 不符合條件，回溯...`, "FAIL");
        }
        return;
      }

      const remainingSlots = 4 - depth;
      for (let i = startNum; i * i * remainingSlots <= target; i++) {
        const nextTarget = target - i * i;
        const nextPath = [...path, i];
        recordStep(depth + 1, nextTarget, nextPath, `選擇平方數 ${i}² (${i*i})，剩餘 ${nextTarget}`);
        solve(depth + 1, nextTarget, i, nextPath);
        recordStep(depth, target, path, `從 ${i}² 回溯，嘗試下一個可能`);
      }
    }

    solve(0, totalTarget, 1, []);

    const finalPath = results.length > 0 ? results[results.length - 1].split(" ").map(Number) : [];
    steps.push({
      totalTarget,
      depth: results.length > 0 ? 4 : 0,
      target: 0,
      path: finalPath,
      results: [...results],
      message: results.length > 0 ? `搜尋完成！展示最終解答：${finalPath.join(" ")}` : "搜尋完成，無合法解。",
      status: "FINISHED"
    });
  }

  function renderStep(index) {
    if (index < 0 || index >= steps.length) return;
    const step = steps[index];

    prevBtn.disabled = index === 0;
    nextBtn.disabled = index === steps.length - 1;
    stepStatus.textContent = `步驟 ${index + 1} / ${steps.length}`;

    valTotal.textContent = step.totalTarget;
    valDepth.textContent = `${step.depth} / 3`;
    valPath.textContent = step.path.length > 0 ? `[${step.path.join(", ")}]` : "[]";
    valRem.textContent = step.target;
    valMsg.textContent = step.message;

    drawPieChart(step.totalTarget, step.path, step.target);

    resultBox.innerHTML = "";
    if (step.results.length === 0) {
      if (step.status === "FINISHED") {
        resultBox.innerHTML = '<div class="result-item" style="color:#ef4444">0 (無合法組合)</div>';
      } else {
        resultBox.innerHTML = '<span class="placeholder">搜尋中...</span>';
      }
    } else {
      step.results.forEach(res => {
        const div = document.createElement("div");
        div.className = "result-item";
        div.textContent = res;
        resultBox.appendChild(div);
      });
    }
  }

  function drawPieChart(total, path, remaining) {
    const cx = canvas.width / 2;
    const cy = canvas.height / 2;
    const radius = 150;

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    let startAngle = -Math.PI / 2;

    path.forEach((num, idx) => {
      const val = num * num;
      const sliceAngle = (val / total) * (2 * Math.PI);
      const endAngle = startAngle + sliceAngle;

      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.arc(cx, cy, radius, startAngle, endAngle);
      ctx.closePath();

      ctx.fillStyle = sliceColors[idx % sliceColors.length];
      ctx.fill();
      ctx.strokeStyle = "#ffffff";
      ctx.lineWidth = 2;
      ctx.stroke();

      const midAngle = startAngle + sliceAngle / 2;
      const labelX = cx + Math.cos(midAngle) * (radius * 0.65);
      const labelY = cy + Math.sin(midAngle) * (radius * 0.65);

      ctx.fillStyle = "#ffffff";
      ctx.font = "bold 14px monospace";
      ctx.textAlign = "center";
      ctx.textBaseline = "middle";
      ctx.fillText(`${num}² (${val})`, labelX, labelY);

      startAngle = endAngle;
    });

    if (remaining > 0) {
      const sliceAngle = (remaining / total) * (2 * Math.PI);
      const endAngle = startAngle + sliceAngle;

      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.arc(cx, cy, radius, startAngle, endAngle);
      ctx.closePath();

      ctx.fillStyle = unassignedColor;
      ctx.fill();
      ctx.strokeStyle = "#ffffff";
      ctx.lineWidth = 2;
      ctx.stroke();

      if (sliceAngle > 0.2) {
        const midAngle = startAngle + sliceAngle / 2;
        const labelX = cx + Math.cos(midAngle) * (radius * 0.5);
        const labelY = cy + Math.sin(midAngle) * (radius * 0.5);

        ctx.fillStyle = "#64748b";
        ctx.font = "13px sans-serif";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillText(`剩餘 ${remaining}`, labelX, labelY);
      }
    }

    ctx.beginPath();
    ctx.arc(cx, cy, radius, 0, 2 * Math.PI);
    ctx.strokeStyle = "#cbd5e1";
    ctx.lineWidth = 3;
    ctx.stroke();
  }
});
```
