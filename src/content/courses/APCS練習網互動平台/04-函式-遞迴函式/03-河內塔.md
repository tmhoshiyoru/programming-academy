```game-js
const colors = ['#e53e3e', '#dd6b20', '#d69e2e', '#38a169', '#3182ce', '#805ad5', '#d69e2e', '#319795'];
const pegNames = ['A (起點)', 'B (中繼)', 'C (目標)'];
const MAX_SAFE_STEPS = 1000;

let diskCount = 3;
let pegs = [[], [], []];
let moveSequence = [];
let currentStep = 0;
let isAutoPlaying = false;
let autoPlayTimer = null;
let selectedPegIndex = null;

const diskCountInput = document.getElementById('disk-count');
const diskCountNum = document.getElementById('disk-count-num');
const btnAutoPlay = document.getElementById('btn-auto-play');
const btnNextStep = document.getElementById('btn-next-step');
const btnReset = document.getElementById('btn-reset');
const messageBanner = document.getElementById('message-banner');
const stepExplanation = document.getElementById('step-explanation');
const pegNodes = document.querySelectorAll('.peg-wrapper');
const modalOverlay = document.getElementById('modal-overlay');
const modalMsg = document.getElementById('modal-msg');
const modalCloseBtn = document.getElementById('modal-close-btn');

function init() {
  diskCount = parseInt(diskCountInput.value);
  diskCountNum.value = diskCount;
  stopAutoPlay();
  
  pegs = [[], [], []];
  for (let i = diskCount; i >= 1; i--) {
    pegs[0].push(i);
  }
  
  selectedPegIndex = null;
  currentStep = 0;
  moveSequence = [];
  
  generateMovesWithPhase(diskCount, 0, 1, 2, 1);
  
  renderDisks();
  highlightCard(0);
  messageBanner.textContent = `盤子數：${diskCount} 個。點擊單步移動或自動播放。`;
  stepExplanation.textContent = `準備就緒！要把 ${diskCount} 個盤子從 A 搬到 C，只需掌握「拆解 ➔ 移動 ➔ 歸位」原則。`;
}

function generateMovesWithPhase(n, from, aux, to, currentPhase) {
  if (moveSequence.length >= MAX_SAFE_STEPS) return;
  
  if (n === 1) {
    moveSequence.push({ 
      disk: 1, 
      from, 
      to, 
      aux, 
      phase: currentPhase,
      subProblemSize: n
    });
    return;
  }
  
  generateMovesWithPhase(n - 1, from, to, aux, 1);
  
  if (moveSequence.length >= MAX_SAFE_STEPS) return;
  moveSequence.push({ 
    disk: n, 
    from, 
    to, 
    aux, 
    phase: 2,
    subProblemSize: n
  });
  
  if (moveSequence.length >= MAX_SAFE_STEPS) return;
  generateMovesWithPhase(n - 1, aux, from, to, 3);
}

function renderDisks() {
  document.querySelectorAll('.disk').forEach(el => el.remove());
  const container = document.getElementById('pegs-container');
  const containerWidth = container.clientWidth;
  const containerHeight = container.clientHeight;
  const pegAreaWidth = containerWidth * 0.32;

  const minDiskWidthPercent = 20;
  const maxDiskWidthPercent = 90;

  const availableHeight = containerHeight - 50; 
  const rawDiskHeight = Math.floor(availableHeight / diskCount);
  const diskHeight = Math.max(2, Math.min(16, rawDiskHeight));
  const fontSize = diskHeight >= 11 ? '0.7rem' : '0px';

  pegs.forEach((peg, pegIndex) => {
    peg.forEach((diskSize, diskIndex) => {
      const diskEl = document.createElement('div');
      diskEl.className = 'disk';
      if (selectedPegIndex === pegIndex && diskIndex === peg.length - 1) {
        diskEl.classList.add('selected');
      }
      
      if (fontSize !== '0px') {
        diskEl.textContent = diskSize;
        diskEl.style.fontSize = fontSize;
      }
      
      let widthRatio;
      if (diskCount === 1) {
        widthRatio = maxDiskWidthPercent / 100;
      } else {
        const stepRatio = (diskSize - 1) / (diskCount - 1);
        widthRatio = (minDiskWidthPercent + stepRatio * (maxDiskWidthPercent - minDiskWidthPercent)) / 100;
      }
      
      const diskPixelWidth = pegAreaWidth * widthRatio;
      diskEl.style.width = `${diskPixelWidth}px`;
      diskEl.style.height = `${diskHeight}px`;
      diskEl.style.borderRadius = `${Math.min(4, diskHeight / 2)}px`;
      diskEl.style.backgroundColor = colors[(diskSize - 1) % colors.length];
      
      const pegCenter = (pegIndex + 0.5) * (containerWidth / 3);
      const diskLeft = pegCenter - (diskPixelWidth / 2);
      diskEl.style.left = `${diskLeft}px`;
      
      const bottomOffset = 18 + diskIndex * (diskHeight + (diskCount > 30 ? 0 : 1));
      diskEl.style.bottom = `${bottomOffset}px`;

      container.appendChild(diskEl);
    });
  });
}

function executeStep() {
  if (currentStep >= MAX_SAFE_STEPS) {
    stopAutoPlay();
    showStepLimitModal();
    return;
  }

  if (currentStep >= moveSequence.length) {
    stopAutoPlay();
    if (diskCount <= 10) {
      messageBanner.textContent = "🎉 太棒了！已成功將所有盤子移動完畢！";
      stepExplanation.textContent = "任務完成！無論多少層，都可以藉由拆解為重複的邏輯來解決。";
      highlightCard(0);
    } else {
      showStepLimitModal();
    }
    return;
  }

  const step = moveSequence[currentStep];
  
  const disk = pegs[step.from].pop();
  pegs[step.to].push(disk);
  renderDisks();
  
  messageBanner.textContent = `步驟 ${currentStep + 1} / ${moveSequence.length}：移動盤子 ${step.disk} (${pegNames[step.from]} ➔ ${pegNames[step.to]})`;
  
  highlightCard(step.phase);

  if (step.phase === 1) {
    stepExplanation.textContent = `【階段 1：移開上方群組】正在將上方小盤子從 ${pegNames[step.from]} 暫存至 ${pegNames[step.to]}（為更下方的大盤子讓路）。`;
  } else if (step.phase === 2) {
    stepExplanation.textContent = `【階段 2：核心搬運】上方障礙已排除！將當前關鍵的 ${step.disk} 號大盤子從 ${pegNames[step.from]} 移至目標柱 ${pegNames[step.to]}。`;
  } else if (step.phase === 3) {
    stepExplanation.textContent = `【階段 3：歸位上方群組】最大盤子已到位！將之前暫存在中繼柱的盤子從 ${pegNames[step.from]} 搬回目標柱 ${pegNames[step.to]} 進行合體。`;
  }

  currentStep++;
}

function highlightCard(cardIndex) {
  for (let i = 1; i <= 3; i++) {
    const card = document.getElementById(`card-step${i}`);
    if (card) card.classList.remove('active');
  }
  if (cardIndex > 0) {
    const activeCard = document.getElementById(`card-step${cardIndex}`);
    if (activeCard) activeCard.classList.add('active');
  }
}

function showStepLimitModal() {
  const totalSteps = (BigInt(2) ** BigInt(diskCount)) - BigInt(1);
  modalMsg.innerHTML = `當前設定圓盤數量為 <b>${diskCount}</b> 個。<br><br>` +
    `完全解開河內塔理論上需要 <b>2<sup>${diskCount}</sup> - 1 = ${totalSteps.toLocaleString()}</b> 次移動！<br>` +
    `若以每秒移動 1 次計算，64 個圓盤約需 <b>5,849 億年</b>（遠超宇宙年齡）！<br><br>` +
    `為了防止瀏覽器卡死，系統已在第 <b>1,000 步</b> 自動終止模擬。這正是演算法中「指數爆炸 <i>O(2<sup>n</sup>)</i>」的最好證明！`;
  modalOverlay.style.display = 'flex';
}

function stopAutoPlay() {
  isAutoPlaying = false;
  btnAutoPlay.textContent = "自動播放";
  btnAutoPlay.classList.remove('secondary');
  btnAutoPlay.classList.add('primary');
  if (autoPlayTimer) {
    clearInterval(autoPlayTimer);
    autoPlayTimer = null;
  }
}

function startAutoPlay() {
  if (currentStep >= moveSequence.length) currentStep = 0;
  isAutoPlaying = true;
  btnAutoPlay.textContent = "暫停播放";
  btnAutoPlay.classList.remove('primary');
  btnAutoPlay.classList.add('secondary');
  
  executeStep();
  const interval = diskCount > 20 ? 80 : 500;
  autoPlayTimer = setInterval(() => {
    if (currentStep < moveSequence.length && currentStep < MAX_SAFE_STEPS) {
      executeStep();
    } else {
      executeStep();
      stopAutoPlay();
    }
  }, interval);
}

pegNodes.forEach(pegNode => {
  pegNode.addEventListener('click', () => {
    if (isAutoPlaying) stopAutoPlay();
    const pegIndex = parseInt(pegNode.getAttribute('data-peg'));
    
    if (selectedPegIndex === null) {
      if (pegs[pegIndex].length > 0) {
        selectedPegIndex = pegIndex;
        messageBanner.textContent = `已選取 ${pegNames[pegIndex]} 的頂端盤子，請點擊目標柱子。`;
      }
    } else {
      if (selectedPegIndex === pegIndex) {
        selectedPegIndex = null;
        messageBanner.textContent = "取消選取。";
      } else {
        const fromPeg = pegs[selectedPegIndex];
        const toPeg = pegs[pegIndex];
        const movingDisk = fromPeg[fromPeg.length - 1];
        const targetTopDisk = toPeg[toPeg.length - 1];
        
        if (!targetTopDisk || movingDisk < targetTopDisk) {
          toPeg.push(fromPeg.pop());
          messageBanner.textContent = `手動移動：${pegNames[selectedPegIndex]} ➔ ${pegNames[pegIndex]}`;
          stepExplanation.textContent = "手動驗證中：請確認遵守「大盤子不可放在小盤子上」的原則。";
          highlightCard(0);
        } else {
          messageBanner.textContent = "⚠️ 違規移動：大盤子不能疊在小盤子上！";
        }
        selectedPegIndex = null;
      }
    }
    renderDisks();
  });
});

function handleInputChange(val) {
  let num = parseInt(val);
  if (isNaN(num)) num = 3;
  if (num < 3) num = 3;
  if (num > 64) num = 64;
  diskCountInput.value = num;
  diskCountNum.value = num;
  init();
}

diskCountInput.addEventListener('input', (e) => handleInputChange(e.target.value));
diskCountNum.addEventListener('change', (e) => handleInputChange(e.target.value));

btnAutoPlay.addEventListener('click', () => {
  if (isAutoPlaying) {
    stopAutoPlay();
  } else {
    startAutoPlay();
  }
});

btnNextStep.addEventListener('click', () => {
  if (isAutoPlaying) stopAutoPlay();
  executeStep();
});

btnReset.addEventListener('click', init);
modalCloseBtn.addEventListener('click', () => {
  modalOverlay.style.display = 'none';
});

window.addEventListener('resize', renderDisks);

init();
```
