```game-js
let currentMatrix = [];
let n = 4;

function generateMatrix(randomSize = false) {
  if (randomSize) {
    // 隨機產生 2 到 10 的 n
    n = Math.floor(Math.random() * 9) + 2;
    document.getElementById('matrixSize').value = n;
  } else {
    const inputN = parseInt(document.getElementById('matrixSize').value, 10);
    n = Math.max(2, Math.min(10, isNaN(inputN) ? 4 : inputN));
  }

  // 1. 建立完美的偶同位矩陣 (OK 狀態)
  let mat = Array.from({ length: n }, () => new Array(n).fill(0));

  // 隨機填充 (n-1) x (n-1) 子矩陣
  for (let i = 0; i < n - 1; i++) {
    for (let j = 0; j < n - 1; j++) {
      mat[i][j] = Math.random() < 0.5 ? 0 : 1;
    }
  }

  // 自動設置第 (n-1) 欄，使前 (n-1) 列的 Sum 均為偶數
  for (let i = 0; i < n - 1; i++) {
    let rowSum = 0;
    for (let j = 0; j < n - 1; j++) rowSum += mat[i][j];
    mat[i][n - 1] = rowSum % 2 === 0 ? 0 : 1;
  }

  // 自動設置第 (n-1) 列，使前 (n-1) 欄的 Sum 均為偶數
  for (let j = 0; j < n - 1; j++) {
    let colSum = 0;
    for (let i = 0; i < n - 1; i++) colSum += mat[i][j];
    mat[n - 1][j] = colSum % 2 === 0 ? 0 : 1;
  }

  // 計算最後右下角 (n-1, n-1) 位元，確保第 (n-1) 列與欄皆為偶數
  let lastRowSum = 0;
  for (let j = 0; j < n - 1; j++) lastRowSum += mat[n - 1][j];
  mat[n - 1][n - 1] = lastRowSum % 2 === 0 ? 0 : 1;

  // 2. 隨機等機率選擇結果類型 (0: OK, 1: Change bit, 2: Corrupt) 各 1/3
  const targetType = Math.floor(Math.random() * 3);

  if (targetType === 1) {
    // Change bit: 翻轉正好 1 個位元
    const r = Math.floor(Math.random() * n);
    const c = Math.floor(Math.random() * n);
    mat[r][c] = 1 - mat[r][c];
  } else if (targetType === 2) {
    // Corrupt: 翻轉 2 個不在同列同欄的位元，造成 2 列 2 欄奇數，無法單一修正
    const r1 = Math.floor(Math.random() * n);
    let r2 = Math.floor(Math.random() * n);
    while (r2 === r1) r2 = Math.floor(Math.random() * n);

    const c1 = Math.floor(Math.random() * n);
    let c2 = Math.floor(Math.random() * n);
    while (c2 === c1) c2 = Math.floor(Math.random() * n);

    mat[r1][c1] = 1 - mat[r1][c1];
    mat[r2][c2] = 1 - mat[r2][c2];
  }

  currentMatrix = mat;
  renderTable();
}

function renderTable() {
  const wrapper = document.getElementById('matrixWrapper');
  let rowSums = new Array(n).fill(0);
  let colSums = new Array(n).fill(0);

  // 計算行列總和
  for (let i = 0; i < n; i++) {
    for (let j = 0; j < n; j++) {
      rowSums[i] += currentMatrix[i][j];
      colSums[j] += currentMatrix[i][j];
    }
  }

  // 找出奇數總和的列與欄 (1-based index)
  let oddRows = [];
  let oddCols = [];
  for (let i = 0; i < n; i++) {
    if (rowSums[i] % 2 !== 0) oddRows.push(i + 1);
    if (colSums[i] % 2 !== 0) oddCols.push(i + 1);
  }

  let html = '<table>';

  for (let i = 0; i < n; i++) {
    html += '<tr>';
    for (let j = 0; j < n; j++) {
      // 若恰好有 1 列 1 欄奇數，高亮顯示交會點位元
      const isTargetBit = (oddRows.length === 1 && oddCols.length === 1 &&
                           oddRows[0] === i + 1 && oddCols[0] === j + 1);
      const highlightClass = isTargetBit ? 'target-bit' : '';

      html += `<td class="cell ${highlightClass}" onclick="toggleBit(${i}, ${j})">${currentMatrix[i][j]}</td>`;
    }
    const isRowOdd = rowSums[i] % 2 !== 0;
    html += `<td class="sum-cell ${isRowOdd ? 'sum-invalid' : ''}">sum=${rowSums[i]}</td>`;
    html += '</tr>';
  }

  // 渲染底部欄總和列
  html += '<tr>';
  for (let j = 0; j < n; j++) {
    const isColOdd = colSums[j] % 2 !== 0;
    html += `<td class="sum-cell ${isColOdd ? 'sum-invalid' : ''}">sum=${colSums[j]}</td>`;
  }
  html += '<td class="corner-cell"></td></tr>';
  html += '</table>';

  wrapper.innerHTML = html;
  evaluateParity(oddRows, oddCols);
}

function toggleBit(r, c) {
  currentMatrix[r][c] = 1 - currentMatrix[r][c];
  renderTable();
}

function evaluateParity(oddRows, oddCols) {
  const resultBox = document.getElementById('resultBox');
  const resultText = document.getElementById('resultText');

  resultBox.className = 'result-box';

  if (oddRows.length === 0 && oddCols.length === 0) {
    resultText.innerText = 'OK';
    resultBox.classList.add('res-ok');
  } else if (oddRows.length === 1 && oddCols.length === 1) {
    resultText.innerText = `Change bit (${oddRows[0]},${oddCols[0]})`;
    resultBox.classList.add('res-change');
  } else {
    resultText.innerText = 'Corrupt';
    resultBox.classList.add('res-corrupt');
  }
}

document.getElementById('btnGenerate').addEventListener('click', () => generateMatrix(false));
document.getElementById('btnRandomGenerate').addEventListener('click', () => generateMatrix(true));

// 初次載入
generateMatrix(false);
```
