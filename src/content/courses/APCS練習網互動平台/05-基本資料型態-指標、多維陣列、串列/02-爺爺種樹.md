```game-js
document.addEventListener('DOMContentLoaded', () => {
  const inputData = document.getElementById('inputData');
  const btnRun = document.getElementById('btnRun');
  const btnReset = document.getElementById('btnReset');
  const gridSizeEl = document.getElementById('gridSize');
  const linesCountEl = document.getElementById('linesCount');
  const treeCountEl = document.getElementById('treeCount');
  const gridContainer = document.getElementById('gridContainer');
  const linesList = document.getElementById('linesList');

  const defaultInput = `7 8
8
2 8 6 8
4 7 2 5
6 7 4 5
1 5 1 7
7 7 7 5
6 4 2 4
2 3 4 1
6 3 4 1`;

  let currentParsedLines = [];
  let selectedLineIndex = -1;

  function playSound() {
    try {
      const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(587.33, audioCtx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.15);
      gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.15);
    } catch (e) {}
  }

  function getLinePoints(x1, y1, x2, y2) {
    const dx = x2 - x1;
    const dy = y2 - y1;

    // 檢查是否符合四種合法角度：橫排、直排、對角線 (斜率必須為 0, ±1 或 垂直)
    if (x1 !== x2 && y1 !== y2 && Math.abs(dx) !== Math.abs(dy)) {
      return null;
    }

    const points = [];
    const stepX = Math.sign(dx);
    const stepY = Math.sign(dy);
    let curX = x1;
    let curY = y1;

    while (true) {
      points.push(`${curX},${curY}`);
      if (curX === x2 && curY === y2) break;
      curX += stepX;
      curY += stepY;
    }
    return points;
  }

  function processGame() {
    selectedLineIndex = -1;
    const rawText = inputData.value.trim();
    const lines = rawText.split('\n').map(l => l.trim()).filter(l => l.length > 0);

    if (lines.length < 2) {
      alert('輸入格式錯誤：請包含田地大小與樹木排數！');
      return;
    }

    const [m, n] = lines[0].split(/\s+/).map(Number);
    const t = parseInt(lines[1], 10);

    if (isNaN(m) || isNaN(n) || isNaN(t) || m <= 1 || n <= 1 || t < 1) {
      alert('輸入格式錯誤：田地尺寸與排數需為有效數值！');
      return;
    }

    const treeSet = new Set();
    currentParsedLines = [];

    for (let i = 2; i < 2 + t && i < lines.length; i++) {
      const coords = lines[i].split(/\s+/).map(Number);
      if (coords.length < 4 || coords.some(isNaN)) {
        alert(`第 ${i - 1} 排樹木資料格式錯誤！`);
        return;
      }

      const [x1, y1, x2, y2] = coords;

      // 檢查座標邊界
      if (x1 < 1 || x1 > m || x2 < 1 || x2 > m || y1 < 1 || y1 > n || y2 < 1 || y2 > n) {
        alert(`輸入測資有誤：第 ${i - 1} 排樹木座標 (${x1}, ${y1}) -> (${x2}, ${y2}) 超出田地範圍 (1~${m}, 1~${n})！`);
        return;
      }

      const pts = getLinePoints(x1, y1, x2, y2);
      if (!pts) {
        alert(`輸入測資有誤：第 ${i - 1} 排樹木座標 (${x1}, ${y1}) -> (${x2}, ${y2}) 不屬於直排、橫排或 45 度對角斜排！`);
        return;
      }

      pts.forEach(p => treeSet.add(p));
      currentParsedLines.push({ x1, y1, x2, y2, points: pts });
    }

    gridSizeEl.textContent = `${m} × ${n}`;
    linesCountEl.textContent = currentParsedLines.length;
    treeCountEl.textContent = treeSet.size;

    renderSidebar();
    renderGrid(m, n, treeSet);
    playSound();
  }

  function renderSidebar() {
    linesList.innerHTML = '';
    currentParsedLines.forEach((line, idx) => {
      const btn = document.createElement('button');
      btn.className = 'line-btn';
      btn.textContent = `#${idx + 1} (${line.x1},${line.y1})→(${line.x2},${line.y2})`;
      btn.addEventListener('click', () => {
        if (selectedLineIndex === idx) {
          selectedLineIndex = -1;
          btn.classList.remove('active');
        } else {
          selectedLineIndex = idx;
          document.querySelectorAll('.line-btn').forEach(b => b.classList.remove('active'));
          btn.classList.add('active');
        }
        updateGridHighlights();
      });
      linesList.appendChild(btn);
    });
  }

  function updateGridHighlights() {
    const activePoints = selectedLineIndex >= 0 ? new Set(currentParsedLines[selectedLineIndex].points) : new Set();
    
    document.querySelectorAll('.grid-cell[data-coord]').forEach(cell => {
      const coord = cell.getAttribute('data-coord');
      if (activePoints.has(coord)) {
        cell.classList.add('highlight-tree');
      } else {
        cell.classList.remove('highlight-tree');
      }
    });
  }

  function renderGrid(cols, rows, treeSet) {
    gridContainer.innerHTML = '';
    
    const displayCols = Math.min(cols, 20);
    const displayRows = Math.min(rows, 20);

    gridContainer.style.gridTemplateColumns = `repeat(${displayCols + 1}, 28px)`;

    // 左上角空白格
    const cornerCell = document.createElement('div');
    cornerCell.className = 'grid-cell header';
    gridContainer.appendChild(cornerCell);

    // 欄標題 (東西向)
    for (let c = 1; c <= displayCols; c++) {
      const headerCell = document.createElement('div');
      headerCell.className = 'grid-cell header';
      headerCell.textContent = c;
      gridContainer.appendChild(headerCell);
    }

    // 列與單元格 (南北向)
    for (let r = 1; r <= displayRows; r++) {
      const rowHeader = document.createElement('div');
      rowHeader.className = 'grid-cell header';
      rowHeader.textContent = r;
      gridContainer.appendChild(rowHeader);

      for (let c = 1; c <= displayCols; c++) {
        const cell = document.createElement('div');
        cell.className = 'grid-cell';
        const key = `${c},${r}`;
        cell.setAttribute('data-coord', key);

        if (treeSet.has(key)) {
          cell.classList.add('has-tree');
        }
        gridContainer.appendChild(cell);
      }
    }
  }

  btnRun.addEventListener('click', processGame);

  btnReset.addEventListener('click', () => {
    inputData.value = defaultInput;
    processGame();
  });

  processGame();
});
```
