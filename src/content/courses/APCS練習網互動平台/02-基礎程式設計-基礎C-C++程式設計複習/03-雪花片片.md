```game-js
const canvas = document.getElementById('snowflake-canvas');
const ctx = canvas.getContext('2d');

const nInput = document.getElementById('n-input');
const btnSet = document.getElementById('btn-set');
const btnNext = document.getElementById('btn-next');
const btnAuto = document.getElementById('btn-auto');
const btnReset = document.getElementById('btn-reset');

const tableBody = document.getElementById('table-body');
const finalResult = document.getElementById('final-result');
const currentRenderedLayerSpan = document.getElementById('current-rendered-layer');

let targetN = 3;
let currentN = 1;
let autoTimer = null;

// 紀錄算過的每一層數據 (BigInt 支援大數)
let dpData = [];

function initData() {
  dpData = [];
  let edge = 3n;          // 第 1 層有 3 個外邊
  let triInLayer = 1n;    // 第 1 層本身有 1 個三角形
  let sumTri = 1n;        // 累計三角形總數 S(1) = 1

  dpData.push({
    layer: 1,
    edges: edge,
    addedTri: 1n,
    triInLayer: triInLayer,
    sumTri: sumTri
  });

  for (let i = 2; i <= 120; i++) {
    let added = edge;                     // 這一層新增的三角形數等於上一層的邊數
    triInLayer = triInLayer + added;      // 該層等邊三角形數量
    sumTri = sumTri + triInLayer;         // 累計總數量
    edge = edge * 4n;                     // 邊數變為 4 倍

    dpData.push({
      layer: i,
      edges: edge,
      addedTri: added,
      triInLayer: triInLayer,
      sumTri: sumTri
    });
  }
}

function renderTable() {
  tableBody.innerHTML = '';
  for (let i = 0; i < currentN; i++) {
    const data = dpData[i];
    const tr = document.createElement('tr');
    if (i === currentN - 1) {
      tr.classList.add('active-row');
    }
    
    tr.innerHTML = `
      <td style="text-align:center;">${data.layer}</td>
      <td>${data.edges.toLocaleString()}</td>
      <td>${data.addedTri.toLocaleString()}</td>
      <td>${data.triInLayer.toLocaleString()}</td>
      <td style="color:#0284c7; font-weight:bold;">${data.sumTri.toLocaleString()}</td>
    `;
    tableBody.appendChild(tr);
  }
  
  // 自動捲動到底部
  const tableContainer = document.querySelector('.table-container');
  tableContainer.scrollTop = tableContainer.scrollHeight;

  // 更新最終答案顯示
  finalResult.innerText = dpData[currentN - 1].sumTri.toString();
}

function drawSnowflake(level) {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  // 限制繪製層數最多為 5，避免繪製過多細節造成視覺混亂與卡頓
  const renderLevel = Math.min(level, 5);
  currentRenderedLayerSpan.innerText = `${currentN} ${level > 5 ? '(畫布僅示範前 5 層構型)' : ''}`;

  const width = canvas.width;
  const height = canvas.height;
  
  // 計算正三角形頂點 (邊長約 260px)
  const side = 260;
  const h = side * Math.sin(Math.PI / 3);
  
  const p1 = { x: (width - side) / 2, y: height / 2 + h / 3 };
  const p2 = { x: (width + side) / 2, y: height / 2 + h / 3 };
  const p3 = { x: width / 2, y: height / 2 - (2 * h) / 3 };

  ctx.strokeStyle = '#38bdf8';
  ctx.fillStyle = 'rgba(56, 189, 248, 0.15)';
  ctx.lineWidth = 1.5;

  ctx.beginPath();
  drawKochSegment(p1, p3, renderLevel);
  drawKochSegment(p3, p2, renderLevel);
  drawKochSegment(p2, p1, renderLevel);
  ctx.closePath();
  ctx.stroke();
  ctx.fill();
}

function drawKochSegment(p1, p2, depth) {
  if (depth === 1) {
    ctx.lineTo(p2.x, p2.y);
    return;
  }

  const dx = (p2.x - p1.x) / 3;
  const dy = (p2.y - p1.y) / 3;

  const a = { x: p1.x + dx, y: p1.y + dy };
  const c = { x: p1.x + 2 * dx, y: p1.y + 2 * dy };

  // 計算突出的三角形頂點 b
  const angle = Math.atan2(p2.y - p1.y, p2.x - p1.x) - Math.PI / 3;
  const dist = Math.hypot(dx, dy);
  const b = {
    x: a.x + Math.cos(angle) * dist,
    y: a.y + Math.sin(angle) * dist
  };

  drawKochSegment(p1, a, depth - 1);
  drawKochSegment(a, b, depth - 1);
  drawKochSegment(b, c, depth - 1);
  drawKochSegment(c, p2, depth - 1);
}

function updateState() {
  renderTable();
  drawSnowflake(currentN);

  btnNext.disabled = currentN >= targetN;
}

function stopAutoPlay() {
  if (autoTimer) {
    clearInterval(autoTimer);
    autoTimer = null;
    btnAuto.innerText = '自動播放';
    btnAuto.classList.replace('danger', 'success');
  }
}

// 事件監聽
btnSet.addEventListener('click', () => {
  stopAutoPlay();
  let val = parseInt(nInput.value, 10);
  if (isNaN(val) || val < 1) val = 1;
  if (val > 120) val = 120;
  
  targetN = val;
  nInput.value = targetN;
  currentN = 1;
  updateState();
});

btnNext.addEventListener('click', () => {
  if (currentN < targetN) {
    currentN++;
    updateState();
  }
});

btnReset.addEventListener('click', () => {
  stopAutoPlay();
  currentN = 1;
  updateState();
});

btnAuto.addEventListener('click', () => {
  if (autoTimer) {
    stopAutoPlay();
  } else {
    if (currentN >= targetN) {
      currentN = 1;
      updateState();
    }
    btnAuto.innerText = '暫停';
    btnAuto.classList.replace('success', 'danger');
    autoTimer = setInterval(() => {
      if (currentN < targetN) {
        currentN++;
        updateState();
      } else {
        stopAutoPlay();
      }
    }, 600);
  }
});

// 初始化
initData();
updateState();
```
