```game-js
let currentStep = 1;
let calculatedDiffSec = 35831; // 預設 09:57:11 的秒數

function switchTab(step) {
  currentStep = step;
  document.querySelectorAll('.tab-btn').forEach((btn, idx) => {
    btn.classList.toggle('active', idx + 1 === step);
  });
  document.querySelectorAll('.step-panel').forEach((panel, idx) => {
    panel.classList.toggle('active', idx + 1 === step);
  });
}

function runCinAnimation() {
  const tokenC1 = document.getElementById('token-c1');
  const tokenC2 = document.getElementById('token-c2');
  
  // 動畫效果：吃掉冒號
  tokenC1.style.transform = 'scale(1.3) translateY(-10px)';
  tokenC1.style.backgroundColor = '#ffcdd2';
  
  setTimeout(() => {
    tokenC1.style.transform = 'scale(1)';
    document.getElementById('val-h1').innerText = '11';
    document.getElementById('val-c1').innerText = "':'";
    document.getElementById('val-m1').innerText = '13';
  }, 400);

  setTimeout(() => {
    tokenC2.style.transform = 'scale(1.3) translateY(-10px)';
    tokenC2.style.backgroundColor = '#ffcdd2';
  }, 800);

  setTimeout(() => {
    tokenC2.style.transform = 'scale(1)';
    document.getElementById('val-c2').innerText = "':'";
    document.getElementById('val-s1').innerText = '20';
  }, 1200);
}

function toggleCrossDayMode() {
  const isCross = document.getElementById('crossDayCheck').checked;
  const explain = document.getElementById('crossDayExplain');
  const timeBlock1 = document.querySelectorAll('.time-block')[0];
  const timeBlock2 = document.querySelectorAll('.time-block')[1];

  if (isCross) {
    explain.classList.remove('hidden');
    timeBlock1.innerHTML = '<h4>時間 1 (02:13:18)</h4><p>2 × 3600 + 13 × 60 + 18</p><div class="sec-result">= <span class="highlight">7998</span> 秒</div>';
    timeBlock2.innerHTML = '<h4>時間 2 (01:20:32)</h4><p>1 × 3600 + 20 × 60 + 32</p><div class="sec-result">= <span class="highlight">4832</span> 秒</div>';
  } else {
    explain.classList.add('hidden');
    timeBlock1.innerHTML = '<h4>時間 1 (11:13:20)</h4><p>11 × 3600 + 13 × 60 + 20</p><div class="sec-result">= <span class="highlight">40400</span> 秒</div>';
    timeBlock2.innerHTML = '<h4>時間 2 (21:10:31)</h4><p>21 × 3600 + 10 × 60 + 31</p><div class="sec-result">= <span class="highlight">76231</span> 秒</div>';
  }
}

function calculateDiff() {
  const isCross = document.getElementById('crossDayCheck').checked;
  const resultDiv = document.getElementById('diff-result');

  if (isCross) {
    // 4832 - 7998 + 86400 = 83234 秒 (23:07:14)
    calculatedDiffSec = (4832 - 7998) + 86400;
    resultDiv.innerHTML = `相差秒數：4832 - 7998 + 86400 (跨日補償) = <strong>${calculatedDiffSec}</strong> 秒`;
  } else {
    // 76231 - 40400 = 35831 秒 (09:57:11)
    calculatedDiffSec = 76231 - 40400;
    resultDiv.innerHTML = `相差秒數：76231 - 40400 = <strong>${calculatedDiffSec}</strong> 秒`;
  }
}

function restoreTime() {
  const hh = Math.floor(calculatedDiffSec / 3600);
  const mm = Math.floor((calculatedDiffSec % 3600) / 60);
  const ss = calculatedDiffSec % 60;

  const pad = (num) => String(num).padStart(2, '0');
  
  const clock = document.getElementById('final-clock');
  clock.innerText = `${pad(hh)}:${pad(mm)}:${pad(ss)}`;
}
```
