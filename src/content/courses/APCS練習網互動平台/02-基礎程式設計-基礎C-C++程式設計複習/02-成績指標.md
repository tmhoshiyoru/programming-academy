```game-js
let currentScores = [];
let currentStep = 0;

const examples = {
  1: { count: 10, scores: [0, 11, 22, 33, 55, 66, 77, 99, 88, 44] },
  2: { count: 1, scores: [13] },
  3: { count: 2, scores: [73, 65] }
};

function init() {
  loadExample(1);
}

function loadExample(id) {
  const ex = examples[id];
  document.getElementById('numInput').value = ex.count;
  document.getElementById('scoresInput').value = ex.scores.join(' ');
  applyCustomInput();
}

function applyCustomInput() {
  const num = parseInt(document.getElementById('numInput').value);
  const rawScores = document.getElementById('scoresInput').value.trim().split(/\s+/).map(Number);

  if (isNaN(num) || num < 1 || num > 20) {
    alert('學生人數請輸入 1~20 之間的整數');
    return;
  }

  if (rawScores.length !== num || rawScores.some(s => isNaN(s) || s < 0 || s > 100)) {
    alert(`請輸入正好 ${num} 個 0~100 之間的整數成績（以空白分隔）`);
    return;
  }

  currentScores = [...rawScores];
  resetSim();
}

function resetSim() {
  currentStep = 0;
  document.getElementById('passLine').style.display = 'none';
  document.getElementById('statusMsg').innerText = '點擊「步驟 1：分數排序」開始觀察解題過程！';
  
  const btn = document.getElementById('stepBtn');
  btn.innerText = '步驟 1：分數排序';
  btn.disabled = false;
  btn.style.opacity = '1';

  renderCards(currentScores);
  updateConsole('等待步驟執行...');
}

function renderCards(scores, highlightInfo = {}) {
  const container = document.getElementById('cardsContainer');
  container.innerHTML = '';

  scores.forEach((score, idx) => {
    const card = document.createElement('div');
    card.className = 'score-card';
    card.innerText = score;

    if (currentStep >= 2) {
      if (score < 60) card.classList.add('fail');
      else card.classList.add('pass');
    }

    if (highlightInfo.maxFailIdx === idx) {
      card.classList.add('highlight-fail');
      const tag = document.createElement('div');
      tag.className = 'tag fail-tag';
      tag.innerText = '最高不及格';
      card.appendChild(tag);
    }

    if (highlightInfo.minPassIdx === idx) {
      card.classList.add('highlight-pass');
      const tag = document.createElement('div');
      tag.className = 'tag pass-tag';
      tag.innerText = '最低及格';
      card.appendChild(tag);
    }

    container.appendChild(card);
  });
}

function nextStep() {
  if (currentStep === 0) {
    // Step 1: Sort
    currentScores.sort((a, b) => a - b);
    renderCards(currentScores);
    document.getElementById('statusMsg').innerText = '第一步完成：將所有成績從小到大進行排序！';
    document.getElementById('stepBtn').innerText = '步驟 2：劃分及格界線';
    currentStep = 1;

  } else if (currentStep === 1) {
    // Step 2: Classify Pass/Fail
    renderCards(currentScores);
    showPassLine();
    document.getElementById('statusMsg').innerText = '第二步完成：以 60 分為基準劃分不及格區與及格區！';
    document.getElementById('stepBtn').innerText = '步驟 3：鎖定指標與邊界條件';
    currentStep = 2;

  } else if (currentStep === 2) {
    // Step 3: Find indicators
    const fails = currentScores.filter(s => s < 60);
    const passes = currentScores.filter(s => s >= 60);

    const maxFail = fails.length > 0 ? Math.max(...fails) : null;
    const minPass = passes.length > 0 ? Math.min(...passes) : null;

    const maxFailIdx = maxFail !== null ? currentScores.lastIndexOf(maxFail) : -1;
    const minPassIdx = minPass !== null ? currentScores.indexOf(minPass) : -1;

    renderCards(currentScores, { maxFailIdx, minPassIdx });

    let line1 = currentScores.join(' ');
    let line2 = maxFail !== null ? maxFail : 'best case';
    let line3 = minPass !== null ? minPass : 'worst case';

    updateConsole(`${line1}\n${line2}\n${line3}`);
    document.getElementById('statusMsg').innerText = '第三步完成：找出最高不及格與最低及格分數！';

    const btn = document.getElementById('stepBtn');
    btn.innerText = '流程結束';
    btn.disabled = true;
    btn.style.opacity = '0.6';
    currentStep = 3;
  }
}

function showPassLine() {
  const container = document.getElementById('cardsContainer');
  const passLine = document.getElementById('passLine');
  const cards = container.children;

  let splitIdx = currentScores.findIndex(s => s >= 60);

  passLine.style.display = 'block';

  if (splitIdx === -1) {
    // All fail -> Line on the far right
    const lastCard = cards[cards.length - 1];
    passLine.style.left = `${lastCard.offsetLeft + lastCard.offsetWidth + 5}px`;
  } else if (splitIdx === 0) {
    // All pass -> Line on the far left
    const firstCard = cards[0];
    passLine.style.left = `${firstCard.offsetLeft - 5}px`;
  } else {
    // Middle
    const prevCard = cards[splitIdx - 1];
    const nextCard = cards[splitIdx];
    const leftPos = (prevCard.offsetLeft + prevCard.offsetWidth + nextCard.offsetLeft) / 2;
    passLine.style.left = `${leftPos}px`;
  }
}

function updateConsole(text) {
  document.getElementById('outputConsole').innerText = text;
}

window.onload = init;
```
