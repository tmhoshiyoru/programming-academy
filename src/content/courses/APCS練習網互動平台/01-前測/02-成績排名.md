```game-js
const inputEl = document.getElementById('custom-input');
const btnApply = document.getElementById('btn-apply');
const btnS1 = document.getElementById('btn-s1');
const btnS2 = document.getElementById('btn-s2');
const errorEl = document.getElementById('error-msg');

const elOriginal = document.getElementById('cards-original');
const elUnique = document.getElementById('cards-unique');
const elSorted = document.getElementById('cards-sorted');
const elRanked = document.getElementById('cards-ranked');
const elFinal = document.getElementById('cards-final');
const elOutput = document.getElementById('output-text');

function parseInput() {
  errorEl.style.display = 'none';
  const rawText = inputEl.value.trim();
  
  if (!rawText) {
    showError('請輸入成績資料！');
    return null;
  }

  const parts = rawText.split(/\s+/);
  const nums = [];

  for (let p of parts) {
    const val = Number(p);
    if (isNaN(val) || !Number.isInteger(val) || val < 0 || val > 100) {
      showError('請輸入 0 ~ 100 之間的整數成績，並用空白隔開。');
      return null;
    }
    nums.push(val);
  }

  if (nums.length === 0) {
    showError('請至少輸入一位學生的成績。');
    return null;
  }

  if (nums.length > 100) {
    showError('學生人數限制最多 100 人。');
    return null;
  }

  return nums;
}

function showError(msg) {
  errorEl.textContent = msg;
  errorEl.style.display = 'block';
}

function updateVisuals() {
  const scores = parseInput();
  if (!scores) return;

  // 1. 原始資料
  elOriginal.innerHTML = '';
  scores.forEach((s, idx) => {
    elOriginal.appendChild(createCard(`s${idx + 1}`, s));
  });

  // 2. 去重
  const uniqueScores = Array.from(new Set(scores));
  elUnique.innerHTML = '';
  uniqueScores.forEach(s => {
    elUnique.appendChild(createCard('不重複分數', s));
  });

  // 3. 排序 (大 -> 小)
  const sortedScores = [...uniqueScores].sort((a, b) => b - a);
  elSorted.innerHTML = '';
  sortedScores.forEach(s => {
    elSorted.appendChild(createCard('降序排列', s));
  });

  // 4. 賦予名次
  const rankMap = {};
  elRanked.innerHTML = '';
  sortedScores.forEach((s, idx) => {
    const rank = idx + 1;
    rankMap[s] = rank;
    elRanked.appendChild(createRankCard(s, rank));
  });

  // 5. 查表還原
  elFinal.innerHTML = '';
  const finalRanks = scores.map(s => rankMap[s]);
  scores.forEach((s, idx) => {
    const r = rankMap[s];
    elFinal.appendChild(createCard(`s${idx + 1} (${s}分)`, `第 ${r} 名`, true));
  });

  // 輸出結果
  elOutput.textContent = finalRanks.join(' ');
}

function createCard(sub, val, isRank = false) {
  const div = document.createElement('div');
  div.className = 'node-card';
  div.innerHTML = `
    <div class="sub">${sub}</div>
    <div class="${isRank ? 'rank-val' : 'val'}">${val}</div>
  `;
  return div;
}

function createRankCard(score, rank) {
  const div = document.createElement('div');
  div.className = 'node-card';
  div.innerHTML = `
    <div class="sub">${score} 分</div>
    <div class="rank-val">第 ${rank} 名</div>
  `;
  return div;
}

btnApply.addEventListener('click', updateVisuals);
btnS1.addEventListener('click', () => {
  inputEl.value = '90 91 88 68 76';
  updateVisuals();
});
btnS2.addEventListener('click', () => {
  inputEl.value = '89 89 93 88';
  updateVisuals();
});

// 初始化執行
updateVisuals();
```
