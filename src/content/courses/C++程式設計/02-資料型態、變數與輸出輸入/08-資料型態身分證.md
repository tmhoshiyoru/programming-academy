function asciiName(n) { return ['NUL','SOH','STX','ETX','EOT','ENQ','ACK','BEL','BS','TAB','LF','VT','FF','CR','SO','SI','DLE','DC1','DC2','DC3','DC4','NAK','SYN','ETB','CAN','EM','SUB','ESC','FS','GS','RS','US'][n] || (n===127?'DEL':''); }
function asciiText(n) { if(n===32) return '空格'; if(n<32 || n===127) return asciiName(n); return String.fromCharCode(n); }
function selectAscii(n) {
  n = Math.max(0, Math.min(127, Number(n)||0));
  const glyph = n===32 ? '␠' : n<32 || n===127 ? '⌁' : String.fromCharCode(n);
  document.querySelectorAll('.ascii-cell').forEach(b => b.classList.toggle('selected', Number(b.dataset.code)===n));
  const escapes = {0:'\\0',7:'\\a',8:'\\b',9:'\\t',10:'\\n',11:'\\v',12:'\\f',13:'\\r',27:'\\x1B',34:'\\"',39:"\\'",92:'\\\\'};
  const charLiteral = escapes[n] ? `'${escapes[n]}'` : n >= 32 && n <= 126 ? `'${String.fromCharCode(n)}'` : '無一般字元常數';
  const kind = n === 32 ? '空格' : n < 32 || n === 127 ? '控制碼' : '可視字元';
  const escapeInfo = {0:['\\0','空字元'],7:['\\a','提示鈴音控制碼'],8:['\\b','退格'],9:['\\t','水平定位（Tab）'],10:['\\n','換行（LF）'],11:['\\v','垂直定位'],12:['\\f','換頁'],13:['\\r','回到行首（CR）'],27:['\\x1B','跳脫控制碼（ESC）'],34:['\\"','雙引號'],39:["\\'",'單引號'],92:['\\\\','反斜線']};
  const escapeDescription = escapeInfo[n] ? `<br>跳脫寫法：<code>${safe(escapeInfo[n][0])}</code> · ${safe(escapeInfo[n][1])}` : '';
  document.querySelector('#ascii-detail').innerHTML = `<span class="ascii-glyph">${safe(glyph)}</span><p><b>${safe(kind)} · ${safe(asciiText(n))}</b><br>十進位 ${n} · 十六進位 0x${n.toString(16).toUpperCase().padStart(2,'0')} · 二進位 ${n.toString(2).padStart(8,'0')}<br>C++ 字元常數：<code>${safe(charLiteral)}</code>${escapeDescription}</p>`;
  document.querySelector('#ascii-search').value = n;
}

  grid.addEventListener('click', e => { const cell=e.target.closest('.ascii-cell'); if(cell) selectAscii(cell.dataset.code); });
  document.querySelector('#ascii-find').addEventListener('click',()=>selectAscii(document.querySelector('#ascii-search').value));
  document.querySelector('#ascii-search').addEventListener('keydown',e=>{if(e.key==='Enter')selectAscii(e.currentTarget.value);});
  selectAscii(65);
}
function runDemo(type) {
  const value = document.querySelector('#demo-input').value;
  let output = value;
  let isError = false;
  if(type.id==='bool') output = value==='1' ? '1（true）' : '0（false）';
  if(type.id==='char') output = Array.from(value)[0] || '（空字元輸入）';
  if(['float','double'].includes(type.id)) { const n=Number(value); output = Number.isFinite(n) ? (type.id==='float'?Math.fround(n):n).toString() : '請輸入有效數字'; }
  if(['int','short','longlong'].includes(type.id)) {
    const result = /^[-+]?\d+$/.test(value) ? BigInt(value) : null;
    if (result === null) { output = '輸入錯誤：請輸入整數。'; isError = true; }
    else if (type.limits && (result < BigInt(type.limits.min) || result > BigInt(type.limits.max))) { output = `輸入失敗：${value} 超出常見 ${type.name} 範圍（${type.range}），這個值無法存入變數。`; isError = true; }
    else output = value;
  }
  const demoOutput = document.querySelector('#demo-output');
  if (type.id === 'string') {
    const line = value.trimStart();
    const firstWord = line.split(/\s+/, 1)[0] || '';
    const remaining = line.slice(firstWord.length).trimStart();
    demoOutput.textContent = firstWord
      ? `輸入資料：${value}\nname 讀入：${firstWord}\ncout 輸出：${firstWord}${remaining ? `\n尚留在輸入串流：${remaining}` : ''}`
      : '沒有可讀取的字串。';
    return;
  }
  demoOutput.textContent = isError ? output : `cout 輸出：${output}`;
}
