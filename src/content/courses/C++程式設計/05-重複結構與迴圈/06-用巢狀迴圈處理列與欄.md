## 一、一列做完，再換下一列

要輸出兩列星號、每列三個，可以讓外層迴圈管理「目前第幾列」，內層迴圈管理「這一列的第幾個」。這就是**巢狀迴圈**。

```cpp
#include <iostream>
using namespace std;

int main() {
    int rows = 2;
    int columns = 3;
    for (int row = 1; row <= rows; row++) {
        for (int column = 1; column <= columns; column++) {
            cout << "*";
        }
        cout << endl;
    }
    return 0;
}
```

輸出結果：

```text
***
***
```

外層每做一輪，內層就會從頭跑完一次。本例的星號輸出指令共執行 2 × 3 = 6 次。

## 二、換行放在哪裡？

內層的 `cout << "*";` 每次只輸出星號，保持在同一行。內層完成後，`cout << endl;` 才替整列換行。

把 endl 放進內層，會變成每個星號各佔一行；把 endl 放到外層迴圈外面，所有列的星號則會接在一起。指令位於哪一層，決定它多久執行一次。

## 三、內外層各有自己的變數

`row` 表示列，`column` 表示欄。每次外層進入新的一列，會重新執行內層的 `int column = 1`。

如果在內層使用 break，會結束這次的內層迴圈，接著執行該列後面的換行；外層仍會依自己的條件決定是否繼續。

```game-js
const patterns = {
  increasing: { label: '遞增（靠左）', rule: 'j ≤ i', stars: (i, n) => i, matches: (i, j, n) => j <= i },
  decreasing: { label: '遞減（靠左）', rule: 'j ≤ n − i + 1', stars: (i, n) => n - i + 1, matches: (i, j, n) => j <= n - i + 1 },
  'right-increasing': { label: '反向遞增（靠右）', rule: 'j ≥ n − i + 1', stars: (i, n) => i, matches: (i, j, n) => j >= n - i + 1 },
  'right-decreasing': { label: '反向遞減（靠右）', rule: 'j ≥ i', stars: (i, n) => n - i + 1, matches: (i, j, n) => j >= i },
};
const tabs = [...document.querySelectorAll('[role="tab"]')];
const sizeField = document.querySelector('#size');
const matrix = document.querySelector('#matrix');
const playButton = document.querySelector('#play');
let n = 5, selected = 'increasing', cursor = 0, timer = null;
let cells = [], rowHeaders = [], columnHeaders = [];
function stop() {
  if (timer !== null) clearInterval(timer);
  timer = null;
  playButton.textContent = '自動走訪';
  playButton.setAttribute('aria-pressed', 'false');
}
function render() {
  const pattern = patterns[selected];
  const i = Math.floor(cursor / n) + 1, j = cursor % n + 1;
  cells.forEach((cell, index) => {
    const active = index === cursor;
    if (active) cell.setAttribute('aria-current', 'step'); else cell.removeAttribute('aria-current');
    cell.tabIndex = active ? 0 : -1;
  });
  rowHeaders.forEach((header, index) => header.classList.toggle('current-axis', index + 1 === i));
  columnHeaders.forEach((header, index) => header.classList.toggle('current-axis', index + 1 === j));
  document.querySelector('#position').textContent = 'i = ' + i + '，j = ' + j + '（' + (cursor + 1) + ' / ' + n * n + ' 格）';
  const star = pattern.matches(i, j, n);
  document.querySelector('#action').textContent = star ? '這一格輸出星號 *' : '這一格輸出一個空格';
  document.querySelector('#condition').textContent = '星號範圍：' + pattern.rule + '。目前這格' + (star ? '符合。' : '在範圍外。');
  const stars = pattern.stars(i, n), spaces = n - stars;
  document.querySelector('#row-summary').textContent = '第 ' + i + ' 列：' + stars + ' 個星號，' + spaces + ' 個空格。';
  document.querySelector('#movement').textContent = cursor === n * n - 1 ? '這是最後一格。本列完成後換行，整張方格走訪結束。' : j === 1 && i > 1 ? '上一列完成並換行；i 增加 1，j 重新回到 1。接著向右走。' : j === n ? '本列已到最後一欄。下一步先換行，i 增加 1，j 回到 1。' : 'i 保持第 ' + i + ' 列；下一步 j 增加 1，往右一格。';
  document.querySelector('#previous').disabled = cursor === 0;
  document.querySelector('#next').disabled = cursor === n * n - 1;
  document.querySelector('#space-note').textContent = selected.startsWith('right-') ? '靠右的圖形需要先輸出左側空格，再輸出星號，才能對齊最右欄。' : '這個 n × n 示範也走過右側空格；實際輸出靠左三角形時，右側空格可以省略，星號後直接換行。';
}
function moveTo(position, focus = false) {
  cursor = Math.max(0, Math.min(n * n - 1, position));
  render();
  if (focus) cells[cursor].focus();
}
function rebuild() {
  stop(); cursor = 0; cells = []; rowHeaders = []; columnHeaders = []; matrix.replaceChildren();
  matrix.setAttribute('aria-label', patterns[selected].label + '，' + n + ' 列 ' + n + ' 欄');
  tabs.forEach(tab => { const active = tab.dataset.pattern === selected; tab.setAttribute('aria-selected', String(active)); tab.tabIndex = active ? 0 : -1; });
  document.querySelector('#pattern-panel').setAttribute('aria-labelledby', 'tab-' + selected);
  const head = document.createElement('thead'), headingRow = document.createElement('tr'), corner = document.createElement('th');
  corner.className = 'corner'; corner.textContent = 'i ↓ / j →'; headingRow.append(corner);
  for (let j = 1; j <= n; j++) { const th = document.createElement('th'); th.scope = 'col'; th.textContent = String(j); columnHeaders.push(th); headingRow.append(th); }
  head.append(headingRow); matrix.append(head);
  const body = document.createElement('tbody');
  for (let i = 1; i <= n; i++) {
    const row = document.createElement('tr'), th = document.createElement('th'); th.scope = 'row'; th.textContent = String(i); rowHeaders.push(th); row.append(th);
    for (let j = 1; j <= n; j++) {
      const td = document.createElement('td'), button = document.createElement('button');
      const star = patterns[selected].matches(i, j, n), index = (i - 1) * n + j - 1;
      button.type = 'button'; button.className = 'cell ' + (star ? 'star' : 'space'); button.textContent = star ? '*' : ' ';
      button.dataset.i = String(i); button.dataset.j = String(j); button.setAttribute('aria-label', '第 ' + i + ' 列、第 ' + j + ' 欄，輸出' + (star ? '星號' : '空格'));
      button.addEventListener('click', () => { stop(); moveTo(index); });
      button.addEventListener('keydown', event => {
        let target = index;
        if (event.key === 'ArrowRight' && j < n) target++;
        else if (event.key === 'ArrowLeft' && j > 1) target--;
        else if (event.key === 'ArrowDown' && i < n) target += n;
        else if (event.key === 'ArrowUp' && i > 1) target -= n;
        else if (event.key === 'Home') target = (i - 1) * n;
        else if (event.key === 'End') target = i * n - 1;
        else return;
        event.preventDefault(); stop(); moveTo(target, true);
      });
      cells.push(button); td.append(button); row.append(td);
    }
    body.append(row);
  }
  matrix.append(body); render();
}
tabs.forEach((tab, index) => {
  tab.addEventListener('click', () => { selected = tab.dataset.pattern; rebuild(); });
  tab.addEventListener('keydown', event => {
    let target;
    if (event.key === 'ArrowRight') target = (index + 1) % tabs.length;
    else if (event.key === 'ArrowLeft') target = (index + tabs.length - 1) % tabs.length;
    else if (event.key === 'Home') target = 0;
    else if (event.key === 'End') target = tabs.length - 1;
    else return;
    event.preventDefault(); selected = tabs[target].dataset.pattern; rebuild(); tabs[target].focus();
  });
});
sizeField.addEventListener('change', () => { n = Number(sizeField.value); rebuild(); });
document.querySelector('#previous').addEventListener('click', () => { stop(); moveTo(cursor - 1); });
document.querySelector('#next').addEventListener('click', () => { stop(); moveTo(cursor + 1); });
document.querySelector('#reset').addEventListener('click', () => { stop(); moveTo(0); });
playButton.addEventListener('click', () => {
  if (timer !== null) { stop(); return; }
  if (n === 1) { moveTo(0); return; }
  moveTo(cursor === n * n - 1 ? 0 : cursor + 1);
  if (cursor === n * n - 1) return;
  playButton.textContent = '暫停走訪'; playButton.setAttribute('aria-pressed', 'true');
  timer = setInterval(() => { moveTo(cursor + 1); if (cursor === n * n - 1) stop(); }, 650);
});
window.addEventListener('pagehide', stop);
rebuild();
```
