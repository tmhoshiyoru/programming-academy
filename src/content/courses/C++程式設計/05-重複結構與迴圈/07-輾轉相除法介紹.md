(() => {
  const $ = (id) => document.getElementById(id);

  const zoomValue = $('zoom-value');
  let steps = [];
  let shownSteps = 0;
  let reverseRows = [];
  let reverseIndex = 0;
  let originalA = 252;
  let originalB = 198;
  let timer = null;

  const fmt = (n) => Number(n).toLocaleString('zh-Hant');
  const gcd = (a, b) => { while (b) [a, b] = [b, a % b]; return a; };

  function makeSteps(a, b) {
    const rows = [];
    let long = Math.max(a, b);
    let short = Math.min(a, b);
    while (short) {
      const quotient = Math.floor(long / short);
      const remainder = long % short;
      rows.push({ long, short, quotient, remainder });
      if (!remainder) break;
      [long, short] = [short, remainder];
    }
    return rows;
  }

  function cameraViewBox() {
    const width = 1000 / (Number(zoom.value) / 100);
    return `${(1000 - width) / 2} 0 ${width} 68`;
  }

  function barSvg(value, maxValue, mode, unit, label, isNew = false) {
    const left = 8;
    const fullWidth = 984;
    const width = Math.max(1, fullWidth * value / maxValue);
    const pieces = [];
    let aria = `${label}，長度 ${fmt(value)}`;
    if (mode === 'whole') {
      pieces.push(`<rect x="${left}" y="30" width="${width}" height="22" rx="7" class="rod-unit"/>`);
    } else {
      const count = Math.round(value / unit);
      const shown = Math.min(count, 18);
      for (let i = 0; i < shown; i += 1) {
        const start = left + width * i / count;
        const end = left + width * (i + 1) / count;
        const klass = mode === 'division' && i >= Math.floor((value - (value % unit)) / unit) ? 'rod-remainder' : (mode === 'reverse' && i === 0 ? 'rod-minimum' : 'rod-unit');
        pieces.push(`<rect x="${start + 1}" y="30" width="${Math.max(1, end - start - 2)}" height="22" rx="5" class="${klass}"/>`);
        if (end - start > 38) pieces.push(`<text x="${(start + end) / 2}" y="45" class="${klass === 'rod-minimum' ? 'minimum-label' : 'segment-label'}">${count > 18 ? '' : fmt(unit)}</text>`);
      }
      if (count > 18) pieces.push(`<text x="${left + width / 2}" y="45" class="segment-label">${fmt(count)} 個 × ${fmt(unit)}${mode === 'division' && value % unit ? `，餘 ${fmt(value % unit)}` : ''}</text>`);
      aria += `，切成 ${fmt(count)} 個長度 ${fmt(unit)} 的小段`;
    }
    const labelText = `${label}　${fmt(value)}`;
    return `<div class="rod-row${mode === 'whole' ? ' is-anchor' : mode === 'reverse' ? ' is-reverse' : ''}${isNew ? ' is-new' : ''}"><svg viewBox="${cameraViewBox()}" role="img" aria-label="${aria}"><text x="8" y="19" class="rod-label">${labelText}</text>${pieces.join('')}</svg></div>`;
  }

  function divisionSvg(step, index) {
    const width = 984 * step.long / maxScale();
    const unitWidth = width / step.long * step.short;
    const remWidth = width / step.long * step.remainder;
    const elems = [];
    const count = step.quotient > 18 ? 0 : step.quotient;
    for (let i = 0; i < count; i += 1) {
      const start = 8 + unitWidth * i;
      const end = 8 + unitWidth * (i + 1);
      elems.push(`<rect x="${start + 1}" y="30" width="${Math.max(1, end - start - 2)}" height="22" rx="5" class="rod-unit"/>`);
      if (step.quotient <= 18 && end - start > 34) elems.push(`<text x="${(start + end) / 2}" y="45" class="segment-label">${fmt(step.short)}</text>`);
    }
    if (step.quotient > 18) {
      elems.push(`<rect x="9" y="30" width="${Math.max(1, unitWidth * step.quotient - 2)}" height="22" rx="5" class="rod-unit"/>`);
      elems.push(`<text x="${8 + unitWidth * step.quotient / 2}" y="45" class="segment-label">${fmt(step.quotient)} 段 × ${fmt(step.short)}</text>`);
    }
    if (step.remainder) {
      const remX = 8 + unitWidth * step.quotient;
      elems.push(`<rect x="${remX + 1}" y="30" width="${Math.max(2, remWidth - 2)}" height="22" rx="5" class="rod-remainder"/>`);
      if (remWidth > 42) elems.push(`<text x="${remX + remWidth / 2}" y="45" class="minimum-label">${fmt(step.remainder)}</text>`);
    }
    const formula = `${fmt(step.long)} = ${fmt(step.quotient)} × ${fmt(step.short)}${step.remainder ? ` + ${fmt(step.remainder)}` : ''}`;
    const label = `第 ${index + 1} 步　${formula}`;
    const aria = `${label}。綠色表示每段 ${fmt(step.short)}，橘色表示剩餘 ${fmt(step.remainder)}。`;
    return `<div class="rod-row is-new"><svg viewBox="${cameraViewBox()}" role="img" aria-label="${aria}"><text x="8" y="19" class="rod-label">${label}</text>${elems.join('')}</svg></div>`;
  }
  function maxScale() { return Math.max(originalA, originalB); }

  function reverseSvg(value, gcdValue, isNew) {
    const width = 984 * value / maxScale();
    const count = value / gcdValue;
    const shown = count > 18 ? 0 : count;
    const unitWidth = width / count;
    const elems = [];
    for (let i = 0; i < shown; i += 1) {
      const start = 8 + unitWidth * i;
      const end = 8 + unitWidth * (i + 1);
      elems.push(`<rect x="${start + 1}" y="30" width="${Math.max(1, end - start - 2)}" height="22" rx="5" class="${i === 0 ? 'rod-minimum' : 'rod-unit'}"/>`);
      if (end - start > 40 && count <= 18) elems.push(`<text x="${(start + end) / 2}" y="45" class="${i === 0 ? 'minimum-label' : 'segment-label'}">${fmt(gcdValue)}</text>`);
    }
    if (count > 18) {
      elems.push(`<rect x="9" y="30" width="${Math.max(1, width - 2)}" height="22" rx="5" class="rod-unit"/>`);
      elems.push(`<text x="${8 + width / 2}" y="45" class="segment-label">${fmt(count)} 個最小單位 × ${fmt(gcdValue)}</text>`);
    }
    return `<div class="rod-row is-reverse${isNew ? ' is-new' : ''}"><svg viewBox="${cameraViewBox()}" role="img" aria-label="長度 ${fmt(value)} 的棍子切成 ${fmt(count)} 個長度 ${fmt(gcdValue)} 的最小單位。"><text x="8" y="19" class="rod-label">反推　${fmt(value)}　＝　${fmt(count)} 個最小單位 × ${fmt(gcdValue)}</text>${elems.join('')}</svg></div>`;
  }

  function reverseValues() {
    const g = gcd(originalA, originalB);
    return [...new Set([originalA, originalB, ...steps.flatMap(s => [s.long, s.short])])]
      .filter(v => v >= g).sort((a, b) => a - b);
  }

  function drawStack(newest = false) {
    const previousTop = viewport.scrollTop;
    const html = [
      barSvg(originalA, maxScale(), 'whole', 0, 'A 棍'),
      barSvg(originalB, maxScale(), 'whole', 0, 'B 棍')
    ];
    for (let i = 0; i < shownSteps; i += 1) html.push(divisionSvg(steps[i], i));
    for (let i = 0; i < reverseIndex; i += 1) html.push(reverseSvg(reverseRows[i], gcd(originalA, originalB), true));
    stack.innerHTML = html.join('');
    if (newest) viewport.scrollTop = viewport.scrollHeight;
    else viewport.scrollTop = previousTop;
  }

  function update() {
    const finished = shownSteps === steps.length && steps.length > 0;
    const reversed = reverseIndex > 0;
    const step = shownSteps ? steps[shownSteps - 1] : null;
    title.textContent = reversed
      ? `反推最小單位：第 ${reverseIndex} / ${reverseRows.length} 列`
      : step ? `切分進度：第 ${shownSteps} / ${steps.length} 步` : '準備開始';
    equation.textContent = reversed
      ? `${fmt(reverseRows[reverseIndex - 1])} = ${fmt(reverseRows[reverseIndex - 1] / gcd(originalA, originalB))} × ${fmt(gcd(originalA, originalB))}`
      : step ? `${fmt(step.long)} = ${fmt(step.quotient)} × ${fmt(step.short)} + ${fmt(step.remainder)}`
      : '先看 A、B 兩根棍子，再逐步切分。';
    if (reversed) caption.textContent = reverseIndex === reverseRows.length
      ? '每根棍子都由相同長度的最小單位組成。'
      : `把長度 ${fmt(reverseRows[reverseIndex - 1])} 的棍子切成相同的小段；再往下看更長的棍子。`;
    else if (step) caption.textContent = step.remainder === 0
      ? `${fmt(step.long)} 剛好切成 ${fmt(step.quotient)} 段，每段長 ${fmt(step.short)}，沒有剩餘。`
      : `把長度 ${fmt(step.long)} 的棍子切成 ${fmt(step.quotient)} 段、每段長 ${fmt(step.short)}，還剩 ${fmt(step.remainder)}；下一步比較 ${fmt(step.short)} 和 ${fmt(step.remainder)}。`;
    else caption.textContent = '按「下一步」逐列增加切分結果。';
    drawStack(false);
    prev.disabled = shownSteps === 0 || reverseIndex > 0;
    next.disabled = !steps.length || shownSteps >= steps.length || reverseIndex > 0;
    play.disabled = !steps.length || shownSteps >= steps.length || reverseIndex > 0;
    play.textContent = timer ? '暫停' : '播放';
    backtrack.disabled = !finished || reverseIndex >= reverseRows.length;
    backtrack.textContent = reverseIndex ? (reverseIndex >= reverseRows.length ? '反推完成' : '反推下一列') : '反推最小單位';
    result.classList.toggle('is-finished', finished);
    result.innerHTML = finished ? `最小共同片段是 <strong>${fmt(gcd(originalA, originalB))}</strong>，也就是 ${fmt(originalA)} 和 ${fmt(originalB)} 的最大公因數。` : '';
  }

  function calculate() {
    const a = Number(aInput.value);
    const b = Number(bInput.value);
    if (!Number.isInteger(a) || !Number.isInteger(b) || a < 1 || b < 1 || a > 999999 || b > 999999) {
      message.textContent = '請輸入 1～999999 之間的正整數。';
      message.classList.add('is-error');
      return;
    }
    stopPlayback();
    originalA = a;
    originalB = b;
    steps = makeSteps(a, b);
    shownSteps = 0;
    reverseRows = reverseValues();
    reverseIndex = 0;
    viewport.scrollTop = 0;
    message.textContent = `已準備好：${fmt(a)} 和 ${fmt(b)}。`;
    message.classList.remove('is-error');
    update();
  }

  function stopPlayback() {
    if (timer) clearInterval(timer);
    timer = null;
    if (steps.length) play.textContent = '播放';
  }
  function advance() {
    if (shownSteps < steps.length) {
      shownSteps += 1;
      update();
      drawStack(true);
    }
  }

  $('load-example').addEventListener('click', () => { aInput.value = '252'; bInput.value = '198'; calculate(); });
  $('random-pair').addEventListener('click', () => {
    aInput.value = String(40 + Math.floor(Math.random() * 461));

    message.classList.remove('is-error');
  });
  $('calculate').addEventListener('click', calculate);

  next.addEventListener('click', () => { stopPlayback(); advance(); });
  play.addEventListener('click', () => {
    if (timer) { stopPlayback(); update(); return; }
    timer = setInterval(() => {
      if (shownSteps >= steps.length) { stopPlayback(); update(); }
      else advance();
    }, 1100);
    update();
  });
  backtrack.addEventListener('click', () => {
    stopPlayback();
    if (reverseIndex < reverseRows.length) reverseIndex += 1;
    update();
    drawStack(true);
  });
  zoom.addEventListener('input', () => {
    zoomValue.value = `${zoom.value}%`;
    zoomValue.textContent = `${zoom.value}%`;
    drawStack(false);
  });
  calculate();
})();

  function euclidRows(a, b) {
    let x = BigInt(a), y = BigInt(b);
    if (x < y) [x, y] = [y, x];
    const rows = [];
    while (y) {
      const q = x / y, r = x % y;
      rows.push({x, y, q, r});
      [x, y] = [y, r];
    }
    return rows;

  function pairCard(title, a, b, mode) {

      const quotientWidth = Number((y * q * 10000n) / x) / 100;
