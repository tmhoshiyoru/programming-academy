## 一、用一個名字管理一排資料

五位同學的分數可以放在五個變數裡，也可以用一個**陣列**整齊地存成一排。陣列中的每個位置叫作**元素**，這一排元素使用相同的資料型態。

```cpp
#include <iostream>
using namespace std;

int main() {
    int score[5] = {72, 85, 60, 91, 78};
    cout << score[1] << endl;
    score[1] = 90;
    cout << score[1] << endl;
    return 0;
}
```

輸出結果：

```text
85
90
```

`int score[5]` 建立五個 int 位置。大括號內的初值依順序放進去，`score[0]` 是 72，`score[1]` 是 85，依此類推。

## 二、索引從 0 開始

方括號中的編號叫作**索引**。長度為 5 的陣列，可用索引為 **0、1、2、3、4**。

- 宣告時的 `[5]`：表示建立五個元素。
- 使用時的 `score[2]`：表示取得第三個元素。
- `score[2] = 100;`：把第三個元素改成 100。

**不要把 `score[5]` 當成第五個元素。** 它已經超出這個陣列的範圍。C++ 的這種陣列不會自動替你擴充位置；存取範圍外的資料會使程式的行為失去保障。

## 三、初始值可以怎麼寫？

~~~cpp
int count[4] = {0};
int number[4] = {7, 8};
~~~

第一排的四個元素都是 0。第二排前兩個是 7、8，剩下的元素也會初始化為 0。先替資料設定初值，再開始使用，狀態會很清楚。

本章先練習長度直接寫在宣告裡的陣列，例如 `int score[5]`。之後再介紹如何依資料量選擇其他容器。

const mode=document.querySelector('#train-mode'),indexInput=document.querySelector('#train-index'),valueInput=document.querySelector('#train-value'),valueLabel=document.querySelector('#train-value-label');
const cars=document.querySelector('#train-cars'),track=document.querySelector('#train-track'),pointer=document.querySelector('#train-pointer'),code=document.querySelector('#train-code'),prompt=document.querySelector('#train-prompt'),range=document.querySelector('#train-range'),status=document.querySelector('#train-status'),runButton=document.querySelector('#train-run');
const nextButton=document.querySelector('#train-next'),duration=window.matchMedia('(prefers-reduced-motion: reduce)').matches?0:620;
function setBusy(value){busy=value;document.querySelectorAll('.train-controls input,.train-controls select,.train-controls button,.train-car').forEach(element=>{element.disabled=value;});}
function declaration(){return 'int score['+values.length+'] = {'+values.join(', ')+'};';}
function updatePrompt(){
  const current=mode.value;
  valueLabel.hidden=current!=='write';valueInput.hidden=current!=='write';
  if(current==='find')prompt.textContent='請找出內容 '+values[targetPosition]+' 位於第幾節，輸入節號後執行。';
  else if(current==='read')prompt.textContent='輸入節號，讀取該節的內容。也可以直接點選車廂。';
  else prompt.textContent='輸入節號與新整數，讓指定車廂改存新的內容。';
  range.textContent='目前共有 '+values.length+' 節：第0節到第'+(values.length-1)+'節。節號就是陣列索引，合法範圍為 0 到 n-1。';
  code.textContent=declaration();
}
function renderCars(){
  cars.replaceChildren();
  values.forEach((value,index)=>{
    const car=document.createElement('button');car.type='button';car.className='train-car'+(index===0?' lead-car':'');car.dataset.index=String(index);car.setAttribute('aria-pressed','false');car.setAttribute('aria-label','第'+index+'節，內容 '+value);
    const title=document.createElement('span');title.className='car-title';title.textContent='第'+index+'節'+(index===0?' · 車頭':'');
    const content=document.createElement('strong');content.className='car-value';content.textContent=String(value);car.append(title,content);
    car.addEventListener('click',()=>{if(busy)return;indexInput.value=String(index);if(mode.value==='write')moveTo(index,()=>{status.textContent='已選取第'+index+'節；輸入新整數後按「執行操作」。';});else perform(index);});
    cars.append(car);
  });
}
function moveTo(index,done){
  if(busy)return;setBusy(true);cars.querySelectorAll('.train-car').forEach(car=>car.setAttribute('aria-pressed','false'));
  const car=cars.querySelector('[data-index="'+index+'"]');
  if(car){car.scrollIntoView({behavior:duration?'smooth':'auto',block:'nearest',inline:'center'});pointer.style.left=(cars.offsetLeft+car.offsetLeft+car.offsetWidth/2)+'px';}
  else{const atStart=index<0;track.scrollTo({left:atStart?0:track.scrollWidth,behavior:duration?'smooth':'auto'});pointer.style.left=(atStart?6:cars.offsetLeft+cars.offsetWidth+10)+'px';}
  window.setTimeout(()=>{const selected=cars.querySelector('[data-index="'+index+'"]');if(selected)selected.setAttribute('aria-pressed','true');setBusy(false);done(selected);},duration);
}
function perform(index){
  if(!Number.isInteger(index)){status.textContent='請輸入整數節號。';return;}
  if(index<0||index>=values.length){moveTo(index,()=>{status.textContent='沒有第'+index+'節。這列只有第0節到第'+(values.length-1)+'節；游標停在軌道邊界，這就是越界存取。';code.textContent=declaration()+'\n// score['+index+'] 超出陣列範圍';});return;}
  if(mode.value==='write'&&(valueInput.value.trim()===''||!Number.isInteger(Number(valueInput.value)))){status.textContent='請輸入要存入的新整數。';return;}
  moveTo(index,()=>{
    if(mode.value==='find'){const target=values[targetPosition];code.textContent=declaration()+'\nint i = '+index+';\ncout << score[i] << endl;';status.textContent=values[index]===target?'答對了！內容 '+target+' 位於第'+index+'節，score['+index+'] 會讀到 '+values[index]+'。':'第'+index+'節的內容是 '+values[index]+'，不是 '+target+'；觀察車廂上顯示的節號再試一次。';}
    else if(mode.value==='read'){code.textContent=declaration()+'\ncout << score['+index+'] << endl;';status.textContent='讀取完成：score['+index+'] 的內容是 '+values[index]+'。';}
    else{const previous=values[index],next=Number(valueInput.value);values[index]=next;renderCars();cars.querySelector('[data-index="'+index+'"]').setAttribute('aria-pressed','true');pointer.style.left=(cars.offsetLeft+cars.querySelector('[data-index="'+index+'"]').offsetLeft+cars.querySelector('[data-index="'+index+'"]').offsetWidth/2)+'px';code.textContent=declaration()+'\nscore['+index+'] = '+next+';';status.textContent='修改完成：第'+index+'節的內容從 '+previous+' 改成 '+next+'。';}
  });
}
function loadTask(){const task=tasks[round%tasks.length];values=task.values.slice();targetPosition=task.targetPosition;indexInput.value='';valueInput.value='';renderCars();updatePrompt();const first=cars.firstElementChild;pointer.style.left=(cars.offsetLeft+first.offsetLeft+first.offsetWidth/2)+'px';status.textContent='點選車廂可以直接讀取內容，也可以在上方輸入節號操作。';}
mode.addEventListener('change',()=>{updatePrompt();status.textContent='選好操作模式，輸入節號，或直接點選車廂。';});
runButton.addEventListener('click',()=>{if(indexInput.value.trim()===''){status.textContent='請輸入節號；例如第0節就輸入 0。';return;}perform(Number(indexInput.value));});
nextButton.addEventListener('click',()=>{round++;loadTask();});
renderCars();loadTask();
